#### `src/hooks/useDraggable.ts`

```typescript
import { useState, useRef, useCallback } from 'react';
import { motion, PanInfo } from 'framer-motion';

interface Position {
  x: number;
  y: number;
}

interface TrailPoint extends Position {
  id: number;
}

interface UseDraggableReturn {
  position: Position;
  isDragging: boolean;
  trail: TrailPoint[];
  dragHandlers: any;
  handlePin: (pinned: boolean) => void;
  edgeSnap: (position: Position) => Position;
}

export const useDraggable = (
  initialPosition: Position,
  isPinned: boolean = false
): UseDraggableReturn => {
  const [position, setPosition] = useState<Position>(initialPosition);
  const [isDragging, setIsDragging] = useState(false);
  const [trail, setTrail] = useState<TrailPoint[]>([]);
  
  const trailIdRef = useRef(0);
  const containerRef = useRef<HTMLElement>(document.documentElement);

  // 边缘吸附功能
  const edgeSnap = useCallback((pos: Position): Position => {
    if (isPinned) return pos;
    
    const threshold = 50;
    const { clientWidth, clientHeight } = containerRef.current;
    
    let snappedPosition = { ...pos };
    
    // 左边缘吸附
    if (pos.x < threshold) {
      snappedPosition.x = 0;
    }
    // 右边缘吸附
    else if (pos.x > clientWidth - 450 - threshold) {
      snappedPosition.x = clientWidth - 450;
    }
    
    // 上边缘吸附
    if (pos.y < threshold) {
      snappedPosition.y = 0;
    }
    // 下边缘吸附
    else if (pos.y > clientHeight - 500 - threshold) {
      snappedPosition.y = clientHeight - 500;
    }
    
    return snappedPosition;
  }, [isPinned]);

  // 处理拖拽
  const handleDrag = useCallback((event: any, info: PanInfo) => {
    if (isPinned) return;
    
    const newPosition = {
      x: position.x + info.delta.x,
      y: position.y + info.delta.y
    };
    
    const snappedPosition = edgeSnap(newPosition);
    setPosition(snappedPosition);
    
    // 添加轨迹点
    const trailPoint: TrailPoint = {
      ...snappedPosition,
      id: trailIdRef.current++
    };
    
    setTrail(prev => [...prev.slice(-5), trailPoint]);
  }, [position, isPinned, edgeSnap]);

  // 拖拽开始
  const handleDragStart = useCallback(() => {
    setIsDragging(true);
    setTrail([]);
  }, []);

  // 拖拽结束
  const handleDragEnd = useCallback(() => {
    setIsDragging(false);
    // 清理轨迹
    setTimeout(() => setTrail([]), 500);
  }, []);

  // 处理固定
  const handlePin = useCallback((pinned: boolean) => {
    if (pinned) {
      // 固定时吸附到边缘
      const snappedPosition = edgeSnap(position);
      setPosition(snappedPosition);
    }
  }, [position, edgeSnap]);

  // 拖拽处理器
  const dragHandlers = {
    drag: true,
    dragMomentum: false,
    dragElastic: 0.1,
    onDrag: handleDrag,
    onDragStart: handleDragStart,
    onDragEnd: handleDragEnd,
  };

  return {
    position,
    isDragging,
    trail,
    dragHandlers,
    handlePin,
    edgeSnap
  };
};

import OpenAI from 'openai';
import * as tf from '@tensorflow/tfjs';
import { io, Socket } from 'socket.io-client';

export interface AIMessage {
  id: string;
  role: 'user' | 'assistant' | 'system';
  content: string;
  timestamp: Date;
  metadata?: any;
}

export interface AIResponse {
  message: string;
  confidence: number;
  processingTime: number;
  model: string;
  suggestions?: string[];
}

export interface VoiceConfig {
  language: string;
  accent: string;
  speed: number;
  pitch: number;
}

class AIService {
  private openai: OpenAI;
  private socket: Socket;
  private model: tf.LayersModel | null = null;
  private isInitialized = false;

  constructor() {
    this.openai = new OpenAI({
      apiKey: process.env.OPENAI_API_KEY,
      dangerouslyAllowBrowser: true
    });
    
    this.socket = io(process.env.AI_SERVICE_URL || 'http://localhost:3002');
    this.initializeModel();
  }

  // 初始化AI模型
  private async initializeModel() {
    try {
      // 加载本地模型
      this.model = await tf.loadLayersModel('/models/ai-assistant/model.json');
      this.isInitialized = true;
    } catch (error) {
      console.warn('Failed to load local model, using API only');
    }
  }

  // 文本对话
  async chat(message: string, context: AIMessage[] = []): Promise<AIResponse> {
    const startTime = Date.now();
    
    try {
      const messages = [
        {
          role: 'system' as const,
          content: '你是YYC³智能助手，专门服务于餐饮行业。请提供专业、准确、有用的回答。'
        },
        ...context.map(msg => ({
          role: msg.role,
          content: msg.content
        })),
        {
          role: 'user' as const,
          content: message
        }
      ];

      const completion = await this.openai.chat.completions.create({
        model: 'gpt-4',
        messages,
        temperature: 0.7,
        max_tokens: 1000,
        presence_penalty: 0.1,
        frequency_penalty: 0.1
      });

      const response = completion.choices[0]?.message?.content || '';
      const processingTime = Date.now() - startTime;

      return {
        message: response,
        confidence: 0.95,
        processingTime,
        model: 'gpt-4',
        suggestions: await this.generateSuggestions(response)
      };
    } catch (error) {
      console.error('Chat error:', error);
      throw new Error('AI服务暂时不可用');
    }
  }

  // 语音识别
  async transcribe(audioBlob: Blob): Promise<string> {
    const formData = new FormData();
    formData.append('audio', audioBlob);
    formData.append('language', 'zh-CN');

    try {
      const response = await fetch('/api/ai/transcribe', {
        method: 'POST',
        body: formData
      });

      const result = await response.json();
      return result.text;
    } catch (error) {
      console.error('Transcription error:', error);
      throw new Error('语音识别失败');
    }
  }

  // 语音合成
  async synthesize(text: string, config: VoiceConfig): Promise<Blob> {
    try {
      const response = await fetch('/api/ai/synthesize', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json'
        },
        body: JSON.stringify({ text, config })
      });

      return await response.blob();
    } catch (error) {
      console.error('Synthesis error:', error);
      throw new Error('语音合成失败');
    }
  }

  // 图像分析
  async analyzeImage(imageBlob: Blob): Promise<any> {
    try {
      // 使用本地模型进行图像分析
      if (this.model && this.isInitialized) {
        const imageElement = await this.blobToImage(imageBlob);
        const tensor = tf.browser.fromPixels(imageElement);
        const processed = tf.image.resizeBilinear(tensor, [224, 224]).expandDims(0);
        const prediction = this.model.predict(processed) as tf.Tensor;
        const result = await prediction.data();
        
        return {
          type: 'food_recognition',
          confidence: Math.max(...result),
          predictions: Array.from(result).map((prob, index) => ({
            label: this.getFoodLabel(index),
            confidence: prob
          }))
        };
      } else {
        // 使用API
        const formData = new FormData();
        formData.append('image', imageBlob);
        
        const response = await fetch('/api/ai/analyze-image', {
          method: 'POST',
          body: formData
        });
        
        return await response.json();
      }
    } catch (error) {
      console.error('Image analysis error:', error);
      throw new Error('图像分析失败');
    }
  }

  // 生成建议
  private async generateSuggestions(response: string): Promise<string[]> {
    const suggestionsPrompt = `
    基于以下回答，生成3个相关的后续问题建议：
    回答：${response}
    
    请以JSON数组格式返回建议：
    ["建议1", "建议2", "建议3"]
    `;

    try {
      const completion = await this.openai.chat.completions.create({
        model: 'gpt-3.5-turbo',
        messages: [
          {
            role: 'system',
            content: '你是一个专业的建议生成助手。'
          },
          {
            role: 'user',
            content: suggestionsPrompt
          }
        ],
        temperature: 0.5,
        max_tokens: 200
      });

      const suggestionsText = completion.choices[0]?.message?.content || '[]';
      return JSON.parse(suggestionsText);
    } catch (error) {
      console.error('Suggestions generation error:', error);
      return [];
    }
  }

  // 实时流式对话
  async streamChat(message: string, onChunk: (chunk: string) => void): Promise<void> {
    try {
      const response = await fetch('/api/ai/stream-chat', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json'
        },
        body: JSON.stringify({ message })
      });

      const reader = response.body?.getReader();
      const decoder = new TextDecoder();

      if (reader) {
        while (true) {
          const { done, value } = await reader.read();
          if (done) break;
          
          const chunk = decoder.decode(value);
          onChunk(chunk);
        }
      }
    } catch (error) {
      console.error('Stream chat error:', error);
      throw new Error('流式对话失败');
    }
  }

  // 知识库查询
  async queryKnowledge(query: string): Promise<any> {
    try {
      const response = await fetch('/api/ai/knowledge-query', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json'
        },
        body: JSON.stringify({ query })
      });

      return await response.json();
    } catch (error) {
      console.error('Knowledge query error:', error);
      throw new Error('知识库查询失败');
    }
  }

  // 工具函数
  private async blobToImage(blob: Blob): Promise<HTMLImageElement> {
    return new Promise((resolve, reject) => {
      const img = new Image();
      img.onload = () => resolve(img);
      img.onerror = reject;
      img.src = URL.createObjectURL(blob);
    });
  }

  private getFoodLabel(index: number): string {
    const labels = [
      '米饭', '面条', '饺子', '包子', '馒头',
      '炒菜', '汤', '烤肉', '海鲜', '蔬菜',
      '水果', '饮料', '甜点', '零食', '调料'
    ];
    return labels[index] || '未知';
  }

  // 获取服务状态
  getStatus() {
    return {
      isConnected: this.socket.connected,
      isModelLoaded: this.isInitialized,
      activeModel: this.isInitialized ? 'local' : 'openai'
    };
  }
}

export const aiService = new AIService();
