<script>
import { ref, computed, onMounted, onUnmounted, watch } from 'vue'
import { Howl } from 'howler'
import { gsap } from 'gsap'
import CosmicButton from './ui/CosmicButton.vue'
import CosmicProgressBar from './ui/CosmicProgressBar.vue'
import CosmicSlider from './ui/CosmicSlider.vue'

export default {
  name: 'CosmicPlayer',
  components: { CosmicButton, CosmicProgressBar, CosmicSlider },
  
  props: {
    track: Object,
    autoPlay: Boolean
  },
  
  emits: ['track-change', 'mheart-update'],
  
  setup(props, { emit }) {
    // 播放器状态
    const isPlaying = ref(false)
    const currentTime = ref(0)
    const duration = ref(0)
    const volume = ref(80)
    const isMuted = ref(false)
    const isShuffled = ref(false)
    const repeatMode = ref('none') // none, one, all
    const isExpanded = ref(false)
    
    // M❤️值系统
    const mHeartValue = ref(0)
    const mHeartLevel = ref(1)
    const mHeartProgress = ref(0)
    const showReward = ref(false)
    const rewardAmount = ref(0)
    
    // 音频相关
    let howl = null
    let animationId = null
    const stardustBg = ref(null)
    const visualizer = ref(null)
    const waveCanvas = ref(null)
    const particleField = ref(null)
    
    // 计算属性
    const currentTrack = computed(() => props.track)
    const progress = computed(() => (currentTime.value / duration.value) * 100)
    
    const volumeIcon = computed(() => {
      if (isMuted.value || volume.value === 0) return 'volume-off'
      if (volume.value < 50) return 'volume-low'
      return 'volume-high'
    })
    
    const repeatIcon = computed(() => {
      if (repeatMode.value === 'one') return 'repeat-one'
      if (repeatMode.value === 'all') return 'repeat'
      return 'repeat'
    })
    
    // 播放控制方法
    const initAudio = (track) => {
      if (howl) howl.unload()
      
      howl = new Howl({
        src: [track.url],
        html5: true,
        onload: () => {
          duration.value = howl.duration()
        },
        onplay: () => {
          isPlaying.value = true
          startVisualization()
        },
        onpause: () => {
          isPlaying.value = false
          stopVisualization()
        },
        onend: () => {
          handleTrackEnd()
        },
        onstop: () => {
          isPlaying.value = false
          currentTime.value = 0
        }
      })
    }
    
    const togglePlay = () => {
      if (!howl) {
        initAudio(currentTrack.value)
      }
      
      if (howl.playing()) {
        howl.pause()
      } else {
        howl.play()
      }
    }
    
    const seekTo = (value) => {
      if (howl) {
        howl.seek((value / 100) * duration.value)
      }
    }
    
    const handleTrackEnd = () => {
      // 计算M❤️值奖励
      const earnedMHeart = calculateMHeartReward()
      if (earnedMHeart > 0) {
        rewardAmount.value = earnedMHeart
        showReward.value = true
        updateMHeart(earnedMHeart)
      }
      
      // 处理播放模式
      if (repeatMode.value === 'one') {
        howl.play()
      } else if (repeatMode.value === 'all' || !isLastTrack()) {
        nextTrack()
      }
    }
    
    // M❤️值计算
    const calculateMHeartReward = () => {
      const baseReward = 10
      const emotionBonus = currentTrack.value?.emotions?.length * 2 || 0
      const timeBonus = Math.floor(currentTime.value / 60) * 0.5
      const rarityBonus = currentTrack.value?.rarity ? 5 : 0
      
      return Math.floor(baseReward + emotionBonus + timeBonus + rarityBonus)
    }
    
    const updateMHeart = (amount) => {
      mHeartValue.value += amount
      mHeartProgress.value = (mHeartValue.value % 100)
      
      // 升级检查
      const newLevel = Math.floor(mHeartValue.value / 100) + 1
      if (newLevel > mHeartLevel.value) {
        mHeartLevel.value = newLevel
        triggerLevelUpAnimation()
      }
      
      emit('mheart-update', {
        value: mHeartValue.value,
        level: mHeartLevel.value,
        progress: mHeartProgress.value
      })
    }
    
    // 可视化效果
    const startVisualization = () => {
      const canvas = waveCanvas.value
      const ctx = canvas.getContext('2d')
      const analyser = new (window.AudioContext || window.webkitAudioContext)()
      const source = analyser.createMediaElementSource(howl._sounds[0]._node)
      source.connect(analyser)
      analyser.connect(analyser.context.destination)
      
      const bufferLength = analyser.frequencyBinCount
      const dataArray = new Uint8Array(bufferLength)
      
      const draw = () => {
        animationId = requestAnimationFrame(draw)
        
        analyser.getByteFrequencyData(dataArray)
        
        ctx.fillStyle = 'rgba(0, 0, 0, 0.2)'
        ctx.fillRect(0, 0, canvas.width, canvas.height)
        
        const barWidth = (canvas.width / bufferLength) * 2.5
        let x = 0
        
        for (let i = 0; i < bufferLength; i++) {
          const barHeight = (dataArray[i] / 255) * canvas.height * 0.7
          
          // 创建渐变色
          const gradient = ctx.createLinearGradient(0, canvas.height - barHeight, 0, canvas.height)
          gradient.addColorStop(0, `hsl(${(i / bufferLength) * 360}, 100%, 50%)`)
          gradient.addColorStop(1, `hsl(${(i / bufferLength) * 360}, 100%, 30%)`)
          
          ctx.fillStyle = gradient
          ctx.fillRect(x, canvas.height - barHeight, barWidth, barHeight)
          
          x += barWidth + 1
        }
      }
      
      draw()
    }
    
    const stopVisualization = () => {
      if (animationId) {
        cancelAnimationFrame(animationId)
        animationId = null
      }
    }
    
    // 星尘动画
    const initStardust = () => {
      const particles = []
      const particleCount = 50
      
      for (let i = 0; i < particleCount; i++) {
        const particle = document.createElement('div')
        particle.className = 'stardust-particle'
        particle.style.cssText = `
          position: absolute;
          width: 2px;
          height: 2px;
          background: white;
          border-radius: 50%;
          opacity: ${Math.random()};
          pointer-events: none;
        `
        stardustBg.value.appendChild(particle)
        
        particles.push({
          element: particle,
          x: Math.random() * 100,
          y: Math.random() * 100,
          vx: (Math.random() - 0.5) * 0.1,
          vy: (Math.random() - 0.5) * 0.1
        })
      }
      
      const animateParticles = () => {
        particles.forEach(p => {
          p.x += p.vx
          p.y += p.vy
          
          if (p.x < 0) p.x = 100
          if (p.x > 100) p.x = 0
          if (p.y < 0) p.y = 100
          if (p.y > 100) p.y = 0
          
          p.element.style.left = `${p.x}%`
          p.element.style.top = `${p.y}%`
        })
        
        requestAnimationFrame(animateParticles)
      }
      
      animateParticles()
    }
    
    // 辅助方法
    const formatTime = (seconds) => {
      const mins = Math.floor(seconds / 60)
      const secs = Math.floor(seconds % 60)
      return `${mins}:${secs.toString().padStart(2, '0')}`
    }
    
    const getEmotionColor = (emotion) => {
      const colors = {
        happy: '#FFD700',
        sad: '#4169E1',
        energetic: '#FF4500',
        calm: '#90EE90',
        mysterious: '#9370DB'
      }
      return colors[emotion] || '#CCCCCC'
    }
    
    const triggerLevelUpAnimation = () => {
      gsap.fromTo('.mheart-icon', 
        { scale: 1, rotation: 0 },
        { 
          scale: 1.5, 
          rotation: 360,
          duration: 1,
          ease: 'back.out(1.7)',
          onComplete: () => {
            gsap.to('.mheart-icon', { scale: 1, duration: 0.3 })
          }
        }
      )
    }
    
    // 生命周期
    onMounted(() => {
      initStardust()
      if (props.track) {
        initAudio(props.track)
      }
    })
    
    onUnmounted(() => {
      if (howl) howl.unload()
      stopVisualization()
    })
    
    watch(() => props.track, (newTrack) => {
      if (newTrack) {
        initAudio(newTrack)
        currentTime.value = 0
      }
    })
    
    return {
      // 状态
      isPlaying,
      currentTime,
      duration,
      volume,
      isMuted,
      isShuffled,
      repeatMode,
      isExpanded,
      mHeartValue,
      mHeartLevel,
      mHeartProgress,
      showReward,
      rewardAmount,
      progress,
      volumeIcon,
      repeatIcon,
      currentTrack,
      
      // 引用
      stardustBg,
      visualizer,
      waveCanvas,
      particleField,
      
      // 方法
      togglePlay,
      seekTo,
      formatTime,
      getEmotionColor
    }
  }
}
</script>

    @click="handleClick"

<script>
import { ref, computed, onMounted } from 'vue'
import { gsap } from 'gsap'

export default {
  name: 'CosmicButton',
  
  props: {
    variant: {
      type: String,
      default: 'primary',
      validator: (value) => ['primary', 'secondary', 'ghost', 'danger'].includes(value)
    },
    size: {
      type: String,
      default: 'medium',
      validator: (value) => ['small', 'medium', 'large', 'xlarge'].includes(value)
    },
    icon: String,
    disabled: Boolean,
    loading: Boolean,
    active: Boolean,
    pulse: Boolean,
    rounded: {
      type: Boolean,
      default: true
    }
  },
  
  emits: ['click'],
  
  setup(props, { emit }) {
    const buttonRef = ref(null)
    const particleCount = 12
    
    const buttonClasses = computed(() => [
      'cosmic-button',
      `cosmic-button--${props.variant}`,
      `cosmic-button--${props.size}`,
      {
        'cosmic-button--disabled': props.disabled,
        'cosmic-button--loading': props.loading,
        'cosmic-button--active': props.active,
        'cosmic-button--rounded': props.rounded,
        'cosmic-button--icon-only': props.icon && !props.$slots?.default
      }
    ])
    
    const iconClass = computed(() => {
      const iconMap = {
        'play': 'fas fa-play',
        'pause': 'fas fa-pause',
        'skip-back': 'fas fa-step-backward',
        'skip-forward': 'fas fa-step-forward',
        'shuffle': 'fas fa-random',
        'repeat': 'fas fa-redo',

        '--ty': `${y}px`,

        )

import { ref, computed, onMounted, onUnmounted } from 'vue'
import { RouterView } from 'vue-router'
import { useUserStore } from '@/store/modules/user'
import { usePlayerStore } from '@/store/modules/player'
import AppHeader from '@/components/layout/AppHeader.vue'
import MiniPlayer from '@/components/player/MiniPlayer.vue'
import NotificationContainer from '@/components/ui/NotificationContainer.vue'
import MHeartFloat from '@/components/ui/MHeartFloat.vue'

export default {
  name: 'App',
  components: {
    RouterView,
    AppHeader,
    MiniPlayer,
    NotificationContainer,
    MHeartFloat
  },
  
  setup() {
    const userStore = useUserStore()
    const playerStore = usePlayerStore()
    const starCanvas = ref(null)
    const isLoading = ref(true)
    const showMHeartFloat = ref(false)
    const mHeartFloatAmount = ref(0)
    
    let animationId = null
    
    const appClasses = computed(() => [
      'app',
      `theme-${userStore.preferences.theme}`,
      {
        'has-player': !!playerStore.currentTrack
      }
    ])
    
    const currentTrack = computed(() => playerStore.currentTrack)
    
    // 星尘背景动画
    const initStarfield = () => {
      const canvas = starCanvas.value
      if (!canvas) return
      
      const ctx = canvas.getContext('2d')
      canvas.width = window.innerWidth
      canvas.height = window.innerHeight
      
      const stars = []
      const starCount = 200
      
      for (let i = 0; i < starCount; i++) {
        stars.push({
          x: Math.random() * canvas.width,
          y: Math.random() * canvas.height,
          size: Math.random() * 2,
          speed: Math.random() * 0.5 + 0.1,
          opacity: Math.random()
        })
      }
      
      const animate = () => {
        ctx.clearRect(0, 0, canvas.width, canvas.height)
        
        stars.forEach(star => {
          ctx.fillStyle = `rgba(255, 255, 255, ${star.opacity})`
          ctx.beginPath()
          ctx.arc(star.x, star.y, star.size, 0, Math.PI * 2)
          ctx.fill()
          
          star.y += star.speed
          star.opacity = Math.sin(Date.now() * 0.001 + star.x) * 0.5 + 0.5
          
          if (star.y > canvas.height) {
            star.y = 0
            star.x = Math.random() * canvas.width
          }
        })
        
        animationId = requestAnimationFrame(animate)
      }
      
      animate()
    }
    
    const handleResize = () => {
      if (starCanvas.value) {
        starCanvas.value.width = window.innerWidth
        starCanvas.value.height = window.innerHeight
      }
    }
    
    const showMHeartReward = (amount) => {
      mHeartFloatAmount.value = amount
      showMHeartFloat.value = true
      
      setTimeout(() => {
        showMHeartFloat.value = false
      }, 3000)
    }
    
    onMounted(() => {
      initStarfield()
      window.addEventListener('resize', handleResize)
      
      // 模拟加载完成
      setTimeout(() => {
        isLoading.value = false
      }, 2000)
      
      // 监听M❤️值变化
      userStore.$onAction(({ name, args }) => {
        if (name === 'updateMHeart' && args[0] > 0) {
          showMHeartReward(args[0])
        }
      })
    })
    
    onUnmounted(() => {
      if (animationId) {
        cancelAnimationFrame(animationId)
      }
      window.removeEventListener('resize', handleResize)
    })
    
    return {
      appClasses,
      currentTrack,
      isLoading,
      showMHeartFloat,
      mHeartFloatAmount,
      starCanvas
    }
  }
}
