```typescript
// src/services/api.ts - 统一API服务
import axios, { AxiosInstance, AxiosRequestConfig, AxiosResponse } from 'axios'
import { ApiResponse, ApiError } from '../types'

class ApiService {
  private client: AxiosInstance
  private baseURL: string

  constructor() {
    this.baseURL = process.env.REACT_APP_API_URL || 'http://localhost:3001/api'
    
    this.client = axios.create({
      baseURL: this.baseURL,
      timeout: 10000,
      headers: {
        'Content-Type': 'application/json'
      }
    })

    this.setupInterceptors()
  }

  private setupInterceptors() {
    // 请求拦截器
    this.client.interceptors.request.use(
      (config) => {
        const token = localStorage.getItem('token')
        if (token) {
          config.headers.Authorization = `Bearer ${token}`
        }
        return config
      },
      (error) => Promise.reject(error)
    )

    // 响应拦截器
    this.client.interceptors.response.use(
      (response: AxiosResponse<ApiResponse>) => response,
      async (error) => {
        const originalRequest = error.config

        if (error.response?.status === 401 && !originalRequest._retry) {
          originalRequest._retry = true
          
          try {
            const refreshToken = localStorage.getItem('refreshToken')
            if (refreshToken) {
              const response = await this.client.post('/auth/refresh', {
                refreshToken
              })
              
              const { token, refreshToken: newRefreshToken } = response.data.data
              localStorage.setItem('token', token)
              localStorage.setItem('refreshToken', newRefreshToken)
              
              originalRequest.headers.Authorization = `Bearer ${token}`
              return this.client(originalRequest)
            }
          } catch (refreshError) {
            // 刷新失败，跳转登录
            localStorage.removeItem('token')
            localStorage.removeItem('refreshToken')
            window.location.href = '/login'
          }
        }

        return Promise.reject(this.handleError(error))
      }
    )
  }

  private handleError(error: any): ApiError {
    if (error.response) {
      return {
        code: error.response.data.error?.code || 'UNKNOWN_ERROR',
        message: error.response.data.error?.message || '请求失败',
        details: error.response.data.error?.details
      }
    } else if (error.request) {
      return {
        code: 'NETWORK_ERROR',
        message: '网络连接失败'
      }
    } else {
      return {
        code: 'CLIENT_ERROR',
        message: error.message || '客户端错误'
      }
    }
  }

  // 通用请求方法
  async get<T>(url: string, config?: AxiosRequestConfig): Promise<ApiResponse<T>> {
    try {
      const response = await this.client.get(url, config)
      return response.data
    } catch (error) {
      throw this.handleError(error)
    }
  }

  async post<T>(url: string, data?: any, config?: AxiosRequestConfig): Promise<ApiResponse<T>> {
    try {
      const response = await this.client.post(url, data, config)
      return response.data
    } catch (error) {
      throw this.handleError(error)
    }
  }

  async put<T>(url: string, data?: any, config?: AxiosRequestConfig): Promise<ApiResponse<T>> {
    try {
      const response = await this.client.put(url, data, config)
      return response.data
    } catch (error) {
      throw this.handleError(error)
    }
  }

  async delete<T>(url: string, config?: AxiosRequestConfig): Promise<ApiResponse<T>> {
    try {
      const response = await this.client.delete(url, config)
      return response.data
    } catch (error) {
      throw this.handleError(error)
    }
  }

  // 文件上传
  async upload(url: string, file: File, onProgress?: (progress: number) => void): Promise<ApiResponse> {
    const formData = new FormData()
    formData.append('file', file)

    try {
      const response = await this.client.post(url, formData, {
        headers: {
          'Content-Type': 'multipart/form-data'
        },
        onUploadProgress: (progressEvent) => {
          if (onProgress && progressEvent.total) {
            const progress = (progressEvent.loaded / progressEvent.total) * 100
            onProgress(progress)
          }
        }
      })
      return response.data
    } catch (error) {
      throw this.handleError(error)
    }
  }
}

export const apiService = new ApiService()
```

```typescript
// src/services/cacheService.ts - 统一缓存服务
import { CacheItem } from '../types'

class CacheService {
  private memoryCache = new Map<string, CacheItem>()
  private readonly defaultTTL = 5 * 60 * 1000 // 5分钟

  // 设置缓存
  set<T>(key: string, data: T, ttl: number = this.defaultTTL): void {
    const item: CacheItem<T> = {
      key,
      data,
      timestamp: Date.now(),
      ttl,
      version: '1.0'
    }

    // 内存缓存
    this.memoryCache.set(key, item)

    // 本地存储
    try {
      localStorage.setItem(`cache_${key}`, JSON.stringify(item))
    } catch (error) {
      console.warn('LocalStorage write failed:', error)
    }
  }

  // 获取缓存
  get<T>(key: string): T | null {
    // 先从内存获取
    const memoryItem = this.memoryCache.get(key) as CacheItem<T>
    if (memoryItem && !this.isExpired(memoryItem)) {
      return memoryItem.data
    }

    // 从本地存储获取
    try {
      const stored = localStorage.getItem(`cache_${key}`)
      if (stored) {
        const item: CacheItem<T> = JSON.parse(stored)
        if (!this.isExpired(item)) {
          // 恢复到内存
          this.memoryCache.set(key, item)
          return item.data
        } else {
          // 过期清理
          localStorage.removeItem(`cache_${key}`)
        }
      }
    } catch (error) {
      console.warn('LocalStorage read failed:', error)
    }

    return null
  }

  // 删除缓存
  delete(key: string): void {
    this.memoryCache.delete(key)
    try {
      localStorage.removeItem(`cache_${key}`)
    } catch (error) {
      console.warn('LocalStorage delete failed:', error)
    }
  }

  // 清空所有缓存
  clear(): void {
    this.memoryCache.clear()
    try {
      const keys = Object.keys(localStorage)
      keys.forEach(key => {
        if (key.startsWith('cache_')) {
          localStorage.removeItem(key)
        }
      })
    } catch (error) {
      console.warn('LocalStorage clear failed:', error)
    }
  }

  // 检查是否过期
  private isExpired(item: CacheItem): boolean {
    return Date.now() - item.timestamp > item.ttl
  }

  // 获取缓存统计
  getStats() {
    const memoryStats = {
      size: this.memoryCache.size,
      keys: Array.from(this.memoryCache.keys())
    }

    let storageStats = { size: 0, keys: [] as string[] }
    try {
      const keys = Object.keys(localStorage)
      storageStats.keys = keys.filter(key => key.startsWith('cache_'))
      storageStats.size = storageStats.keys.length
    } catch (error) {
      console.warn('LocalStorage stats failed:', error)
    }

    return {
      memory: memoryStats,
      storage: storageStats
    }
  }
}

export const cacheService = new CacheService()
```

```typescript
// src/components/player/AudioPlayer.tsx - 统一音频播放器
import React, { useState, useRef, useEffect, useCallback } from 'react'
import { Box, Group, Slider, Progress, Text, ActionIcon, Tooltip } from '@mantine/core'
import {
  IconPlayerPlay,
  IconPlayerPause,
  IconPlayerSkipBack,
  IconPlayerSkipForward,
  IconVolume,
  IconRepeat,
  IconShuffle,
  IconHeart
} from '@tabler/icons-react'
import { Howl } from 'howler'
import { Track } from '../../types'
import { Button } from '../ui/Button'
import { cacheService } from '../../services/cacheService'
import { notifications } from '@mantine/notifications'

interface AudioPlayerProps {
  track: Track
  onNext?: () => void
  onPrevious?: () => void
  onLike?: (trackId: string) => void
}

export const AudioPlayer: React.FC<AudioPlayerProps> = ({
  track,
  onNext,
  onPrevious,
  onLike
}) => {
  const [isPlaying, setIsPlaying] = useState(false)
  const [currentTime, setCurrentTime] = useState(0)
  const [duration, setDuration] = useState(0)
  const [volume, setVolume] = useState(0.8)
  const [isRepeat, setIsRepeat] = useState(false)
  const [isShuffle, setIsShuffle] = useState(false)
  const [isLiked, setIsLiked] = useState(false)

  const howlRef = useRef<Howl | null>(null)
  const intervalRef = useRef<NodeJS.Timeout | null>(null)

  // 初始化音频
  useEffect(() => {
    if (track?.audioUrl) {
      // 检查缓存
      const cacheKey = `track_${track.id}`
      const cachedTrack = cacheService.get(cacheKey)
      
      if (howlRef.current) {
        howlRef.current.unload()
      }

      const howl = new Howl({
        src: [track.audioUrl],
        html5: true,
        preload: true,
        onload: () => {
          setDuration(howl.duration())
          // 缓存音频数据
          cacheService.set(cacheKey, {
            ...track,
            duration: howl.duration()
          }, 10 * 60 * 1000) // 10分钟
        },
        onplay: () => {
          setIsPlaying(true)
          startProgressUpdate()
          // 记录播放历史
          recordPlayHistory(track.id)
        },
        onpause: () => {
          setIsPlaying(false)
          stopProgressUpdate()
        },
        onend: () => {
          setIsPlaying(false)
          stopProgressUpdate()
          if (isRepeat) {
            howl.seek(0)
            howl.play()
          } else {
            onNext?.()
          }
        },
        onloaderror: (id, error) => {
          console.error('Audio load error:', error)
          notifications.show({
            title: '加载失败',
            message: '音频加载失败，请检查网络连接',
            color: 'red'
          })
        }
      })

      howlRef.current = howl
      howl.volume(volume)

      // 检查是否已收藏
      checkLikedStatus(track.id)
    }

    return () => {
      if (howlRef.current) {
        howlRef.current.unload()
      }
      stopProgressUpdate()
    }
  }, [track])

  // 更新播放进度
  const startProgressUpdate = () => {
    intervalRef.current = setInterval(() => {
      if (howlRef.current) {
        setCurrentTime(howlRef.current.seek() as number)
      }
    }, 100)
  }

  const stopProgressUpdate = () => {
    if (intervalRef.current) {
      clearInterval(intervalRef.current)
      intervalRef.current = null
    }
  }

  // 播放/暂停
  const togglePlay = () => {
    if (!howlRef.current) return

    if (isPlaying) {
      howlRef.current.pause()
    } else {
      howlRef.current.play()
    }
  }

  // 跳转到指定时间
  const seek = (time: number) => {
    if (howlRef.current) {
      howlRef.current.seek(time)
      setCurrentTime(time)
    }
  }

  // 调整音量
  const adjustVolume = (newVolume: number) => {
    setVolume(newVolume)
    if (howlRef.current) {
      howlRef.current.volume(newVolume)
    }
  }

  // 记录播放历史
  const recordPlayHistory = async (trackId: string) => {
    try {
      await apiService.post('/music/history', { trackId })
    } catch (error) {
      console.error('Record play history failed:', error)
    }
  }

  // 检查收藏状态
  const checkLikedStatus = async (trackId: string) => {
    try {
      const response = await apiService.get(`/music/liked/${trackId}`)
      setIsLiked(response.data?.liked || false)
    } catch (error) {
      console.error('Check liked status failed:', error)
    }
  }

  // 切换收藏
  const toggleLike = () => {
    const newLikedStatus = !isLiked
    setIsLiked(newLikedStatus)
    
    try {
      apiService.post(`/music/like/${track.id}`, {
        liked: newLikedStatus
      })
      onLike?.(track.id)
      
      notifications.show({
        title: newLikedStatus ? '已收藏' : '取消收藏',
        message: track.title,
        color: newLikedStatus ? 'green' : 'gray'
      })
    } catch (error) {
      console.error('Toggle like failed:', error)
      setIsLiked(!newLikedStatus) // 回滚状态
    }
  }

  // 格式化时间
  const formatTime = (seconds: number): string => {
    const mins = Math.floor(seconds / 60)
    const secs = Math.floor(seconds % 60)
    return `${mins}:${secs.toString().padStart(2, '0')}`
  }

  return (
    <Box className="bg-gradient-to-r from-indigo-600 to-purple-600 p-6 rounded-xl shadow-2xl">
      {/* 歌曲信息 */}
      <Group position="apart" align="center" mb="md">
        <div className="flex items-center space-x-4">
          <img
            src={track?.cover || '/default-cover.jpg'}
            alt={track?.title}
            className="w-16 h-16 rounded-lg shadow-lg"
          />
          <div>
            <Text color="white" weight={600} size="lg">
              {track?.title}
            </Text>
            <Text color="white" opacity={0.8}>
              {track?.artist}
            </Text>
          </div>
        </div>

        <Tooltip label={isLiked ? '取消收藏' : '收藏'}>
          <ActionIcon
            variant="transparent"
            color="white"
            onClick={toggleLike}
          >
            <IconHeart
              size={20}
              fill={isLiked ? 'white' : 'none'}
            />
          </ActionIcon>
        </Tooltip>
      </Group>

      {/* 播放控制 */}
      <Group position="center" spacing="xl" mb="md">
        <ActionIcon
          variant="transparent"
          color="white"
          onClick={() => setIsShuffle(!isShuffle)}
          className={isShuffle ? 'opacity-100' : 'opacity-50'}
        >
          <IconShuffle size={20} />
        </ActionIcon>

        <ActionIcon
          variant="transparent"
          color="white"
          onClick={onPrevious}
        >
          <IconPlayerSkipBack size={24} />
        </ActionIcon>

        <ActionIcon
          variant="filled"
          color="white"
          size="xl"
          radius="xl"
          onClick={togglePlay}
        >
          {isPlaying ? <IconPlayerPause size={24} /> : <IconPlayerPlay size={24} />}
        </ActionIcon>

        <ActionIcon
          variant="transparent"
          color="white"
          onClick={onNext}
        >
          <IconPlayerSkipForward size={24} />
        </ActionIcon>

        <ActionIcon
          variant="transparent"
          color="white"
          onClick={() => setIsRepeat(!isRepeat)}
          className={isRepeat ? 'opacity-100' : 'opacity-50'}
        >
          <IconRepeat size={20} />
        </ActionIcon>
      </Group>

      {/* 进度条 */}
      <Group spacing="xs" mb="md">
        <Text size="xs" color="white" opacity={0.8}>
          {formatTime(currentTime)}
        </Text>
        <Slider
          value={currentTime}
          onChange={seek}
          max={duration || 0}
          step={0.1}
```

```typescript
  const [emotionAnalysis, setEmotionAnalysis] = useState<number | null>(null)

  const form = useForm<LyricFormValues>({
    initialValues: {
      theme: '',
      style: 'pop',
      emotion: 50,
      keywords: '',
      length: 200
    },
    validate: {
      theme: (value) => (!value ? '请输入主题' : null),
      keywords: (value) => (!value ? '请输入关键词' : null)
    }
  })

  const generateLyrics = async (values: LyricFormValues) => {
    setIsGenerating(true)
    
    try {
      // 检查缓存
      const cacheKey = `lyrics_${JSON.stringify(values)}`
      const cachedResult = cacheService.get(cacheKey)
      
      if (cachedResult) {
        setGeneratedLyrics(prev => [cachedResult, ...prev])
        notifications.show({
          title: '歌词生成成功',
          message: '已为您生成新的歌词',
          color: 'green'
        })
        setIsGenerating(false)
        return
      }

      // 调用API生成歌词
      const response = await apiService.post('/ai/generate-lyrics', {
        theme: values.theme,
        style: values.style,
        emotion: values.emotion / 100,
        keywords: values.keywords.split(',').map(k => k.trim()),
        length: values.length
      })

      if (response.success && response.data) {
        const newLyric: LyricData = {
          id: Date.now().toString(),
          content: response.data.content,
          theme: values.theme,
          emotion: values.emotion,
          style: values.style,
          keywords: values.keywords.split(',').map(k => k.trim()),
          metadata: response.data.metadata,
          createdAt: new Date().toISOString()
        }
        
        setGeneratedLyrics(prev => [newLyric, ...prev])
        setEmotionAnalysis(response.data.emotionScore)
        
        // 缓存结果
        cacheService.set(cacheKey, newLyric, 30 * 60 * 1000) // 30分钟
        
        notifications.show({
          title: '歌词生成成功',
          message: '已为您生成新的歌词',
          color: 'green'
        })
      }
    } catch (error) {
      console.error('Generation failed:', error)
      notifications.show({
        title: '生成失败',
        message: error.message || '请稍后重试',
        color: 'red'
      })
    } finally {
      setIsGenerating(false)
    }
  }

  const saveLyrics = async (lyric: LyricData) => {
    try {
      const response = await apiService.post('/create/lyrics/save', lyric)
      
      if (response.success) {
        notifications.show({
          title: '保存成功',
          message: '歌词已保存到您的创作库',
          color: 'green'
        })
      }
    } catch (error) {
      notifications.show({
        title: '保存失败',
        message: error.message || '请稍后重试',
        color: 'red'
      })
    }
  }

  return (
    <Stack spacing="xl">
      <Card variant="elevated">
        <Card.Section withBorder inheritPadding py="xs">
          <Text weight={600} size="lg">AI歌词生成器</Text>
        </Card.Section>

        <form onSubmit={form.onSubmit(generateLyrics)}>
          <Stack>
            <TextInput
              label="主题"
              placeholder="如：爱情、梦想、自然..."
              {...form.getInputProps('theme')}
              required
            />

            <Select
              label="音乐风格"
              data={[
                { value: 'pop', label: '流行' },
                { value: 'rock', label: '摇滚' },
                { value: 'jazz', label: '爵士' },
                { value: 'classical', label: '古典' },
                { value: 'electronic', label: '电子' }
              ]}
              {...form.getInputProps('style')}
            />

            <div>
              <Text size="sm" weight={500} mb={5}>情感强度</Text>
              <Slider
                {...form.getInputProps('emotion')}
                marks={[
                  { value: 0, label: '平静' },
                  { value: 50, label: '适中' },
                  { value: 100, label: '激昂' }
                ]}
              />
            </div>

            <TextInput
              label="关键词"
              placeholder="用逗号分隔，如：星空, 徘徊, 温暖"
              {...form.getInputProps('keywords')}
              required
            />

            <div>
              <Text size="sm" weight={500} mb={5}>歌词长度</Text>
              <Slider
                {...form.getInputProps('length')}
                min={50}
                max={500}
                marks={[
                  { value: 50, label: '短' },
                  { value: 250, label: '中' },
                  { value: 500, label: '长' }
                ]}
              />
            </div>

            <UIComponent
              variant="gradient"
              onClick={() => generateLyrics(form.values)}
              loading={isGenerating}
              disabled={!form.isValid()}
            >
              生成歌词
            </UIComponent>
          </Stack>
        </form>
      </Card>

      {emotionAnalysis !== null && (
        <Card variant="elevated">
          <Text weight={600} mb="md">情感分析结果</Text>
          <Progress
            value={emotionAnalysis}
            color={emotionAnalysis > 60 ? 'green' : emotionAnalysis > 40 ? 'yellow' : 'red'}
            label={`${emotionAnalysis}%`}
            size="xl"
          />
          <Text size="sm" color="dimmed" mt="sm">
            情感倾向：{emotionAnalysis > 60 ? '积极' : emotionAnalysis > 40 ? '中性' : '消极'}
          </Text>
        </Card>
      )}

      {generatedLyrics.map((lyric) => (
        <Card key={lyric.id} variant="elevated">
          <Group position="apart" mb="md">
            <Group>
              <Badge color="blue">{lyric.style}</Badge>
              <Badge color="purple">{lyric.theme}</Badge>
            </Group>
            <UIComponent
              variant="outline"
              size="sm"
              onClick={() => saveLyrics(lyric)}
            >
              保存
            </UIComponent>
          </Group>
          
          <Textarea
            value={lyric.content}
            readOnly
            minRows={4}
            styles={{ input: { backgroundColor: 'transparent' } }}
          />
          
          <Text size="xs" color="dimmed" mt="sm">
            生成时间：{new Date(lyric.createdAt).toLocaleString()}
          </Text>
        </Card>
      ))}
    </Stack>
  )
}
```

```typescript
// src/components/error/ErrorBoundary.tsx - 统一错误边界
import React, { Component, ErrorInfo, ReactNode } from 'react'
import { Alert, Button, Group, Text, Title } from '@mantine/core'
import { notifications } from '@mantine/notifications'

interface Props {
  children: ReactNode
}

interface State {
  hasError: boolean
  error: Error | null
  errorInfo: ErrorInfo | null
}

export class ErrorBoundary extends Component<Props, State> {
  constructor(props: Props) {
    super(props)
    this.state = {
      hasError: false,
      error: null,
      errorInfo: null
    }
  }

  static getDerivedStateFromError(error: Error): State {
    return {
      hasError: true,
      error,
      errorInfo: null
    }
  }

  componentDidCatch(error: Error, errorInfo: ErrorInfo) {
    this.setState({
      error,
      errorInfo
    })

    // 上报错误
    this.reportError(error, errorInfo)
  }

  reportError = async (error: Error, errorInfo: ErrorInfo) => {
    try {
      await fetch('/api/v1/log/error', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json'
        },
        body: JSON.stringify({
          error: error.message,
          stack: error.stack,
          componentStack: errorInfo.componentStack,
          timestamp: new Date().toISOString(),
          userAgent: navigator.userAgent,
          url: window.location.href
        })
      })
    } catch (reportError) {
      console.error('Failed to report error:', reportError)
    }
  }

  handleReset = () => {
    this.setState({
      hasError: false,
      error: null,
      errorInfo: null
    })
  }

  render() {
    if (this.state.hasError) {
      return (
        <div className="min-h-screen flex items-center justify-center bg-gray-50">
          <div className="max-w-md w-full p-6">
            <Alert
              icon="⚠️"
              title="应用遇到错误"
              color="red"
              mb="md"
            >
              <Text mb="md">
                很抱歉，应用遇到了一个意外错误。我们已经记录了这个问题，请稍后重试。
              </Text>

              {process.env.NODE_ENV === 'development' && this.state.error && (
                <details className="mt-4">
                  <summary>错误详情（开发模式）</summary>
                  <pre className="mt-2 p-4 bg-gray-100 rounded text-xs overflow-auto">
                    {this.state.error.toString()}
                    {this.state.errorInfo?.componentStack}
                  </pre>
                </details>
              )}

              <Group position="center" mt="md">
                <Button onClick={this.handleReset} variant="outline">
                  重新加载
                </Button>
                <Button onClick={() => window.location.reload()}>
                  刷新页面
                </Button>
              </Group>
            </Alert>
          </div>
        </div>
      )
    }

    return this.props.children
  }
}

// 全局错误处理器
export const setupGlobalErrorHandling = () => {
  // 捕获未处理的Promise错误
  window.addEventListener('unhandledrejection', (event) => {
    console.error('Unhandled promise rejection:', event.reason)
    notifications.show({
      title: '系统错误',
      message: '发生了一个意外错误，请刷新页面重试',
      color: 'red'
    })
  })

  // 捕获全局错误
  window.addEventListener('error', (event) => {
    console.error('Global error:', event.error)
    notifications.show({
      title: '系统错误',
      message: '发生了一个意外错误，请刷新页面重试',
      color: 'red'
    })
  })
}
```
