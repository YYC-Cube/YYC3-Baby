```typescript
// src/store/modules/auth.ts
import { defineStore } from 'pinia'
import { authApi } from '@/api/auth'
import { storage } from '@/utils/storage'

interface User {
  id: string
  username: string
  email: string
  avatar: string
  role: string
  permissions: string[]
  createdAt: string
  lastLoginAt: string
}

interface AuthState {
  user: User | null
  token: string | null
  refreshToken: string | null
  isAuthenticated: boolean
  isLoading: boolean
  loginAttempts: number
  lockUntil: number | null
}

export const useAuthStore = defineStore('auth', {
  state: (): AuthState => ({
    user: null,
    token: storage.get('token') || null,
    refreshToken: storage.get('refreshToken') || null,
    isAuthenticated: false,
    isLoading: false,
    loginAttempts: 0,
    lockUntil: null
  }),

  getters: {
    isLocked: (state) => state.lockUntil ? Date.now() < state.lockUntil : false,
    remainingLockTime: (state) => {
      if (!state.lockUntil) return 0
      const remaining = state.lockUntil - Date.now()
      return Math.max(0, Math.ceil(remaining / 1000))
    },
    hasPermission: (state) => (permission: string) => {
      return state.user?.permissions.includes(permission) || false
    },
    isAdmin: (state) => state.user?.role === 'admin',
    isCreator: (state) => ['admin', 'creator'].includes(state.user?.role || '')
  },

  actions: {
    async login(credentials: LoginCredentials): Promise<void> {
      if (this.isLocked) {
        throw new Error(`账户已锁定，请${this.remainingLockTime}秒后再试`)
      }

      this.isLoading = true
      
      try {
        const response = await authApi.login(credentials)
        
        this.setAuthData(response.data)
        this.loginAttempts = 0
        this.lockUntil = null
        
        // 记录登录成功
        await this.recordLoginSuccess()
        
      } catch (error: any) {
        this.handleLoginError(error)
        throw error
      } finally {
        this.isLoading = false
      }
    },

    async register(userData: RegisterData): Promise<void> {
      this.isLoading = true
      
      try {
        const response = await authApi.register(userData)
        
        // 自动登录
        await this.login({
          username: userData.username,
          password: userData.password
        })
        
      } catch (error) {
        this.isLoading = false
        throw error
      }
    },

    async logout(): Promise<void> {
      try {
        if (this.token) {
          await authApi.logout()
        }
      } catch (error) {
        console.error('Logout error:', error)
      } finally {
        this.clearAuthData()
      }
    },

    async refreshToken(): Promise<void> {
      if (!this.refreshToken) {
        throw new Error('No refresh token available')
      }

      try {
        const response = await authApi.refreshToken(this.refreshToken)
        this.setAuthData(response.data)
      } catch (error) {
        this.clearAuthData()
        throw error
      }
    },

    async updateProfile(profileData: Partial<User>): Promise<void> {
      if (!this.user) throw new Error('User not authenticated')

      try {
        const response = await authApi.updateProfile(profileData)
        this.user = { ...this.user, ...response.data }
        
        // 更新本地存储
        storage.set('user', this.user)
      } catch (error) {
        throw error
      }
    },

    async changePassword(passwordData: ChangePasswordData): Promise<void> {
      try {
        await authApi.changePassword(passwordData)
        
        // 密码修改后需要重新登录
        this.clearAuthData()
      } catch (error) {
        throw error
      }
    },

    async enableTwoFactor(): Promise<string> {
      try {
        const response = await authApi.enableTwoFactor()
        return response.data.qrCode
      } catch (error) {
        throw error
      }
    },

    async verifyTwoFactor(code: string): Promise<void> {
      try {
        await authApi.verifyTwoFactor(code)
        
        if (this.user) {
          this.user.twoFactorEnabled = true
          storage.set('user', this.user)
        }
      } catch (error) {
        throw error
      }
    },

    setAuthData(data: AuthResponse): void {
      this.token = data.token
      this.refreshToken = data.refreshToken
      this.user = data.user
      this.isAuthenticated = true

      // 持久化存储
      storage.set('token', this.token)
      storage.set('refreshToken', this.refreshToken)
      storage.set('user', this.user)

      // 设置axios默认header
      if (this.token) {
        apiClient.defaults.headers.common['Authorization'] = `Bearer ${this.token}`
      }
    },

    clearAuthData(): void {
      this.user = null
      this.token = null
      this.refreshToken = null
      this.isAuthenticated = false

      // 清除存储
      storage.remove('token')
      storage.remove('refreshToken')
      storage.remove('user')

      // 清除axios header
      delete apiClient.defaults.headers.common['Authorization']
    },

    handleLoginError(error: any): void {
      this.loginAttempts++
      
      if (error.response?.status === 401) {
        // 连续失败5次锁定账户30分钟
        if (this.loginAttempts >= 5) {
          this.lockUntil = Date.now() + 30 * 60 * 1000
        }
      }
      
      this.isLoading = false
    },

    async recordLoginSuccess(): Promise<void> {
      try {
        await authApi.recordLogin({
          userId: this.user?.id,
          timestamp: new Date().toISOString(),
          userAgent: navigator.userAgent,
          ip: await this.getClientIP()
        })
      } catch (error) {
        console.error('Failed to record login:', error)
      }
    },

    async getClientIP(): Promise<string> {
      try {
        const response = await fetch('https://api.ipify.org?format=json')
        const data = await response.json()
        return data.ip
      } catch {
        return 'unknown'
      }
    },

    // 初始化认证状态
    async initAuth(): Promise<void> {
      if (this.token && this.user) {
        try {
          // 验证token是否有效
          await authApi.verify()
          this.isAuthenticated = true
          
          // 设置axios header
          apiClient.defaults.headers.common['Authorization'] = `Bearer ${this.token}`
          
          // 检查是否需要刷新token
          await this.checkTokenExpiry()
        } catch (error) {
          this.clearAuthData()
        }
      }
    },

    async checkTokenExpiry(): Promise<void> {
      if (!this.token) return

      try {
        const payload = JSON.parse(atob(this.token.split('.')[1]))
        const expiryTime = payload.exp * 1000
        const currentTime = Date.now()
        const timeUntilExpiry = expiryTime - currentTime

        // 如果token在5分钟内过期，自动刷新
        if (timeUntilExpiry < 5 * 60 * 1000) {
          await this.refreshToken()
        }
      } catch (error) {
        console.error('Token validation error:', error)
      }
    }
  }
})

// 路由守卫
// src/router/guards.ts
import { useAuthStore } from '@/store/modules/auth'

export const authGuard = async (to: any, from: any, next: any) => {
  const authStore = useAuthStore()

  // 初始化认证状态
  if (!authStore.isAuthenticated && authStore.token) {
    await authStore.initAuth()
  }

  // 检查路由是否需要认证
  if (to.meta.requiresAuth && !authStore.isAuthenticated) {
    next({
      name: 'Login',
      query: { redirect: to.fullPath }
    })
    return
  }

  // 检查权限
  if (to.meta.permission && !authStore.hasPermission(to.meta.permission)) {
    next({ name: 'Forbidden' })
    return
  }

  // 检查角色
  if (to.meta.role && !to.meta.role.includes(authStore.user?.role || '')) {
    next({ name: 'Forbidden' })
    return
  }

  next()
}
```

```plaintext
<script setup lang="ts">
import { ref, computed, onMounted } from 'vue'
import { useRouter } from 'vue-router'
import { useAuthStore } from '@/store/modules/auth'
import { validateEmail, validatePassword } from '@/utils/validation'
import ForgotPasswordModal from './ForgotPasswordModal.vue'

const router = useRouter()
const authStore = useAuthStore()

// 表单数据
const form = ref({
  username: '',
  password: '',
  twoFactorCode: '',
  rememberMe: false
})

// 状态
const showPassword = ref(false)
const requiresTwoFactor = ref(false)
const showForgotPassword = ref(false)
const errors = ref<Record<string, string>>({})

// 星空背景
const starCanvas = ref<HTMLCanvasElement>()

// 计算属性
const isFormValid = computed(() => {
  return form.value.username.trim() && 
         form.value.password.trim() && 
         (!requiresTwoFactor.value || form.value.twoFactorCode.length === 6) &&
         Object.keys(errors.value).length === 0
})

// 方法
const handleLogin = async () => {
  if (!isFormValid.value) return

  try {
    await authStore.login({
      username: form.value.username,
      password: form.value.password,
      twoFactorCode: form.value.twoFactorCode || undefined,
      rememberMe: form.value.rememberMe
    })

    // 登录成功后跳转
    const redirect = router.currentRoute.value.query.redirect as string
    router.push(redirect || '/')
  } catch (error: any) {
    if (error.response?.status === 206) {
      // 需要双因素认证
      requiresTwoFactor.value = true
    } else {
      // 显示错误消息
      showError(error.message || '登录失败，请检查用户名和密码')
    }
  }
}

const validateField = (field: string) => {
  errors.value[field] = ''

  switch (field) {
    case 'username':
      if (!form.value.username.trim()) {
        errors.value.username = '请输入用户名或邮箱'
      } else if (form.value.username.includes('@') && !validateEmail(form.value.username)) {
        errors.value.username = '请输入有效的邮箱地址'
      }
      break
      
    case 'password':
      if (!form.value.password.trim()) {
        errors.value.password = '请输入密码'
      } else if (form.value.password.length < 6) {
        errors.value.password = '密码长度至少6位'
      }
      break
  }
}

const handleSocialLogin = async (provider: string) => {
  try {
    // 构建OAuth URL
    const authUrl = `/api/v1/auth/oauth/${provider}`
    
    // 打开新窗口进行OAuth认证
    const popup = window.open(
      authUrl,
      'oauth',
      'width=500,height=600,scrollbars=yes,resizable=yes'
    )

    // 监听认证结果
    const checkClosed = setInterval(() => {
      if (popup?.closed) {
        clearInterval(checkClosed)
        // 检查认证状态
        authStore.initAuth()
      }
    }, 1000)
  } catch (error) {
    showError('第三方登录失败')
  }
}

const handleForgotPasswordSuccess = () => {
  showForgotPassword.value = false
  showSuccess('密码重置邮件已发送，请检查邮箱')
}

const showError = (message: string) => {
  // 使用全局通知组件显示错误
  console.error(message)
}

const showSuccess = (message: string) => {
  // 使用全局通知组件显示成功
  console.log(message)
}

// 初始化星空背景
const initStarBackground = () => {
  if (!starCanvas.value) return

  const canvas = starCanvas.value
  const ctx = canvas.getContext('2d')
  
  canvas.width = window.innerWidth
  canvas.height = window.innerHeight

  const stars = []
  const starCount = 200

  for (let i = 0; i < starCount; i++) {
    stars.push({
      x: Math.random() * canvas.width,
      y: Math.random() * canvas.height,
      size: Math.random() * 2,
      speed: Math.random() * 0.5 + 0.1,
      opacity: Math.random()
    })
  }

  const animate = () => {
    ctx.fillStyle = 'rgba(10, 14, 39, 0.1)'
    ctx.fillRect(0, 0, canvas.width, canvas.height)

    stars.forEach(star => {
      ctx.fillStyle = `rgba(255, 255, 255, ${star.opacity})`
      ctx.beginPath()
      ctx.arc(star.x, star.y, star.size, 0, Math.PI * 2)
      ctx.fill()

      star.y += star.speed
      star.opacity = Math.sin(Date.now() * 0.001 + star.x) * 0.5 + 0.5

      if (star.y > canvas.height) {
        star.y = 0
        star.x = Math.random() * canvas.width
      }
    })

    requestAnimationFrame(animate)
  }

  animate()
}

onMounted(() => {
  initStarBackground()
  
  // 检查是否已登录
  if (authStore.isAuthenticated) {
    router.push('/')
  }
})
```

```typescript
// src/services/aiMusicGenerator.ts
import * as tf from '@tensorflow/tfjs'
import { MusicTheory } from '@/utils/musicTheory'
import { AudioManager } from '@/utils/audioManager'

export class AIMusicGenerator {
  private model: tf.LayersModel | null = null
  private audioContext: AudioContext
  private musicTheory: MusicTheory
  private audioManager: AudioManager

  constructor() {
    this.audioContext = new AudioContext()
    this.musicTheory = new MusicTheory()
    this.audioManager = new AudioManager()
    this.initializeModels()
  }

  private async initializeModels(): Promise<void> {
    try {
      // 加载预训练的音乐生成模型
      this.model = await tf.loadLayersModel('/models/music-generator/model.json')
      console.log('AI音乐生成模型加载成功')
    } catch (error) {
      console.error('模型加载失败:', error)
      // 创建回退模型
      this.createFallbackModel()
    }
  }

  private createFallbackModel(): void {
    this.model = tf.sequential({
      layers: [
        tf.layers.lstm({
          units: 256,
          returnSequences: true,
          inputShape: [null, 128] // MIDI音符向量
        }),
        tf.layers.dropout({ rate: 0.3 }),
        tf.layers.lstm({
          units: 128,
          returnSequences: true
        }),
        tf.layers.dropout({ rate: 0.3 }),
        tf.layers.timeDistributed({
          layer: tf.layers.dense({ units: 128, activation: 'softmax' })
        })
      ]
    })
  }

  /**
   * 生成完整音乐作品
   */
  async generateMusic(params: MusicGenerationParams): Promise<GeneratedMusic> {
    const {
      theme,
      emotion,
      style,
      duration,
      tempo,
      key,
      timeSignature,
      instruments
    } = params

    // 1. 生成和弦进行
    const chordProgression = await this.generateChordProgression({
      style,
      emotion,
      key,
      duration
    })

    // 2. 生成旋律
    const melody = await this.generateMelody({
      chordProgression,
      emotion,
      style,
      key,
      duration
    })

    // 3. 生成节奏
    const rhythm = await this.generateRhythm({
      style,
      tempo,
      timeSignature,
      duration
    })

    // 4. 生成 bassline
    const bassline = await this.generateBassline({
      chordProgression,
      style,
      key,
      duration
    })

    // 5. 生成和声
    const harmony = await this.generateHarmony({
      chordProgression,
      melody,
      style,
      duration
    })

    // 6. 编排乐器
    const arrangement = await this.arrangeInstruments({
      melody,
      harmony,
      bassline,
      rhythm,
      instruments
    })

    // 7. 生成音频
    const audioBuffer = await this.renderToAudio(arrangement)

    return {
      id: generateId(),
      theme,
      emotion,
      style,
      duration,
      tempo,
      key,
      timeSignature,
      chordProgression,
      melody,
      harmony,
      bassline,
      rhythm,
      arrangement,
      audioBuffer,
      createdAt: new Date().toISOString()
    }
  }

  /**
   * 生成和弦进行
   */
  private async generateChordProgression(params: {
    style: string
    emotion: string
    key: string
    duration: number
  }): Promise<ChordProgression> {
    const { style, emotion, key, duration } = params

    // 根据风格和情感选择和弦进行模板
    const templates = this.getChordTemplates(style, emotion)
    const template = templates[Math.floor(Math.random() * templates.length)]

    // 将模板转换为实际和弦
    const chords = this.applyChordTemplate(template, key)
    
    // 计算每个和弦的时长
    const chordDuration = duration / chords.length
    const chordProgression: ChordProgression = []

    chords.forEach((chord, index) => {
      chordProgression.push({
        chord,
        startTime: index * chordDuration,
        duration: chordDuration,
        inversion: Math.floor(Math.random() * 3),
        velocity: this.getVelocityByEmotion(emotion)
      })
    })

    return chordProgression
  }

  /**
   * 生成旋律
   */
  private async generateMelody(params: {
    chordProgression: ChordProgression
    emotion: string
    style: string
    key: string
    duration: number
  }): Promise<Melody> {
    const { chordProgression, emotion, style, key, duration } = params

    // 准备输入数据
    const inputData = this.prepareMelodyInput(chordProgression, key)
    const inputTensor = tf.tensor3d([inputData])

    // 使用模型生成旋律
    const prediction = this.model!.predict(inputTensor) as tf.Tensor
    const outputData = await prediction.data()

    // 将输出转换为音符序列
    const notes = this.decodeMelodyOutput(outputData, chordProgression, emotion, style)

    prediction.dispose()
    inputTensor.dispose()

    return {
      notes,
      key,
      scale: this.musicTheory.getScale(key),
      totalDuration: duration
    }
  }

  /**
   * 生成节奏
   */
  private async generateRhythm(params: {
    style: string
    tempo: number
    timeSignature: string
    duration: number
  }): Promise<Rhythm> {
    const { style, tempo, timeSignature, duration } = params

    // 根据风格生成节奏模式
    const patterns = this.getRhythmPatterns(style)
    const pattern = patterns[Math.floor(Math.random() * patterns.length)]

    // 计算小节数
    const [beatsPerBar, beatUnit] = timeSignature.split('/').map(Number)
    const barDuration = (beatsPerBar * 60) / tempo
    const totalBars = Math.ceil(duration / barDuration)

    const rhythm: Rhythm = {
      pattern,
      tempo,
      timeSignature,
      totalBars,
      tracks: []
    }

    // 生成各个打击乐轨道
    const instruments = this.getRhythmInstruments(style)
    instruments.forEach(instrument => {
      const track = this.generateRhythmTrack(
        instrument,
        pattern,
        totalBars,
        tempo,
        beatsPerBar
      )
      rhythm.tracks.push(track)
    })

    return rhythm
  }

  /**
   * 生成 Bassline
   */
  private async generateBassline(params: {
    chordProgression: ChordProgression
    style: string
    key: string
    duration: number
  }): Promise<Bassline> {
    const { chordProgression, style, key, duration } = params

    const notes: Note[] = []

    chordProgression.forEach((chordInfo, index) => {
      // 获取和弦根音
      const rootNote = this.musicTheory.getChordRoot(chordInfo.chord)
      
      // 生成 bassline 音符
      const bassNotes = this.generateBassNotes(
        rootNote,
        chordInfo,
        style,
        index
      )

      notes.push(...bassNotes)
    })

    return {
      notes,
      instrument: this.getBassInstrument(style),
      key
    }
  }

  /**
   * 生成和声
   */
  private async generateHarmony(params: {
    chordProgression: ChordProgression
    melody: Melody
    style: string
    duration: number
  }): Promise<Harmony> {
    const { chordProgression, melody, style, duration } = params

    const harmonyNotes: Note[] = []

    // 为每个和弦生成和声
    chordProgression.forEach((chordInfo, index) => {
      const chordNotes = this.musicTheory.getChordNotes(chordInfo.chord)
      
      // 根据旋律音符调整和声
      const melodyNotesInChord = melody.notes.filter(note => 
        note.startTime >= chordInfo.startTime &&
        note.startTime < chordInfo.startTime + chordInfo.duration
      )

      const harmonyChord = this.adjustHarmonyToMelody(
        chordNotes,
        melodyNotesInChord,
        chordInfo,
        style
      )

      harmonyNotes.push(...harmonyChord)
    })

    return {
      notes: harmonyNotes,
      instrument: this.getHarmonyInstrument(style)
    }
  }

  /**
   * 编排乐器
   */
  private async arrangeInstruments(params: {
    melody: Melody
    harmony: Harmony
    bassline: Bassline
    rhythm: Rhythm
    instruments: string[]
  }): Promise<Arrangement> {
    const { melody, harmony, bassline, rhythm, instruments } = params

    const arrangement: Arrangement = {
      tracks: [],
      tempo: rhythm.tempo,
      timeSignature: rhythm.timeSignature,
      duration: Math.max(
        melody.totalDuration,
        bassline.notes[bassline.notes.length - 1]?.startTime || 0
      )
    }

    // 添加旋律轨道
    arrangement.tracks.push({
      id: 'melody',
      name: '主旋律',
      instrument: instruments[0] || 'piano',
      notes: melody.notes,
      volume: 0.8,
      pan: 0,
      effects: ['reverb', 'delay']
    })

    // 添加和声轨道
    arrangement.tracks.push({
      id: 'harmony',
      name: '和声',
      instrument: instruments[1] || 'strings',
      notes: harmony.notes,
      volume: 0.6,
      pan: 0,
      effects: ['reverb']
    })

    // 添加 Bass 轨道
    arrangement.tracks.push({
      id: 'bass',
      name: '贝斯',
      instrument: bassline.instrument,
      notes: bassline.notes,
      volume: 0.7,
      pan: 0,
      effects: []
    })

    // 添加节奏轨道
    rhythm.tracks.forEach((rhythmTrack, index) => {
      arrangement.tracks.push({
        id: `rhythm-${index}`,
        name: rhythmTrack.instrument,
        instrument: rhythmTrack.instrument,
        notes: rhythmTrack.notes,
        volume: rhythmTrack.volume,
        pan: rhythmTrack.pan || 0,
        effects: []
      })
    })

    return arrangement
  }

  /**
   * 渲染音频
   */
  private async renderToAudio(arrangement: Arrangement): Promise<AudioBuffer> {
    const sampleRate = this.audioContext.sampleRate
    const duration = arrangement.duration
    const numberOfChannels = 2
    const audioBuffer = this.audioContext.createBuffer(
      numberOfChannels,
      sampleRate * duration,
      sampleRate
    )

    // 渲染每个轨道
    for (const track of arrangement.tracks) {
      await this.renderTrack(track, audioBuffer, sampleRate)
    }

    return audioBuffer
  }

  /**
   * 渲染单个轨道
   */
  private async renderTrack(
    track: Track,
    audioBuffer: AudioBuffer,
    sampleRate: number
  ): Promise<void> {
    for (const note of track.notes) {
      const startTime = note.startTime
      const duration = note.duration || 0.5
      const frequency = this.musicTheory.midiToFrequency(note.pitch)

      // 创建音符振荡器
      const oscillator = this.audioContext.createOscillator()
      const gainNode = this.audioContext.createGain()

      oscillator.type = this.getOscillatorType(track.instrument)
      oscillator.frequency.setValueAtTime(frequency, startTime)

      // 设置音量包络
      gainNode.gain.setValueAtTime(0, startTime)
      gainNode.gain.linearRampToValueAtTime(track.volume, startTime + 0.01)
      gainNode.gain.exponentialRampToValueAtTime(track.volume * 0.7, startTime + duration * 0.3)
      gainNode.gain.exponentialRampToValueAtTime(0.01, startTime + duration)

      // 连接节点
      oscillator.connect(gainNode)
      gainNode.connect(this.audioContext.destination)

      // 播放音符
      oscillator.start(startTime)
      oscillator.stop(startTime + duration)
    }
  }

  // 辅助方法
  private getChordTemplates(style: string, emotion: string): string[][] {
    const templates: Record<string, Record<string, string[][]>> = {
      pop: {
        happy: [['I', 'V', 'vi', 'IV'], ['I', 'IV', 'V', 'I']],
        sad: [['vi', 'IV', 'I', 'V'], ['i', 'VI', 'III', 'VII']],
        energetic: [['I', 'V', 'vi', 'IV'], ['I', 'VI', 'IV', 'V']],
        calm: [['I', 'vi', 'IV', 'V'], ['I', 'IV', 'vi', 'ii']]
      },
      rock: {
        happy: [['I', 'IV', 'V', 'IV'], ['I', 'V', 'vi', 'IV']],
        sad: [['vi', 'IV', 'I', 'V'], ['i', 'VII', 'VI', 'V']],
        energetic: [['I', 'V', 'vi', 'IV'], ['I', 'IV', 'V', 'vi']],
        calm: [['I', 'vi', 'IV', 'V'], ['I', 'IV', 'ii', 'V']]
      },
      jazz: {
        happy: [['Imaj7', 'VIm7', 'IIm7', 'V7'], ['IIIm7', 'VIm7', 'IIm7', 'V7']],
        sad: [['Im7', 'VIm7b5', 'IIm7b5', 'V7b9'], ['Im7', 'VIm7b5', 'IIm7b5', 'V7alt']],
        energetic: [['Imaj7', 'VI7', 'IIm7', 'V7'], ['IIIm7', 'VI7', 'IIm7', 'V7']],
        calm: [['Imaj7', 'VIm7', 'IIm7', 'V7'], ['Imaj7', 'IVmaj7', 'IIIm7', 'VIm7']]
      }
    }

    return templates[style]?.[emotion] || templates.pop.happy
  }

  private applyChordTemplate(template: string[], key: string): string[] {
    return template.map(roman => this.musicTheory.romanToChord(roman, key))
  }

  private getVelocityByEmotion(emotion: string): number {
    const velocities = {
      happy: 0.8,
      sad: 0.5,
      energetic: 1.0,
      calm: 0.6,
      romantic: 0.7,
      mysterious: 0.6
    }
    return velocities[emotion] || 0.7
  }

  private getOscillatorType(instrument: string): OscillatorType {
    const types: Record<string, OscillatorType> = {
      piano: 'triangle',
      strings: 'sawtooth',
      bass: 'sine',
      drums: 'square',
      synth: 'sawtooth'
    }
    return types[instrument] || 'sine'
  }

  private generateId(): string {
    return Math.random().toString(36).substr(2, 9)
  }
```

```plaintext
const audioBufferToWav = async (buffer: AudioBuffer): Promise<Blob> => {
  const length = buffer.length * buffer.numberOfChannels * 2 + 44
  const arrayBuffer = new ArrayBuffer(length)
  const view = new DataView(arrayBuffer)
  const channels: Float32Array[] = []
  let offset = 0
  let pos = 0

  // 写入WAV头
  const setUint16 = (data: number) => {
    view.setUint16(pos, data, true)
    pos += 2
  }
  
  const setUint32 = (data: number) => {
    view.setUint32(pos, data, true)
    pos += 4
  }

  // RIFF identifier
  setUint32(0x46464952)
  // file length
  setUint32(length - 8)
  // WAVE identifier
  setUint32(0x45564157)
  // fmt chunk identifier
  setUint32(0x20746d66)
  // chunk length
  setUint32(16)
  // sample format (raw)
  setUint16(1)
  // channel count
  setUint16(buffer.numberOfChannels)
  // sample rate
  setUint32(buffer.sampleRate)
  // byte rate
  setUint32(buffer.sampleRate * 2 * buffer.numberOfChannels)
  // block align
  setUint16(buffer.numberOfChannels * 2)
  // bits per sample
  setUint16(16)
  // data chunk identifier
  setUint32(0x61746164)
  // data chunk length
  setUint32(length - pos - 4)

  // 写入音频数据
  for (let i = 0; i < buffer.numberOfChannels; i++) {
    channels.push(buffer.getChannelData(i))
  }

  while (pos < length) {
    for (let i = 0; i < buffer.numberOfChannels; i++) {
      let sample = Math.max(-1, Math.min(1, channels[i][offset]))
      sample = sample < 0 ? sample * 0x8000 : sample * 0x7FFF
      view.setInt16(pos, sample, true)
      pos += 2
    }
    offset++
  }

  return new Blob([arrayBuffer], { type: 'audio/wav' })
}
```

```typescript
// src/services/recommendationEngine.ts
import { userBehaviorApi } from '@/api/userBehavior'
import { musicApi } from '@/api/music'
import { Redis } from 'ioredis'

export class RecommendationEngine {
  private redis: Redis
  private userProfiles: Map<string, UserProfile> = new Map()
  private itemFeatures: Map<string, ItemFeatures> = new Map()
  
  constructor(redis: Redis) {
    this.redis = redis
    this.initializeRecommendationSystem()
  }

  private async initializeRecommendationSystem(): Promise<void> {
    // 加载用户画像
    await this.loadUserProfiles()
    
    // 加载音乐特征
    await this.loadItemFeatures()
    
    // 启动定时更新任务
    this.startPeriodicUpdates()
  }

  /**
   * 获取用户个性化推荐
   */
  async getRecommendations(userId: string, options: RecommendationOptions = {}): Promise<RecommendationResult> {
    const {
      count = 20,
      strategy = 'hybrid',
      excludeSeen = true,
      context = {}
    } = options

    // 获取用户画像
    const userProfile = await this.getUserProfile(userId)
    if (!userProfile) {
      return this.getDefaultRecommendations(count)
    }

    let recommendations: MusicItem[] = []

    switch (strategy) {
      case 'collaborative':
        recommendations = await this.getCollaborativeRecommendations(userProfile, count)
        break
      case 'content':
        recommendations = await this.getContentBasedRecommendations(userProfile, count)
        break
      case 'contextual':
        recommendations = await this.getContextualRecommendations(userProfile, context, count)
        break
      case 'hybrid':
      default:
        recommendations = await this.getHybridRecommendations(userProfile, count, context)
        break
    }

    // 过滤已见内容
    if (excludeSeen) {
      recommendations = await this.filterSeenContent(userId, recommendations)
    }

    // 应用多样性策略
    recommendations = await this.applyDiversityStrategy(recommendations, userProfile)

    // 生成推荐解释
    const explanations = await this.generateExplanations(recommendations, userProfile)

    return {
      recommendations,
      explanations,
      strategy,
      timestamp: new Date().toISOString(),
      userId
    }
  }

  /**
   * 协同过滤推荐
   */
  private async getCollaborativeRecommendations(
    userProfile: UserProfile, 
    count: number
  ): Promise<MusicItem[]> {
    // 找到相似用户
    const similarUsers = await this.findSimilarUsers(userProfile, 50)
    
    // 获取相似用户喜欢的音乐
    const candidateItems = await this.getCandidateItemsFromUsers(similarUsers, userProfile.userId)
    
    // 计算预测评分
    const scoredItems = await this.predictRatings(userProfile, candidateItems)
    
    // 返回最高评分的音乐
    return scoredItems
      .sort((a, b) => b.score - a.score)
      .slice(0, count)
      .map(item => item.music)
  }

  /**
   * 基于内容的推荐
   */
  private async getContentBasedRecommendations(
    userProfile: UserProfile, 
    count: number
  ): Promise<MusicItem[]> {
    // 获取用户喜欢的音乐特征
    const preferredFeatures = await this.extractUserPreferredFeatures(userProfile)
    
    // 查找相似特征的音乐
    const candidateItems = await this.findSimilarContent(preferredFeatures, userProfile.userId)
    
    // 计算内容相似度
    const scoredItems = await this.calculateContentSimilarity(candidateItems, preferredFeatures)
    
    return scoredItems
      .sort((a, b) => b.similarity - a.similarity)
      .slice(0, count)
      .map(item => item.music)
  }

  /**
   * 上下文感知推荐
   */
  private async getContextualRecommendations(
    userProfile: UserProfile,
    context: RecommendationContext,
    count: number
  ): Promise<MusicItem[]> {
    const {
      timeOfDay,
      location,
      weather,
      activity,
      mood
    } = context

    // 根据上下文调整推荐权重
    const contextWeights = await this.calculateContextWeights({
      timeOfDay,
      weather,
      activity,
      mood
    })

    // 获取基础推荐
    const baseRecommendations = await this.getHybridRecommendations(userProfile, count * 2)

    // 应用上下文权重
    const contextualScores = await this.applyContextWeights(
      baseRecommendations,
      contextWeights,
      userProfile
    )

    // 返回调整后的推荐
    return contextualScores
      .sort((a, b) => b.adjustedScore - a.adjustedScore)
      .slice(0, count)
      .map(item => item.music)
  }

  /**
   * 混合推荐策略
   */
  private async getHybridRecommendations(
    userProfile: UserProfile,
    count: number,
    context: RecommendationContext = {}
  ): Promise<MusicItem[]> {
    // 获取多种策略的推荐结果
    const collaborativeRecs = await this.getCollaborativeRecommendations(userProfile, count)
    const contentRecs = await this.getContentBasedRecommendations(userProfile, count)
    
    let contextualRecs: MusicItem[] = []
    if (Object.keys(context).length > 0) {
      contextualRecs = await this.getContextualRecommendations(userProfile, context, count)
    }

    // 融合推荐结果
    const fusedRecommendations = await this.fuseRecommendations([
      { items: collaborativeRecs, weight: 0.4, strategy: 'collaborative' },
      { items: contentRecs, weight: 0.4, strategy: 'content' },
      { items: contextualRecs, weight: 0.2, strategy: 'contextual' }
    ])

    return fusedRecommendations.slice(0, count)
  }

  /**
   * 更新用户行为
   */
  async updateUserBehavior(userId: string, behavior: UserBehavior): Promise<void> {
    // 实时更新用户画像
    await this.updateUserProfile(userId, behavior)
    
    // 记录行为日志
    await this.logBehavior(userId, behavior)
    
    // 更新Redis缓存
    await this.updateBehaviorCache(userId, behavior)
    
    // 触发实时推荐更新（如果需要）
    if (this.shouldUpdateRealtime(behavior)) {
      await this.updateRealtimeRecommendations(userId)
    }
  }

  /**
   * 生成推荐解释
   */
  private async generateExplanations(
    recommendations: MusicItem[],
    userProfile: UserProfile
  ): Promise<RecommendationExplanation[]> {
    const explanations: RecommendationExplanation[] = []

    for (const music of recommendations) {
      const explanation = await this.generateExplanation(music, userProfile)
      explanations.push(explanation)
    }

    return explanations
  }

  private async generateExplanation(
    music: MusicItem,
    userProfile: UserProfile
  ): Promise<RecommendationExplanation> {
    const reasons: string[] = []

    // 分析推荐原因
    if (await this.isBecauseSimilarUsers(music, userProfile)) {
      reasons.push('喜欢相似音乐的用户也喜欢这首')
    }

    if (await this.isBecauseSimilarContent(music, userProfile)) {
      reasons.push('与你喜欢的音乐风格相似')
    }

    if (await this.isBecauseTrending(music)) {
      reasons.push('正在热门趋势中')
    }

    if (await this.isBecauseNewRelease(music, userProfile)) {
      reasons.push('你喜欢的艺人新发布')
    }

    if (await this.isBecauseContext(music, userProfile)) {
      reasons.push('适合当前场景')
    }

    return {
      musicId: music.id,
      reasons,
      confidence: this.calculateExplanationConfidence(reasons.length),
      type: this.getExplanationType(reasons)
    }
  }

  /**
   * 获取用户画像
   */
  private async getUserProfile(userId: string): Promise<UserProfile | null> {
    // 先从缓存获取
    let profile = this.userProfiles.get(userId)
    
    if (!profile) {
      // 从Redis获取
      const cached = await this.redis.hgetall(`user_profile:${userId}`)
      if (Object.keys(cached).length > 0) {
        profile = JSON.parse(cached.profile)
        this.userProfiles.set(userId, profile)
      } else {
        // 从数据库构建画像
        profile = await this.buildUserProfile(userId)
        if (profile) {
          this.userProfiles.set(userId, profile)
          await this.cacheUserProfile(userId, profile)
        }
      }
    }

    return profile
  }

  /**
   * 构建用户画像
   */
  private async buildUserProfile(userId: string): Promise<UserProfile | null> {
    try {
      // 获取用户行为数据
      const behaviors = await userBehaviorApi.getUserBehaviors(userId)
      
      // 获取用户偏好
      const preferences = await userBehaviorApi.getUserPreferences(userId)
      
      // 计算特征向量
      const features = await this.calculateUserFeatures(behaviors, preferences)
      
      // 分析音乐品味
      const taste = await this.analyzeMusicTaste(behaviors)
      
      // 计算活跃度
      const activity = await this.calculateUserActivity(behaviors)

      return {
        userId,
        features,
        taste,
        activity,
        preferences,
        lastUpdated: new Date().toISOString()
      }
    } catch (error) {
      console.error('Failed to build user profile:', error)
      return null
    }
  }

  /**
   * 计算用户特征向量
   */
  private async calculateUserFeatures(
    behaviors: UserBehavior[],
    preferences: UserPreferences
  ): Promise<number[]> {
    // 特征维度：音乐类型、情感、节奏、年代、语言等
    const features = new Array(128).fill(0)
    
    // 基于行为更新特征
    for (const behavior of behaviors) {
      const itemFeatures = await this.getItemFeatures(behavior.itemId)
      if (itemFeatures) {
        const weight = this.getBehaviorWeight(behavior.type)
        for (let i = 0; i < features.length; i++) {
          features[i] += itemFeatures.vector[i] * weight
        }
      }
    }
    
    // 归一化
    const norm = Math.sqrt(features.reduce((sum, val) => sum + val * val, 0))
    if (norm > 0) {
      for (let i = 0; i < features.length; i++) {
        features[i] /= norm
      }
    }
    
    return features
  }

  /**
   * 查找相似用户
   */
  private async findSimilarUsers(
    targetUser: UserProfile, 
    limit: number
  ): Promise<SimilarUser[]> {
    const similarities: SimilarUser[] = []
    
    // 获取所有用户画像（实际实现中应该使用更高效的算法）
    const allUsers = await this.getAllUserProfiles()
    
    for (const user of allUsers) {
      if (user.userId === targetUser.userId) continue
      
      const similarity = this.calculateUserSimilarity(targetUser, user)
      if (similarity > 0.1) { // 相似度阈值
        similarities.push({
          userId: user.userId,
          similarity,
          commonInterests: await this.findCommonInterests(targetUser, user)
        })
      }
    }
    
    return similarities
      .sort((a, b) => b.similarity - a.similarity)
      .slice(0, limit)
  }

  /**
   * 计算用户相似度（余弦相似度）
   */
  private calculateUserSimilarity(user1: UserProfile, user2: UserProfile): number {
    const features1 = user1.features
    const features2 = user2.features
    
    let dotProduct = 0
    let norm1 = 0
    let norm2 = 0
    
    for (let i = 0; i < features1.length; i++) {
      dotProduct += features1[i] * features2[i]
      norm1 += features1[i] * features1[i]
      norm2 += features2[i] * features2[i]
    }
    
    if (norm1 === 0 || norm2 === 0) return 0
    
    return dotProduct / (Math.sqrt(norm1) * Math.sqrt(norm2))
  }

  /**
   * 应用多样性策略
   */
  private async applyDiversityStrategy(
    recommendations: MusicItem[],
    userProfile: UserProfile
  ): Promise<MusicItem[]> {
    const diverseRecommendations: MusicItem[] = []
    const usedGenres = new Set<string>()
    const usedArtists = new Set<string>()
    const maxSameGenre = Math.ceil(recommendations.length * 0.3)
    const maxSameArtist = Math.ceil(recommendations.length * 0.2)
    
    let genreCount: Record<string, number> = {}
    let artistCount: Record<string, number> = {}
    
    for (const music of recommendations) {
      const genreCount = genreCount[music.genre] || 0
      const artistCount = artistCount[music.artist] || 0
      
      // 检查多样性约束
      if (genreCount < maxSameGenre && artistCount < maxSameArtist) {
        diverseRecommendations.push(music)
        usedGenres.add(music.genre)
        usedArtists.add(music.artist)
        genreCount[music.genre] = (genreCount[music.genre] || 0) + 1
        artistCount[music.artist] = (artistCount[music.artist] || 0) + 1
      }
    }
    
    return diverseRecommendations
  }

  /**
   * 启动定时更新任务
   */
  private startPeriodicUpdates(): void {
    // 每小时更新热门推荐
    setInterval(async () => {
      await this.updateTrendingRecommendations()
    }, 60 * 60 * 1000)
    
    // 每天更新用户画像
    setInterval(async () => {
      await this.updateAllUserProfiles()
    }, 24 * 60 * 60 * 1000)
    
    // 每周更新音乐特征
    setInterval(async () => {
      await this.updateItemFeatures()
    }, 7 * 24 * 60 * 60 * 1000)
  }

  /**
   * 处理用户反馈
   */
  async handleFeedback(userId: string, feedback: RecommendationFeedback): Promise<void> {
    // 更新推荐模型
    await this.updateRecommendationModel(userId, feedback)
    
    // 调整用户偏好
    await this.adjustUserPreferences(userId, feedback)
    
    // 记录反馈日志
    await this.logFeedback(userId, feedback)
    
    // 实时调整推荐策略
    await this.adjustRecommendationStrategy(userId, feedback)
  }

  // 辅助方法
  private getBehaviorWeight(behaviorType: string): number {
    const weights: Record<string, number> = {
      'play': 1.0,
      'like': 2.0,
      'share': 1.5,
      'download': 3.0,
      'add_to_playlist': 2.5,
      'skip': -0.5,
      'dislike': -2.0
    }
    return weights[behaviorType] || 0
  }

  private async getDefaultRecommendations(count: number): Promise<MusicItem[]> {
    // 返回热门音乐作为默认推荐
    return await musicApi.getTrendingMusic(count)
  }

  private async filterSeenContent(userId: string, recommendations: MusicItem[]): Promise<MusicItem[]> {
    const seenItems = await this.getUserSeenItems(userId)
    const seenIds = new Set(seenItems.map(item => item.id))
    
    return recommendations.filter(music => !seenIds.has(music.id))
  }

  private async cacheUserProfile(userId: string, profile: UserProfile): Promise<void> {
    await this.redis.hset(
      `user_profile:${userId}`,
      'profile',
      JSON.stringify(profile)
    )
    await this.redis.expire(`user_profile:${userId}`, 24 * 60 * 60) // 24小时过期
  }
```

```typescript
  (config) => {
    // 添加认证token
    const authStore = useAuthStore()
    if (authStore.token) {
      config.headers.Authorization = `Bearer ${authStore.token}`
    }

    // 添加请求ID用于追踪
    config.headers['X-Request-ID'] = generateRequestId()

    // 添加时间戳

    return config
  },
  (error) => {
    return Promise.reject(error)
  }
)

// 响应拦截器
apiClient.interceptors.response.use(
  (response) => {
    return response
  },
  async (error) => {
    const authStore = useAuthStore()
    const notificationStore = useNotificationStore()

    if (error.response) {
      const { status, data } = error.response

      switch (status) {
        case 401:
          // Token过期或无效
          if (data.code === 'TOKEN_EXPIRED') {
            try {
              await authStore.refreshToken()
              // 重试原请求
              return apiClient.request(error.config)
            } catch (refreshError) {
              authStore.logout()
              router.push('/login')
            }
          } else {
            authStore.logout()
            router.push('/login')
          }
          break

        case 403:
          // 权限不足
          notificationStore.showError('权限不足，无法访问该资源')
          break

        case 404:
          // 资源不存在
          notificationStore.showError('请求的资源不存在')
          break

        case 429:
          // 请求过于频繁
          notificationStore.showError('请求过于频繁，请稍后再试')
          break

        case 500:
          // 服务器错误
          notificationStore.showError('服务器错误，请稍后再试')
          break

        default:
          // 其他错误
          notificationStore.showError(data.message || '请求失败')
      }
    } else if (error.request) {
      // 网络错误
      notificationStore.showError('网络连接失败，请检查网络')
    } else {
      // 其他错误
      notificationStore.showError('请求配置错误')
    }

    return Promise.reject(error)
  }
```

```typescript
    this.logError(appError)
    
    // 显示用户友好的错误信息
    this.showUserError(appError)
    
    // 上报错误到服务器
    this.reportError(appError)
  }

  /**
   * 标准化错误对象
   */
  private normalizeError(error: any, context?: string): AppError {
    if (error.isAxiosError) {
      return this.handleAxiosError(error, context)
    }

    if (error instanceof Error) {
      return {
        code: 'CLIENT_ERROR',
        message: error.message,
        details: error.stack,
        timestamp: Date.now(),
        requestId: this.getRequestId(error)
      }
    }

    return {
      code: 'UNKNOWN_ERROR',
      message: '未知错误',
      details: error,
      timestamp: Date.now()
    }
  }

  /**
   * 处理Axios错误
   */
  private handleAxiosError(error: any, context?: string): AppError {
    const { response, request, config } = error
    
    if (response) {
      // 服务器响应错误
      return {
        code: response.data?.code || `HTTP_${response.status}`,
        message: response.data?.message || this.getHttpErrorMessage(response.status),
        details: response.data,
        timestamp: Date.now(),
        requestId: config?.headers?.['X-Request-ID']
      }
    } else if (request) {
      // 网络错误
      return {
        code: 'NETWORK_ERROR',
        message: '网络连接失败',
        details: { url: config?.url, method: config?.method },
        timestamp: Date.now(),
        requestId: config?.headers?.['X-Request-ID']
      }
    } else {
      // 请求配置错误
      return {
        code: 'REQUEST_ERROR',
        message: '请求配置错误',
        details: error.message,
        timestamp: Date.now()
      }
    }
  }

  /**
   * 获取HTTP错误消息
   */
  private getHttpErrorMessage(status: number): string {
    const messages: Record<number, string> = {
      400: '请求参数错误',
      401: '身份验证失败',
      403: '权限不足',
      404: '资源不存在',
      405: '请求方法不允许',
      408: '请求超时',
      409: '资源冲突',
      422: '请求数据验证失败',
      429: '请求过于频繁',
      500: '服务器内部错误',
      502: '网关错误',
      503: '服务不可用',
      504: '网关超时'
    }
    
    return messages[status] || '请求失败'
  }

  /**
   * 记录错误日志
   */
  private logError(error: AppError): void {
    // 添加到错误队列
    this.errorQueue.push(error)
    
    // 限制队列大小
    if (this.errorQueue.length > this.maxQueueSize) {
      this.errorQueue.shift()
    }
    
    // 控制台输出
    console.error(`[${error.code}] ${error.message}`, error.details)
    
    // 本地存储
    try {
      const errors = JSON.parse(localStorage.getItem('error_logs') || '[]')
      errors.push(error)
      localStorage.setItem('error_logs', JSON.stringify(errors.slice(-50))) // 保留最近50个
    } catch (e) {
      console.warn('Failed to save error log to localStorage:', e)
    }
  }

  /**
   * 显示用户友好的错误信息
   */
  private showUserError(error: AppError): void {
    const { code, message } = error
    
    // 根据错误类型选择显示方式
    switch (code) {
      case 'NETWORK_ERROR':
        ElNotification({
          title: '网络错误',
          message: '请检查网络连接后重试',
          type: 'error',
          duration: 5000
        })
        break
        
      case 'TOKEN_EXPIRED':
        ElMessage({
          message: '登录已过期，请重新登录',
          type: 'warning'
        })
        break
        
      case 'HTTP_429':
        ElMessage({
          message: '操作过于频繁，请稍后再试',
          type: 'warning'
        })
        break
        
      default:
        ElMessage({
          message: message || '操作失败',
          type: 'error',
          duration: 3000
        })
    }
  }

  /**
   * 上报错误到服务器
   */
  private async reportError(error: AppError): void {
    // 只在生产环境上报
    if (import.meta.env.MODE !== 'production') return
    
    try {
      const errorReport = {
        ...error,
        userAgent: navigator.userAgent,
        url: window.location.href,
        userId: this.getCurrentUserId(),
        version: import.meta.env.VITE_APP_VERSION
      }
      
      await fetch('/api/v1/log/error', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json'
        },
        body: JSON.stringify(errorReport)
      })
    } catch (e) {
      console.warn('Failed to report error:', e)
    }
  }

  /**
   * 获取请求ID
   */
  private getRequestId(error: any): string | undefined {
    return error.config?.headers?.['X-Request-ID']
  }

  /**
   * 获取当前用户ID
   */
  private getCurrentUserId(): string | null {
    try {
      const authStore = JSON.parse(localStorage.getItem('auth-store') || '{}')
      return authStore.user?.id || null
    } catch {
      return null
    }
  }

  /**
   * 获取错误统计
   */
  getErrorStats(): any {
    const stats = {
      total: this.errorQueue.length,
      byCode: {} as Record<string, number>,
      recent: this.errorQueue.slice(-10)
    }
    
    this.errorQueue.forEach(error => {
      stats.byCode[error.code] = (stats.byCode[error.code] || 0) + 1
    })
    
    return stats
  }

  /**
   * 清空错误队列
   */
  clearErrors(): void {
    this.errorQueue = []
    localStorage.removeItem('error_logs')
  }
}
```

```typescript
// src/utils/cacheManager.ts
import Redis from 'ioredis'

export interface CacheOptions {
  ttl?: number // 生存时间（秒）
  prefix?: string // 键前缀
  tags?: string[] // 缓存标签
}

export interface CacheItem<T> {
  data: T
  timestamp: number
  ttl?: number
  tags?: string[]
}

export class CacheManager {
  private static instance: CacheManager
  private redis: Redis
  private memoryCache: Map<string, CacheItem<any>> = new Map()
  private defaultTTL = 3600 // 默认1小时
  private maxMemoryItems = 1000

  constructor(redis: Redis) {
    this.redis = redis
    this.startMemoryCleanup()
  }

  static getInstance(redis?: Redis): CacheManager {
    if (!CacheManager.instance) {
      if (!redis) {
        throw new Error('Redis instance is required for first initialization')
      }
      CacheManager.instance = new CacheManager(redis)
    }
    return CacheManager.instance
  }

  /**
   * 设置缓存
   */
  async set<T>(key: string, data: T, options: CacheOptions = {}): Promise<void> {
    const { ttl = this.defaultTTL, prefix = 'cache:', tags = [] } = options
    const fullKey = prefix + key
    
    const cacheItem: CacheItem<T> = {
      data,
      timestamp: Date.now(),
      ttl,
      tags
    }

    // 内存缓存
    this.setMemoryCache(fullKey, cacheItem)
    
    // Redis缓存
    try {
      const serialized = JSON.stringify(cacheItem)
      if (ttl > 0) {
        await this.redis.setex(fullKey, ttl, serialized)
      } else {
        await this.redis.set(fullKey, serialized)
      }
      
      // 设置标签索引
      if (tags.length > 0) {
        await this.setTagIndex(fullKey, tags)
      }
    } catch (error) {
      console.warn('Redis cache set failed:', error)
    }
  }

  /**
   * 获取缓存
   */
  async get<T>(key: string, options: CacheOptions = {}): Promise<T | null> {
    const { prefix = 'cache:' } = options
    const fullKey = prefix + key

    // 先从内存缓存获取
    const memoryItem = this.getMemoryCache(fullKey)
    if (memoryItem !== null) {
      return memoryItem
    }

    // 从Redis获取
    try {
      const serialized = await this.redis.get(fullKey)
      if (serialized) {
        const cacheItem: CacheItem<T> = JSON.parse(serialized)
        
        // 检查是否过期
        if (this.isExpired(cacheItem)) {
          await this.delete(fullKey)
          return null
        }
        
        // 更新内存缓存
        this.setMemoryCache(fullKey, cacheItem)
        
        return cacheItem.data
      }
    } catch (error) {
      console.warn('Redis cache get failed:', error)
    }

    return null
  }

  /**
   * 删除缓存
   */
  async delete(key: string, options: CacheOptions = {}): Promise<void> {
    const { prefix = 'cache:' } = options
    const fullKey = prefix + key

    // 删除内存缓存
    this.memoryCache.delete(fullKey)
    
    // 删除Redis缓存
    try {
      await this.redis.del(fullKey)
    } catch (error) {
      console.warn('Redis cache delete failed:', error)
    }
  }

  /**
   * 批量删除（通过标签）
   */
  async deleteByTag(tag: string): Promise<void> {
    try {
      const tagKey = `tag:${tag}`
      const keys = await this.redis.smembers(tagKey)
      
      if (keys.length > 0) {
        // 删除所有相关缓存
        await this.redis.del(...keys)
        
        // 删除标签索引
        await this.redis.del(tagKey)
      }
      
      // 删除内存缓存中的相关项
      for (const [key, item] of this.memoryCache.entries()) {
        if (item.tags?.includes(tag)) {
          this.memoryCache.delete(key)
        }
      }
    } catch (error) {
      console.warn('Delete by tag failed:', error)
    }
  }

  /**
   * 检查缓存是否存在
   */
  async exists(key: string, options: CacheOptions = {}): Promise<boolean> {
    const { prefix = 'cache:' } = options
    const fullKey = prefix + key

    // 检查内存缓存
    if (this.memoryCache.has(fullKey)) {
      const item = this.memoryCache.get(fullKey)!
      return !this.isExpired(item)
    }

    // 检查Redis缓存
    try {
      const exists = await this.redis.exists(fullKey)
      return exists > 0
    } catch (error) {
      console.warn('Redis cache exists check failed:', error)
      return false
    }
  }

  /**
   * 设置内存缓存
   */
  private setMemoryCache<T>(key: string, item: CacheItem<T>): void {
    // 如果缓存已满，删除最旧的项
    if (this.memoryCache.size >= this.maxMemoryItems) {
      const oldestKey = this.memoryCache.keys().next().value
      this.memoryCache.delete(oldestKey)
    }
    
    this.memoryCache.set(key, item)
  }

  /**
   * 获取内存缓存
   */
  private getMemoryCache<T>(key: string): T | null {
    const item = this.memoryCache.get(key)
    if (!item) return null
    
    if (this.isExpired(item)) {
      this.memoryCache.delete(key)
      return null
    }
    
    return item.data
  }

  /**
   * 检查缓存是否过期
   */
  private isExpired(item: CacheItem<any>): boolean {
    if (!item.ttl) return false
    return Date.now() - item.timestamp > item.ttl * 1000
  }

  /**
   * 设置标签索引
   */
  private async setTagIndex(key: string, tags: string[]): Promise<void> {
    for (const tag of tags) {
      const tagKey = `tag:${tag}`
      await this.redis.sadd(tagKey, key)
      
      // 设置标签索引的过期时间
      await this.redis.expire(tagKey, 7 * 24 * 3600) // 7天
    }
  }

  /**
   * 启动内存缓存清理
   */
  private startMemoryCleanup(): void {
    setInterval(() => {
      for (const [key, item] of this.memoryCache.entries()) {
        if (this.isExpired(item)) {
          this.memoryCache.delete(key)
        }
      }
    }, 60000) // 每分钟清理一次
  }

  /**
   * 获取缓存统计
   */
  async getStats(): Promise<any> {
    try {
      const redisInfo = await this.redis.info('memory')
      const memoryUsage = this.memoryCache.size
      
      return {
        memoryCache: {
          items: memoryUsage,
          maxSize: this.maxMemoryItems,
          usage: (memoryUsage / this.maxMemoryItems * 100).toFixed(2) + '%'
        },
        redis: {
          usedMemory: this.parseRedisMemory(redisInfo)
        }
      }
    } catch (error) {
      console.warn('Get cache stats failed:', error)
      return null
    }
  }

  /**
   * 解析Redis内存信息
   */
  private parseRedisMemory(info: string): any {
    const lines = info.split('\r\n')
    const memory: any = {}
    
    for (const line of lines) {
      if (line.startsWith('used_memory_human:')) {
        memory.used = line.split(':')[1]
      } else if (line.startsWith('used_memory_peak_human:')) {
        memory.peak = line.split(':')[1]
      }
    }
    
    return memory
  }

  /**
   * 清空所有缓存
   */
  async clear(): Promise<void> {
    this.memoryCache.clear()
    
    try {
      // 只删除缓存相关的键
      const keys = await this.redis.keys('cache:*')
      if (keys.length > 0) {
        await this.redis.del(...keys)
      }
      
      const tagKeys = await this.redis.keys('tag:*')
      if (tagKeys.length > 0) {
        await this.redis.del(...tagKeys)
      }
    } catch (error) {
      console.warn('Clear cache failed:', error)
    }
  }
}

// 缓存装饰器
export function Cacheable(options: CacheOptions = {}) {
  return function (target: any, propertyName: string, descriptor: PropertyDescriptor) {
    const method = descriptor.value
    
    descriptor.value = async function (...args: any[]) {
      const cacheManager = CacheManager.getInstance()
      const key = `${target.constructor.name}:${propertyName}:${JSON.stringify(args)}`
      
      // 尝试从缓存获取
      const cached = await cacheManager.get(key)
      if (cached !== null) {
        return cached
      }
      
      // 执行原方法
      const result = await method.apply(this, args)
      
      // 缓存结果
      await cacheManager.set(key, result, options)
      
      return result
    }
  }
}

// 缓存失效装饰器
export function CacheEvict(tags: string[] = []) {
  return function (target: any, propertyName: string, descriptor: PropertyDescriptor) {
    const method = descriptor.value
    
    descriptor.value = async function (...args: any[]) {
      const cacheManager = CacheManager.getInstance()
      
      // 执行原方法
      const result = await method.apply(this, args)
      
      // 清除相关缓存
      for (const tag of tags) {
        await cacheManager.deleteByTag(tag)
      }
      
      return result
    }
  }
}
```

### 1. 多端同步系统
```typescript
// src/services/syncManager.ts
import { useAuthStore } from '@/store/modules/auth'
import { IndexedDBManager } from '@/utils/indexedDB'

export interface SyncOperation {
  id: string
  type: 'create' | 'update' | 'delete'
  entity: string
  entityId: string
  data: any
  timestamp: number
  deviceId: string
  status: 'pending' | 'syncing' | 'completed' | 'failed'
}

export interface SyncConflict {
  operationId: string
  localData: any
  remoteData: any
  conflictType: 'version' | 'data' | 'delete'
  timestamp: number
}

export class SyncManager {
  private static instance: SyncManager
  private dbManager: IndexedDBManager
  private deviceId: string
  private syncInProgress = false
  private syncQueue: SyncOperation[] = []
  private conflicts: SyncConflict[] = []

  constructor() {
    this.dbManager = new IndexedDBManager()
    this.deviceId = this.getOrCreateDeviceId()
    this.initializeSync()
  }

  static getInstance(): SyncManager {
    if (!SyncManager.instance) {
      SyncManager.instance = new SyncManager()
    }
    return SyncManager.instance
  }

  /**
   * 初始化同步
   */
  private async initializeSync(): Promise<void> {
    // 加载待同步操作
    await this.loadSyncQueue()
    
    // 加载冲突
    await this.loadConflicts()
    
    // 监听在线状态
    window.addEventListener('online', () => this.startSync())
    window.addEventListener('offline', () => this.stopSync())
    
    // 定期同步
    setInterval(() => {
      if (navigator.onLine && !this.syncInProgress) {
        this.startSync()
      }
    }, 30000) // 30秒检查一次
  }

  /**
   * 记录操作
   */
  async recordOperation(operation: Omit<SyncOperation, 'id' | 'timestamp' | 'deviceId' | 'status'>): Promise<void> {
    const syncOp: SyncOperation = {
      ...operation,
      id: this.generateOperationId(),
      timestamp: Date.now(),
      deviceId: this.deviceId,
      status: 'pending'
    }

    // 添加到同步队列
    this.syncQueue.push(syncOp)
    
    // 保存到本地数据库
    await this.dbManager.add('syncQueue', syncOp)
    
    // 如果在线，立即同步
    if (navigator.onLine && !this.syncInProgress) {
      this.startSync()
    }
  }

  /**
   * 开始同步
   */
  async startSync(): Promise<void> {
    if (this.syncInProgress || this.syncQueue.length === 0) return

    this.syncInProgress = true

    try {
      // 获取远程数据的时间戳
      const remoteTimestamps = await this.getRemoteTimestamps()
      
      // 处理同步队列
      const operationsToSync = this.syncQueue.filter(op => op.status === 'pending')
      
      for (const operation of operationsToSync) {
        await this.syncOperation(operation, remoteTimestamps)
      }
      
      // 拉取远程更新
      await this.pullRemoteUpdates()
      
    } catch (error) {
      console.error('Sync failed:', error)
    } finally {
      this.syncInProgress = false
    }
  }

  /**
   * 停止同步
   */
  stopSync(): void {
    this.syncInProgress = false
  }

  /**
   * 同步单个操作
   */
  private async syncOperation(operation: SyncOperation, remoteTimestamps: Map<string, number>): Promise<void> {
    try {
      operation.status = 'syncing'
      await this.dbManager.update('syncQueue', operation)

      const remoteTimestamp = remoteTimestamps.get(`${operation.entity}:${operation.entityId}`)
      
      // 检查是否有冲突
      if (remoteTimestamp && remoteTimestamp > operation.timestamp) {
        const conflict = await this.detectConflict(operation, remoteTimestamp)
        if (conflict) {
          this.handleConflict(conflict)
          return
        }
      }

      // 执行同步
      await this.executeSyncOperation(operation)
      
      // 更新状态
      operation.status = 'completed'
      await this.dbManager.update('syncQueue', operation)
      
      // 从队列中移除
      const index = this.syncQueue.findIndex(op => op.id === operation.id)
      if (index > -1) {
        this.syncQueue.splice(index, 1)
      }
      
    } catch (error) {
      console.error('Sync operation failed:', operation, error)
      operation.status = 'failed'
      await this.dbManager.update('syncQueue', operation)
    }
  }

  /**
   * 执行同步操作
   */
  private async executeSyncOperation(operation: SyncOperation): Promise<void> {
    const authStore = useAuthStore()
    
    switch (operation.type) {
      case 'create':
        await apiClient.post(`/sync/${operation.entity}`, operation.data, {
          headers: {
            'Authorization': `Bearer ${authStore.token}`,
            'X-Device-ID': this.deviceId,
            'X-Operation-ID': operation.id,
            'X-Timestamp': operation.timestamp.toString()
          }
        })
        break
        
      case 'update':
        await apiClient.put(`/sync/${operation.entity}/${operation.entityId}`, operation.data, {
          headers: {
            'Authorization': `Bearer ${authStore.token}`,
            'X-Device-ID': this.deviceId,
            'X-Operation-ID': operation.id,
            'X-Timestamp': operation.timestamp.toString()
          }
        })
        break
        
      case 'delete':
        await apiClient.delete(`/sync/${operation.entity}/${operation.entityId}`, {
          headers: {
            'Authorization': `Bearer ${authStore.token}`,
            'X-Device-ID': this.deviceId,
            'X-Operation-ID': operation.id,
            'X-Timestamp': operation.timestamp.toString()
          }
        })
        break
    }
  }

  /**
   * 检测冲突
   */
  private async detectConflict(operation: SyncOperation, remoteTimestamp: number): Promise<SyncConflict | null> {
    // 获取远程数据
    try {
      const response = await apiClient.get(`/sync/${operation.entity}/${operation.entityId}`, {
        headers: {
          'Authorization': `Bearer ${useAuthStore().token}`,
          'X-Device-ID': this.deviceId
        }
      })
      
      const remoteData = response.data
      
      // 检查数据是否有差异
      if (this.hasDataConflict(operation.data, remoteData)) {
        return {
          operationId: operation.id,
          localData: operation.data,
          remoteData,
          conflictType: 'data',
          timestamp: Date.now()
        }
      }
    } catch (error) {
      // 如果远程数据不存在，可能是删除冲突
      if (error.response?.status === 404) {
        return {
          operationId: operation.id,
          localData: operation.data,
          remoteData: null,
          conflictType: 'delete',
          timestamp: Date.now()
        }
      }
    }
    
    return null
  }

  /**
   * 处理冲突
   */
  private async handleConflict(conflict: SyncConflict): Promise<void> {
    this.conflicts.push(conflict)
    await this.dbManager.add('conflicts', conflict)
    
    // 显示冲突通知
    this.showConflictNotification(conflict)
  }

  /**
   * 解决冲突
   */
  async resolveConflict(conflictId: string, resolution: 'local' | 'remote' | 'merge'): Promise<void> {
    const conflictIndex = this.conflicts.findIndex(c => c.operationId === conflictId)
    if (conflictIndex === -1) return

    const conflict = this.conflicts[conflictIndex]
    const operation = this.syncQueue.find(op => op.id === conflict.operationId)
    
    if (!operation) return

    try {
      switch (resolution) {
        case 'local':
          // 强制使用本地数据
          await this.forceSyncOperation(operation)
          break
          
        case 'remote':
          // 使用远程数据，更新本地
          await this.updateLocalData(operation.entity, operation.entityId, conflict.remoteData)
          break
          
        case 'merge':
          // 合并数据
          const mergedData = this.mergeData(conflict.localData, conflict.remoteData)
          operation.data = mergedData
          await this.executeSyncOperation(operation)
          break
      }
      
      // 移除冲突
      this.conflicts.splice(conflictIndex, 1)
      await this.dbManager.delete('conflicts', conflictId)
      
      // 更新操作状态
      operation.status = 'completed'
      await this.dbManager.update('syncQueue', operation)
      
    } catch (error) {
      console.error('Resolve conflict failed:', error)
    }
  }

  /**
   * 拉取远程更新
   */
  private async pullRemoteUpdates(): Promise<void> {
    try {
      const lastSyncTime = await this.getLastSyncTime()
      const response = await apiClient.get('/sync/updates', {
        params: {
          since: lastSyncTime,
          deviceId: this.deviceId
        },
        headers: {
          'Authorization': `Bearer ${useAuthStore().token}`
        }
      })
      
      const updates = response.data
      
      for (const update of updates) {
        await this.applyRemoteUpdate(update)
      }
      
      // 更新最后同步时间
      await this.setLastSyncTime(Date.now())
      
    } catch (error) {
      console.error('Pull remote updates failed:', error)
    }
  }

  /**
   * 应用远程更新
   */
  private async applyRemoteUpdate(update: any): Promise<void> {
    const { entity, entityId, type, data, timestamp } = update
    
    switch (type) {
      case 'create':
        await this.dbManager.add(entity, { ...data, id: entityId, synced: true })
        break
        
      case 'update':
        await this.dbManager.update(entity, entityId, { ...data, synced: true })
        break
        
      case 'delete':
        await this.dbManager.delete(entity, entityId)
        break
    }
  }

  /**
   * 获取远程时间戳
   */
  private async getRemoteTimestamps(): Promise<Map<string, number>> {
    try {
      const response = await apiClient.get('/sync/timestamps', {
        headers: {
          'Authorization': `Bearer ${useAuthStore().token}`,
          'X-Device-ID': this.deviceId
        }
      })
      
      const timestamps = new Map<string, number>()
      response.data.forEach((item: any) => {
        timestamps.set(`${item.entity}:${item.entityId}`, item.timestamp)
      })
      
      return timestamps
    } catch (error) {
      console.warn('Get remote timestamps failed:', error)
      return new Map()
    }
  }

  /**
   * 获取或创建设备ID
   */
  private getOrCreateDeviceId(): string {
    let deviceId = localStorage.getItem('device_id')
    if (!deviceId) {
      deviceId = this.generateDeviceId()
      localStorage.setItem('device_id', deviceId)
    }
    return deviceId
  }

  /**
   * 生成设备ID
   */
  private generateDeviceId(): string {
    return Date.now().toString(36) + Math.random().toString(36).substr(2)
  }

  /**
   * 生成操作ID
   */
  private generateOperationId(): string {
    return 'op_' + Date.now().toString(36) + Math.random().toString(36).substr(2)
  }

  /**
   * 检查数据冲突
   */
  private hasDataConflict(localData: any, remoteData: any): boolean {
    // 简单的冲突检测逻辑
    return JSON.stringify(localData) !== JSON.stringify(remoteData)
  }

  /**
   * 合并数据
   */
  private mergeData(localData: any, remoteData: any): any {
    // 简单的合并策略：远程数据优先，保留本地特有的字段
    return {
      ...remoteData,
      ...localData,
      id: remoteData.id || localData.id,
      updatedAt: Date.now()
    }
  }

  /**
   * 强制同步操作
   */
  private async forceSyncOperation(operation: SyncOperation): Promise<void> {
    const authStore = useAuthStore()
    
    await apiClient.post(`/sync/${operation.entity}/force`, operation.data, {
      headers: {
        'Authorization': `Bearer ${authStore.token}`,
        'X-Device-ID': this.deviceId,
        'X-Operation-ID': operation.id,
        'X-Timestamp': operation.timestamp.toString()
      }
    })
  }

  /**
   * 更新本地数据
   */
  private async updateLocalData(entity: string, entityId: string, data: any): Promise<void> {
    if (data) {
      await this.dbManager.update(entity, entityId, { ...data, synced: true })
    } else {
      await this.dbManager.delete(entity, entityId)
    }
  }

  /**
   * 显示冲突通知
   */
  private showConflictNotification(conflict: SyncConflict): void {
    // 这里可以集成通知组件
    console.warn('Sync conflict detected:', conflict)
  }

  /**
   * 加载同步队列
   */
  private async loadSyncQueue(): Promise<void> {
    try {
      const operations = await this.dbManager.getAll('syncQueue')
      this.syncQueue = operations.filter(op => op.status !== 'completed')
    } catch (error) {
      console.warn('Load sync queue failed:', error)
    }
  }

  /**
   * 加载冲突
   */
  private async loadConflicts(): Promise<void> {
    try {
      this.conflicts = await this.dbManager.getAll('conflicts')
    } catch (error) {
      console.warn('Load conflicts failed:', error)
    }
  }

  /**
   * 获取最后同步时间
   */
  private async getLastSyncTime(): Promise<number> {
    try {
      const setting = await this.dbManager.get('settings', 'lastSyncTime')
      return setting?.value || 0
    } catch (error) {
      return 0
    }
  }

  /**
   * 设置最后同步时间
   */
  private async setLastSyncTime(timestamp: number): Promise<void> {
    try {
      await this.dbManager.put('settings', {
        id: 'lastSyncTime',
        value: timestamp
      })
    } catch (error) {
      console.warn('Set last sync time failed:', error)
    }
  }

  /**
   * 获取同步状态
   */
  getSyncStatus(): any {
    return {
      inProgress: this.syncInProgress,
      queueLength: this.syncQueue.length,
      conflictsCount: this.conflicts.length,
      deviceId: this.deviceId
    }
  }
}
```
