// client/src/components/player/WaveformPlayer.tsx - 波形播放器
import React, { useRef, useEffect, useState } from 'react'
import { Button, Group, Progress, Box } from '@mantine/core'
import { Howl } from 'howler'
import * as d3 from 'd3'

interface WaveformPlayerProps {
  audioUrl: string
  title: string
  artist: string
}

export const WaveformPlayer: React.FC<WaveformPlayerProps> = ({ audioUrl, title, artist }) => {
  const [isPlaying, setIsPlaying] = useState(false)
  const [progress, setProgress] = useState(0)
  const [duration, setDuration] = useState(0)
  const waveformRef = useRef<HTMLDivElement>(null)
  const howlRef = useRef<Howl | null>(null)

  useEffect(() => {
    if (audioUrl) {
      howlRef.current = new Howl({
        src: [audioUrl],
        html5: true,
        onload: () => setDuration(howlRef.current?.duration() || 0),
        onplay: () => setIsPlaying(true),
        onpause: () => setIsPlaying(false),
        onend: () => setIsPlaying(false)
      })

      // 生成波形可视化
      generateWaveform()
    }

    return () => {
      if (howlRef.current) {
        howlRef.current.unload()
      }
    }
  }, [audioUrl])

  const generateWaveform = () => {
    if (!waveformRef.current) return

    const width = waveformRef.current.clientWidth
    const height = 100
    const data = Array.from({ length: 100 }, () => Math.random() * 100)

    const svg = d3.select(waveformRef.current)
      .append('svg')
      .attr('width', width)
      .attr('height', height)

    const x = d3.scaleLinear()
      .domain([0, data.length])
      .range([0, width])

    const y = d3.scaleLinear()
      .domain([0, 100])
      .range([height, 0])

    svg.selectAll('rect')
      .data(data)
      .enter()
      .append('rect')
      .attr('x', (d, i) => x(i))
      .attr('y', d => y(d))
      .attr('width', width / data.length - 1)
      .attr('height', d => height - y(d))
      .attr('fill', '#8b5cf6')
      .attr('opacity', 0.8)
  }

  const togglePlay = () => {
    if (howlRef.current) {
      if (isPlaying) {
        howlRef.current.pause()
      } else {
        howlRef.current.play()
        updateProgress()
      }
    }
  }

  const updateProgress = () => {
    if (howlRef.current) {
      const current = howlRef.current.seek() as number
      setProgress((current / duration) * 100)
      
      if (isPlaying) {
        requestAnimationFrame(updateProgress)
      }
    }
  }

  return (
    <Box className="bg-gray-900 rounded-xl p-6">
      <div className="mb-4">
        <h3 className="text-xl font-bold text-white">{title}</h3>
        <p className="text-gray-400">{artist}</p>
      </div>

      <div ref={waveformRef} className="mb-4 bg-gray-800 rounded-lg p-2" />

      <Progress 
        value={progress} 
        size="sm" 
        className="mb-4"
        color="violet"
      />

      <Group position="center">
        <Button
          onClick={togglePlay}
          variant="filled"
          color="violet"
          size="lg"
          leftIcon={isPlaying ? '⏸' : '▶'}
        >
          {isPlaying ? '暂停' : '播放'}
        </Button>
      </Group>
    </Box>
  )
}

// server/src/modules/ai/lyrics/lyrics.service.ts - AI歌词服务
import { Injectable } from '@nestjs/common'
import { ConfigService } from '@nestjs/config'
import axios from 'axios'

@Injectable()
export class LyricsService {
  constructor(private configService: ConfigService) {}

  async generateLyrics(params: {
    theme: string
    emotion: number
    style: string
    keywords: string[]
  }) {
    const prompt = this.buildPrompt(params)
    
    try {
      const response = await axios.post(
        'https://api.openai.com/v1/chat/completions',
        {
          model: 'gpt-4',
          messages: [
            {
              role: 'system',
              content: '你是一个专业的歌词创作AI，擅长创作富有情感和诗意的歌词。'
            },
            {
              role: 'user',
              content: prompt
            }
          ],
          max_tokens: 1000,
          temperature: 0.8
        },
        {
          headers: {
            'Authorization': `Bearer ${this.configService.get('OPENAI_API_KEY')}`,
            'Content-Type': 'application/json'
          }
        }
      )

      return {
        lyrics: response.data.choices[0].message.content,
        metadata: {
          theme: params.theme,
          emotion: params.emotion,
          style: params.style,
          keywords: params.keywords,
          generatedAt: new Date()
        }
      }
    } catch (error) {
      throw new Error('歌词生成失败')
    }
  }

  private buildPrompt(params: {
    theme: string
    emotion: number
    style: string
    keywords: string[]
  }): string {
    const emotionMap = {
      0.2: '平静温柔',
      0.5: '适中抒情',
      0.8: '激昂热烈'
    }

    return `
请创作一首${params.style}风格的歌词，主题是"${params.theme}"，
情感基调为${emotionMap[params.emotion as keyof typeof emotionMap]}。

请融入以下关键词：${params.keywords.join('、')}

要求：
1. 结构完整，包含主歌、副歌、桥段
2. 韵律优美，朗朗上口
3. 情感真挚，富有画面感
4. 长度控制在200-300字

请直接输出歌词内容，不需要其他说明。
    `
  }
}

// server/src/modules/energy/energy.service.ts - 能量系统服务
import { Injectable } from '@nestjs/common'
import { InjectRedis } from '@nestjs/redis'
import Redis from 'ioredis'

@Injectable()
export class EnergyService {
  constructor(@InjectRedis() private readonly redis: Redis) {}

  async getUserEnergy(userId: string): Promise<number> {
    const energy = await this.redis.zscore('user_energy', userId)
    return energy ? parseFloat(energy) : 0
  }

  async boostEnergy(userId: string, amount: number, reason: string): Promise<void> {
    const multi = this.redis.multi()
    
    // 增加用户能量
    multi.zincrby('user_energy', amount, userId)
    
    // 记录能量变动历史
    multi.lpush(
      `energy_history:${userId}`,
      JSON.stringify({
        amount,
        reason,
        timestamp: new Date().toISOString()
      })
    )
    
    // 更新排行榜
    multi.zadd('energy_leaderboard', amount, userId)
    
    await multi.exec()
  }

  async getLeaderboard(limit: number = 10): Promise<Array<{userId: string, energy: number}>> {
    const leaderboard = await this.redis.zrevrange(
      'energy_leaderboard',
      0,
      limit - 1,
      'WITHSCORES'
    )
    
    return leaderboard.map((item, index) => ({
      userId: item,
      energy: parseFloat(leaderboard[index + 1] as string),
      rank: index + 1
    })).filter((_, index) => index % 2 === 0)
  }

  async processEnergyBoost(userId: string, action: string): Promise<number> {
    const boostMap = {
      'create_song': 50,
      'daily_login': 10,
      'share_song': 20,
      'comment': 5,
      'like': 2
    }

    const boostAmount = boostMap[action as keyof typeof boostMap] || 0
    
    if (boostAmount > 0) {
      await this.boostEnergy(userId, boostAmount, `完成${action}任务`)
    }

    return boostAmount
  }
}

# ai_services/lyric_generator/app.py - 歌词生成微服务
from flask import Flask, request, jsonify
from transformers import GPT2LMHeadModel, GPT2Tokenizer
import torch

app = Flask(__name__)

# 加载微调后的模型
tokenizer = GPT2Tokenizer.from_pretrained('./models/lyric-gpt2')
model = GPT2LMHeadModel.from_pretrained('./models/lyric-gpt2')

@app.route('/generate', methods=['POST'])
def generate_lyrics():
    data = request.json
    theme = data.get('theme', '')
    emotion = data.get('emotion', 0.5)
    keywords = data.get('keywords', [])
    
    # 构建提示词
    prompt = build_prompt(theme, emotion, keywords)
    
    # 生成歌词
    inputs = tokenizer.encode(prompt, return_tensors='pt')
    outputs = model.generate(
        inputs,
        max_length=500,
        temperature=0.8,
        num_return_sequences=1,
        pad_token_id=tokenizer.eos_token_id
    )
    
    lyrics = tokenizer.decode(outputs[0], skip_special_tokens=True)
    
    return jsonify({
        'lyrics': lyrics,
        'metadata': {
            'theme': theme,
            'emotion': emotion,
            'keywords': keywords
        }
    })

def build_prompt(theme: str, emotion: float, keywords: list) -> str:
    emotion_desc = "平静" if emotion < 0.3 else "激昂" if emotion > 0.7 else "适中"
    keyword_str = "、".join(keywords)
    
    return f"""
主题：{theme}
情感：{emotion_desc}
关键词：{keyword_str}

歌词：
"""

if __name__ == '__main__':
    app.run(host='0.0.0.0', port=5000)

      const response = await api.post('/api/v1/auth/refresh', { refreshToken })
      const { token: newToken, refreshToken: newRefreshToken } = response.data
      
      localStorage.setItem('token', newToken)
      localStorage.setItem('refreshToken', newRefreshToken)
      
      return { token: newToken, refreshToken: newRefreshToken }
    } catch (error: any) {
      return rejectWithValue('Token刷新失败')
    }
  }

interface ProtectedRouteProps {
  children: React.ReactNode
  requiredRole?: string[]
}

export const ProtectedRoute: React.FC<ProtectedRouteProps> = ({ 
  children, 
  requiredRole = [] 
}) => {
  const dispatch = useDispatch<AppDispatch>()
  const location = useLocation()
  const { isAuthenticated, isLoading, user, token } = useSelector(
    (state: RootState) => state.auth
  )
  const [isRefreshing, setIsRefreshing] = useState(false)

  useEffect(() => {
    const checkAuth = async () => {
      if (token && !isAuthenticated && !isLoading) {
        setIsRefreshing(true)
        try {
          await dispatch(refreshAuthToken()).unwrap()
        } catch (error) {
          console.error('Token refresh failed:', error)
        } finally {
          setIsRefreshing(false)
        }
      }
    }

    checkAuth()
  }, [token, isAuthenticated, isLoading, dispatch])

  if (isLoading || isRefreshing) {
    return (
      <div className="flex items-center justify-center min-h-screen">
        <Loader size="xl" color="violet" />
      </div>
    )
  }

  if (!isAuthenticated) {
    return <Navigate to="/login" state={{ from: location }} replace />
  }

  // 角色权限检查
  if (requiredRole.length > 0 && user && !requiredRole.includes(user.role)) {
    return <Navigate to="/unauthorized" replace />
  }

  return <>{children}</>
}

          return this.handleForbidden(data)
        case 404:

          return this.handleTooManyRequests(data)

          return this.handleServerError(data)

          return this.handleUnknownError(data)

import { useState, useEffect, useCallback } from 'react'
import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query'

interface CacheOptions {
  ttl?: number // 生存时间（毫秒）
  storage?: 'localStorage' | 'sessionStorage' | 'memory'
}

interface CacheItem<T> {
  data: T
  timestamp: number
  ttl: number
}

export const useCache = <T>(
  key: string,
  fetcher: () => Promise<T>,
  options: CacheOptions = {}
) => {
  const {
    ttl = 5 * 60 * 1000, // 默认5分钟
    storage = 'localStorage'
  } = options

  const queryClient = useQueryClient()

  // 获取缓存数据
  const getCacheData = useCallback((): T | null => {
    try {
      if (storage === 'memory') {
        const memoryCache = queryClient.getQueryData<T>([key])
        return memoryCache || null
      }

      const storageObj = window[storage]
      const cached = storageObj.getItem(`cache_${key}`)
      
      if (!cached) return null

      const parsed: CacheItem<T> = JSON.parse(cached)
      const now = Date.now()

      if (now - parsed.timestamp > parsed.ttl) {
        storageObj.removeItem(`cache_${key}`)
        return null
      }

      return parsed.data
    } catch (error) {
      console.error('Cache read error:', error)
      return null
    }
  }, [key, storage, queryClient])

  // 设置缓存数据
  const setCacheData = useCallback((data: T) => {
    try {
      const cacheItem: CacheItem<T> = {
        data,
        timestamp: Date.now(),
        ttl
      }

      if (storage === 'memory') {
        queryClient.setQueryData([key], data)
      } else {
        const storageObj = window[storage]
        storageObj.setItem(`cache_${key}`, JSON.stringify(cacheItem))
      }
    } catch (error) {
      console.error('Cache write error:', error)
    }
  }, [key, ttl, storage, queryClient])

  // 清除缓存
  const clearCache = useCallback(() => {
    try {
      if (storage === 'memory') {
        queryClient.removeQueries([key])
      } else {
        const storageObj = window[storage]
        storageObj.removeItem(`cache_${key}`)
      }
    } catch (error) {
      console.error('Cache clear error:', error)
    }
  }, [key, storage, queryClient])

  // 使用React Query进行数据获取和缓存
  const {
    data,
    isLoading,
    error,
    refetch
  } = useQuery({
    queryKey: [key],
    queryFn: async () => {
      // 先检查缓存
      const cachedData = getCacheData()
      if (cachedData) {
        return cachedData
      }

      // 缓存不存在或已过期，重新获取
      const freshData = await fetcher()
      setCacheData(freshData)
      return freshData
    },
    staleTime: ttl,
    cacheTime: ttl * 2
  })

  // 手动刷新缓存
  const refresh = useCallback(async () => {
    try {
      const freshData = await fetcher()
      setCacheData(freshData)
      return freshData
    } catch (error) {
      console.error('Cache refresh error:', error)
      throw error
    }
  }, [fetcher, setCacheData])

  return {
    data,
    isLoading,
    error,
    refetch,
    refresh,
    clearCache
  }
}

import { api } from './api'

interface SyncOperation {
  id: string
  type: 'create' | 'update' | 'delete'
  entityType: string
  entityId: string
  data: any
  timestamp: number
  deviceId: string
}

interface SyncConflict {
  entityType: string
  entityId: string
  localData: any
  remoteData: any
  timestamp: number
}

export class SyncService {
  private deviceId: string
  private pendingOperations: SyncOperation[] = []
  private isOnline: boolean = navigator.onLine

  constructor() {
    this.deviceId = this.getOrCreateDeviceId()
    this.setupEventListeners()
    this.loadPendingOperations()
  }

  private getOrCreateDeviceId(): string {
    let deviceId = localStorage.getItem('deviceId')
    if (!deviceId) {
      deviceId = `device_${Date.now()}_${Math.random().toString(36).substr(2, 9)}`
      localStorage.setItem('deviceId', deviceId)
    }
    return deviceId
  }

  private setupEventListeners() {
    window.addEventListener('online', () => {
      this.isOnline = true
      this.syncPendingOperations()
    })

    window.addEventListener('offline', () => {
      this.isOnline = false
    })
  }

  private loadPendingOperations() {
    try {
      const stored = localStorage.getItem('pendingSyncOperations')
      if (stored) {
        this.pendingOperations = JSON.parse(stored)
      }
    } catch (error) {
      console.error('Load pending operations error:', error)
    }
  }

  private savePendingOperations() {
    try {
      localStorage.setItem(
        'pendingSyncOperations',
        JSON.stringify(this.pendingOperations)
      )
    } catch (error) {
      console.error('Save pending operations error:', error)
    }
  }

  // 记录同步操作
  public async recordOperation(
    type: 'create' | 'update' | 'delete',
    entityType: string,
    entityId: string,
    data: any
  ) {
    const operation: SyncOperation = {
      id: `${Date.now()}_${Math.random().toString(36).substr(2, 9)}`,
      type,
      entityType,
      entityId,
      data,
      timestamp: Date.now(),
      deviceId: this.deviceId
    }

    this.pendingOperations.push(operation)
    this.savePendingOperations()

    if (this.isOnline) {
      await this.syncPendingOperations()
    }
  }

  // 同步待处理操作
  private async syncPendingOperations() {
    if (!this.isOnline || this.pendingOperations.length === 0) {
      return
    }

    try {
      const operationsToSync = [...this.pendingOperations]
      
      const response = await api.post('/api/v1/sync/record', {
        operations: operationsToSync,
        deviceId: this.deviceId
      })

      const { conflicts, synced } = response.data

      // 处理冲突
      if (conflicts && conflicts.length > 0) {
        await this.handleConflicts(conflicts)
      }

      // 移除已同步的操作
      this.pendingOperations = this.pendingOperations.filter(
        op => !synced.includes(op.id)
      )
      this.savePendingOperations()

    } catch (error) {
      console.error('Sync operations error:', error)
    }
  }

  // 处理同步冲突
  private async handleConflicts(conflicts: SyncConflict[]) {
    for (const conflict of conflicts) {
      const resolution = await this.resolveConflict(conflict)
      
      if (resolution) {
        await api.post('/api/v1/sync/resolve', {
          conflictId: conflict.entityId,
          resolution,
          deviceId: this.deviceId
        })
      }
    }
  }

  // 解决冲突（这里简化处理，实际应该让用户选择）
  private async resolveConflict(conflict: SyncConflict): Promise<any> {
    // 简单策略：使用最新的数据
    if (conflict.remoteData.timestamp > conflict.localData.timestamp) {
      return conflict.remoteData
    } else {
      return conflict.localData
    }
  }

  // 获取同步数据
  public async fetchSyncData(lastSyncTime?: number) {
    try {
      const response = await api.get('/api/v1/sync/data', {
        params: {
          deviceId: this.deviceId,
          lastSyncTime
        }
      })

      return response.data
    } catch (error) {
      console.error('Fetch sync data error:', error)
      throw error
    }
  }

  // 手动触发同步
  public async forceSync() {
    if (this.isOnline) {
      await this.syncPendingOperations()
      const data = await this.fetchSyncData()
      return data
    } else {
      throw new Error('设备离线，无法同步')
    }
  }

  // 获取同步状态
  public getSyncStatus() {
    return {
      isOnline: this.isOnline,
      deviceId: this.deviceId,
      pendingOperations: this.pendingOperations.length,
      lastSyncTime: localStorage.getItem('lastSyncTime')
    }
  }
}

    try {
      const permission = await this.requestPermission()
      if (!permission) {
        return false
      }

      const registration = await navigator.serviceWorker.ready
      const subscription = await registration.pushManager.subscribe({
        userVisibleOnly: true,
        applicationServerKey: this.urlBase64ToUint8Array(
          process.env.REACT_APP_VAPID_PUBLIC_KEY!
        )
      })

      this.subscription = subscription

      // 发送订阅信息到服务器
      await fetch('/api/v1/notification/subscribe', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Authorization': `Bearer ${localStorage.getItem('token')}`
        },
        body: JSON.stringify({
          subscription: subscription.toJSON()
        })
      })

      return true
    } catch (error) {
      console.error('Subscribe failed:', error)
      return false
    }
  }

  // 取消订阅
  public async unsubscribe(): Promise<boolean> {
    if (!this.subscription) {
      return true
    }

    try {
      await this.subscription.unsubscribe()
      this.subscription = null

      // 通知服务器
      await fetch('/api/v1/notification/unsubscribe', {
        method: 'POST',
        headers: {
          'Authorization': `Bearer ${localStorage.getItem('token')}`
        }
      })

      return true
    } catch (error) {
      console.error('Unsubscribe failed:', error)
      return false
    }
  }

  // 获取通知列表
  public async getNotifications() {
    try {
      const response = await fetch('/api/v1/notification/list', {
        headers: {
          'Authorization': `Bearer ${localStorage.getItem('token')}`
        }
      })
      return response.json()
    } catch (error) {
      console.error('Get notifications failed:', error)
      return []
    }
  }

  // 标记通知已读
  public async markAsRead(notificationId: string) {
    try {
      await fetch(`/api/v1/notification/${notificationId}/read`, {
        method: 'POST',
        headers: {
          'Authorization': `Bearer ${localStorage.getItem('token')}`
        }
      })
    } catch (error) {
      console.error('Mark as read failed:', error)
    }
  }

  // 工具方法：VAPID密钥转换
  private urlBase64ToUint8Array(base64String: string) {
    const padding = '='.repeat((4 - base64String.length % 4) % 4)
    const base64 = (base64String + padding)
      .replace(/-/g, '+')
      .replace(/_/g, '/')

    const rawData = window.atob(base64)
    const outputArray = new Uint8Array(rawData.length)

    for (let i = 0; i < rawData.length; ++i) {
      outputArray[i] = rawData.charCodeAt(i)
    }

    return outputArray
  }

  const [totalGifts, setTotalGifts] = useState(0)

  const remoteVideoRef = useRef<HTMLVideoElement>(null)
  const socketRef = useRef<Socket | null>(null)
  const peerConnectionRef = useRef<RTCPeerConnection | null>(null)

    // 初始化WebSocket连接
    socketRef.current = io(`${process.env.REACT_APP_WS_URL}/live`, {
      auth: {
        token: localStorage.getItem('token')
      }
    })

    // 监听事件
    socketRef.current.on('viewer-count', (count: number) => {
      setViewerCount(count)
    })

      setMessages(prev => [...prev.slice(-99), message]) // 保留最新100条
    })

    socketRef.current.on('like', () => {
      setLikes(prev => prev + 1)
    })

    socketRef.current.on('gift', (giftData: any) => {
      setTotalGifts(prev => prev + giftData.value)
    })

    if (isHost) {
      initializeHostStream()
    } else {
      initializeViewer()
    }

    return () => {
      socketRef.current?.disconnect()
      if (peerConnectionRef.current) {
        peerConnectionRef.current.close()
      }
    }
  }, [roomId, isHost])

    try {
      const stream = await navigator.mediaDevices.getUserMedia({
        video: true,
        audio: true
      })

      if (localVideoRef.current) {
        localVideoRef.current.srcObject = stream
      }

      // 创建WebRTC连接
      const peerConnection = new RTCPeerConnection({
        iceServers: [
          { urls: 'stun:stun.l.google.com:19302' }
        ]
      })

      stream.getTracks().forEach(track => {
        peerConnection.addTrack(track, stream)
      })

      peerConnectionRef.current = peerConnection

      console.error('Failed to initialize host stream:', error)

      })

        if (remoteVideoRef.current && event.streams[0]) {
          remoteVideoRef.current.srcObject = event.streams[0]
        }

      console.error('Failed to initialize viewer:', error)
