```typescript
// src/models/Playlist.ts
import { Schema, model, Document } from 'mongoose'
import { User } from './User'
import { Music } from './Music'

export interface IPlaylist extends Document {
  _id: string
  name: string
  description: string
  cover: string
  creator: IUser['_id']
  musics: IMusic['_id'][]
  tags: string[]
  isPublic: boolean
  isCollaborative: boolean
  collaborators: IUser['_id'][]
  playCount: number
  likeCount: number
  duration: number // 总时长（秒）
  trackCount: number // 歌曲数量
  createdAt: Date
  updatedAt: Date
  lastPlayedAt?: Date
}

const PlaylistSchema = new Schema<IPlaylist>({
  name: {
    type: String,
    required: true,
    trim: true,
    maxlength: 100
  },
  description: {
    type: String,
    trim: true,
    maxlength: 500
  },
  cover: {
    type: String,
    default: '/images/default-playlist.jpg'
  },
  creator: {
    type: Schema.Types.ObjectId,
    ref: 'User',
    required: true
  },
  musics: [{
    type: Schema.Types.ObjectId,
    ref: 'Music'
  }],
  tags: [{
    type: String,
    trim: true
  }],
  isPublic: {
    type: Boolean,
    default: true
  },
  isCollaborative: {
    type: Boolean,
    default: false
  },
  collaborators: [{
    type: Schema.Types.ObjectId,
    ref: 'User'
  }],
  playCount: {
    type: Number,
    default: 0
  },
  likeCount: {
    type: Number,
    default: 0
  },
  duration: {
    type: Number,
    default: 0
  },
  trackCount: {
    type: Number,
    default: 0
  },
  createdAt: {
    type: Date,
    default: Date.now
  },
  updatedAt: {
    type: Date,
    default: Date.now
  },
  lastPlayedAt: {
    type: Date
  }
}, {
  timestamps: true,
  toJSON: { virtuals: true },
  toObject: { virtuals: true }
})

// 索引
PlaylistSchema.index({ creator: 1, createdAt: -1 })
PlaylistSchema.index({ tags: 1 })
PlaylistSchema.index({ isPublic: 1, playCount: -1 })
PlaylistSchema.index({ name: 'text', description: 'text' })

// 虚拟字段
PlaylistSchema.virtual('isLiked', {
  ref: 'PlaylistLike',
  localField: '_id',
  foreignField: 'playlist',
  justOne: true
})

// 中间件
PlaylistSchema.pre('save', async function(next) {
  // 更新歌曲数量和总时长
  if (this.isModified('musics')) {
    const musics = await Music.find({ _id: { $in: this.musics } })
    this.trackCount = musics.length
    this.duration = musics.reduce((total, music) => total + music.duration, 0)
  }
  next()
})

export const Playlist = model<IPlaylist>('Playlist', PlaylistSchema)
```

```typescript
// src/services/playlistService.ts
import { Playlist, IPlaylist } from '@/models/Playlist'
import { Music } from '@/models/Music'
import { User } from '@/models/User'
import { PlaylistLike } from '@/models/PlaylistLike'
import { CacheManager } from '@/utils/cacheManager'
import { mongoose } from 'mongoose'

export class PlaylistService {
  private cache: CacheManager

  constructor() {
    this.cache = CacheManager.getInstance()
  }

  /**
   * 创建播放列表
   */
  async createPlaylist(userId: string, playlistData: CreatePlaylistDto): Promise<IPlaylist> {
    const playlist = new Playlist({
      ...playlistData,
      creator: userId,
      musics: playlistData.musicIds || [],
      trackCount: playlistData.musicIds?.length || 0
    })

    await playlist.save()
    await this.populatePlaylist(playlist)

    // 清除用户播放列表缓存
    await this.cache.deleteByTag(`user:${userId}:playlists`)

    return playlist
  }

  /**
   * 获取播放列表详情
   */
  async getPlaylist(playlistId: string, userId?: string): Promise<IPlaylist | null> {
    const cacheKey = `playlist:${playlistId}`
    
    // 尝试从缓存获取
    let playlist = await this.cache.get<IPlaylist>(cacheKey)
    
    if (!playlist) {
      playlist = await Playlist.findById(playlistId)
        .populate('creator', 'username avatar')
        .populate('musics')
        .populate('collaborators', 'username avatar')
      
      if (playlist) {
        // 缓存1小时
        await this.cache.set(cacheKey, playlist, { ttl: 3600 })
      }
    }

    // 检查访问权限
    if (playlist && !playlist.isPublic && playlist.creator.toString() !== userId) {
      const isCollaborator = playlist.collaborators.some(
        collaborator => collaborator.toString() === userId
      )
      if (!isCollaborator) {
        return null
      }
    }

    return playlist
  }

  /**
   * 获取用户播放列表
   */
  async getUserPlaylists(
    userId: string, 
    options: {
      page?: number
      limit?: number
      type?: 'created' | 'liked' | 'collaborative'
    } = {}
  ): Promise<{ playlists: IPlaylist[], total: number }> {
    const { page = 1, limit = 20, type = 'created' } = options
    const cacheKey = `user:${userId}:playlists:${type}:${page}:${limit}`

    // 尝试从缓存获取
    const cached = await this.cache.get(cacheKey)
    if (cached) return cached

    let playlists: IPlaylist[] = []
    let total = 0

    switch (type) {
      case 'created':
        const createdResult = await Playlist.find({ creator: userId })
          .populate('musics')
          .sort({ updatedAt: -1 })
          .skip((page - 1) * limit)
          .limit(limit)
        
        playlists = createdResult
        total = await Playlist.countDocuments({ creator: userId })
        break

      case 'liked':
        const likedPlaylists = await PlaylistLike.find({ user: userId })
          .populate({
            path: 'playlist',
            populate: {
              path: 'musics'
            }
          })
          .sort({ createdAt: -1 })
          .skip((page - 1) * limit)
          .limit(limit)
        
        playlists = likedPlaylists.map(item => item.playlist as IPlaylist)
        total = await PlaylistLike.countDocuments({ user: userId })
        break

      case 'collaborative':
        const collabResult = await Playlist.find({
          $or: [
            { creator: userId },
            { collaborators: userId }
          ],
          isCollaborative: true
        })
          .populate('musics')
          .sort({ updatedAt: -1 })
          .skip((page - 1) * limit)
          .limit(limit)
        
        playlists = collabResult
        total = await Playlist.countDocuments({
          $or: [
            { creator: userId },
            { collaborators: userId }
          ],
          isCollaborative: true
        })
        break
    }

    const result = { playlists, total }
    
    // 缓存15分钟
    await this.cache.set(cacheKey, result, { 
      ttl: 900, 
      tags: [`user:${userId}:playlists`] 
    })

    return result
  }

  /**
   * 添加音乐到播放列表
   */
  async addMusicToPlaylist(
    playlistId: string, 
    musicId: string, 
    userId: string,
    position?: number
  ): Promise<IPlaylist> {
    const playlist = await Playlist.findById(playlistId)
    if (!playlist) {
      throw new Error('播放列表不存在')
    }

    // 检查权限
    if (!this.canEditPlaylist(playlist, userId)) {
      throw new Error('无权限编辑此播放列表')
    }

    // 检查音乐是否已存在
    if (playlist.musics.includes(musicId as any)) {
      throw new Error('音乐已在播放列表中')
    }

    // 添加音乐
    if (position !== undefined && position >= 0 && position <= playlist.musics.length) {
      playlist.musics.splice(position, 0, musicId as any)
    } else {
      playlist.musics.push(musicId as any)
    }

    await playlist.save()
    await this.populatePlaylist(playlist)

    // 清除相关缓存
    await this.cache.delete(`playlist:${playlistId}`)
    await this.cache.deleteByTag(`user:${userId}:playlists`)

    // 记录行为
    await this.recordPlaylistBehavior(userId, playlistId, 'add_music', {
      musicId,
      position
    })

    return playlist
  }

  /**
   * 从播放列表移除音乐
   */
  async removeMusicFromPlaylist(
    playlistId: string, 
    musicId: string, 
    userId: string
  ): Promise<IPlaylist> {
    const playlist = await Playlist.findById(playlistId)
    if (!playlist) {
      throw new Error('播放列表不存在')
    }

    // 检查权限
    if (!this.canEditPlaylist(playlist, userId)) {
      throw new Error('无权限编辑此播放列表')
    }

    // 移除音乐
    const index = playlist.musics.indexOf(musicId as any)
    if (index === -1) {
      throw new Error('音乐不在播放列表中')
    }

    playlist.musics.splice(index, 1)
    await playlist.save()
    await this.populatePlaylist(playlist)

    // 清除相关缓存
    await this.cache.delete(`playlist:${playlistId}`)
    await this.cache.deleteByTag(`user:${userId}:playlists`)

    // 记录行为
    await this.recordPlaylistBehavior(userId, playlistId, 'remove_music', {
      musicId
    })

    return playlist
  }

  /**
   * 调整播放列表音乐顺序
   */
  async reorderPlaylist(
    playlistId: string, 
    musicIds: string[], 
    userId: string
  ): Promise<IPlaylist> {
    const playlist = await Playlist.findById(playlistId)
    if (!playlist) {
      throw new Error('播放列表不存在')
    }

    // 检查权限
    if (!this.canEditPlaylist(playlist, userId)) {
      throw new Error('无权限编辑此播放列表')
    }

    // 验证音乐ID列表
    const currentMusicIds = playlist.musics.map(id => id.toString())
    const isValid = musicIds.every(id => currentMusicIds.includes(id)) &&
                   currentMusicIds.every(id => musicIds.includes(id))
    
    if (!isValid) {
      throw new Error('无效的音乐顺序')
    }

    // 更新顺序
    playlist.musics = musicIds as any[]
    await playlist.save()
    await this.populatePlaylist(playlist)

    // 清除相关缓存
    await this.cache.delete(`playlist:${playlistId}`)
    await this.cache.deleteByTag(`user:${userId}:playlists`)

    return playlist
  }

  /**
   * 复制播放列表
   */
  async duplicatePlaylist(
    playlistId: string, 
    userId: string, 
    name?: string
  ): Promise<IPlaylist> {
    const originalPlaylist = await this.getPlaylist(playlistId, userId)
    if (!originalPlaylist) {
      throw new Error('播放列表不存在或无权限访问')
    }

    const duplicateData = {
      name: name || `${originalPlaylist.name} - 副本`,
      description: originalPlaylist.description,
      cover: originalPlaylist.cover,
      musicIds: originalPlaylist.musics.map(music => music._id.toString()),
      tags: [...originalPlaylist.tags],
      isPublic: false
    }

    const duplicate = await this.createPlaylist(userId, duplicateData)

    // 记录行为
    await this.recordPlaylistBehavior(userId, playlistId, 'duplicate', {
      duplicateId: duplicate._id.toString()
    })

    return duplicate
  }

  /**
   * 删除播放列表
   */
  async deletePlaylist(playlistId: string, userId: string): Promise<void> {
    const playlist = await Playlist.findById(playlistId)
    if (!playlist) {
      throw new Error('播放列表不存在')
    }

    // 只有创建者可以删除
    if (playlist.creator.toString() !== userId) {
      throw new Error('只有创建者可以删除播放列表')
    }

    await Playlist.findByIdAndDelete(playlistId)

    // 清除相关缓存
    await this.cache.delete(`playlist:${playlistId}`)
    await this.cache.deleteByTag(`user:${userId}:playlists`)

    // 删除相关点赞记录
    await PlaylistLike.deleteMany({ playlist: playlistId })
  }

  /**
   * 搜索播放列表
   */
  async searchPlaylists(
    query: string,
    options: {
      page?: number
      limit?: number
      tags?: string[]
      userId?: string
    } = {}
  ): Promise<{ playlists: IPlaylist[], total: number }> {
    const { page = 1, limit = 20, tags, userId } = options

    const searchQuery: any = {
      isPublic: true,
      $text: { $search: query }
    }

    if (tags && tags.length > 0) {
      searchQuery.tags = { $in: tags }
    }

    if (userId) {
      searchQuery.creator = userId
    }

    const playlists = await Playlist.find(searchQuery)
      .populate('creator', 'username avatar')
      .populate('musics', 'title artist duration coverUrl')
      .sort({ score: { $meta: 'textScore' }, playCount: -1 })
      .skip((page - 1) * limit)
      .limit(limit)

    const total = await Playlist.countDocuments(searchQuery)

    return { playlists, total }
  }

  /**
   * 获取热门播放列表
   */
  async getTrendingPlaylists(limit: number = 20): Promise<IPlaylist[]> {
    const cacheKey = `trending:playlists:${limit}`
    
    let playlists = await this.cache.get<IPlaylist[]>(cacheKey)
    
    if (!playlists) {
      playlists = await Playlist.find({ isPublic: true })
        .populate('creator', 'username avatar')
        .populate('musics', 'title artist duration coverUrl')
        .sort({ playCount: -1, likeCount: -1, createdAt: -1 })
        .limit(limit)

      // 缓存30分钟
      await this.cache.set(cacheKey, playlists, { ttl: 1800 })
    }

    return playlists
  }

  /**
   * 更新播放列表信息
   */
  async updatePlaylist(
    playlistId: string, 
    updateData: UpdatePlaylistDto, 
    userId: string
  ): Promise<IPlaylist> {
    const playlist = await Playlist.findById(playlistId)
    if (!playlist) {
      throw new Error('播放列表不存在')
    }

    // 检查权限
    if (!this.canEditPlaylist(playlist, userId)) {
      throw new Error('无权限编辑此播放列表')
    }

    Object.assign(playlist, updateData)
    await playlist.save()
    await this.populatePlaylist(playlist)

    // 清除相关缓存
    await this.cache.delete(`playlist:${playlistId}`)
    await this.cache.deleteByTag(`user:${userId}:playlists`)

    return playlist
  }

  /**
   * 检查是否可以编辑播放列表
   */
  private canEditPlaylist(playlist: IPlaylist, userId: string): boolean {
    return playlist.creator.toString() === userId ||
           playlist.collaborators.some(collaborator => collaborator.toString() === userId)
  }

  /**
   * 填充播放列表数据
   */
  private async populatePlaylist(playlist: IPlaylist): Promise<void> {
    await playlist.populate([
      { path: 'creator', select: 'username avatar' },
      { path: 'musics', select: 'title artist duration coverUrl' },
      { path: 'collaborators', select: 'username avatar' }
    ])
  }

  /**
   * 记录播放列表行为
   */
  private async recordPlaylistBehavior(
    userId: string,
    playlistId: string,
    action: string,
    metadata?: any
  ): Promise<void> {
    // 这里可以集成行为分析系统
    console.log('Playlist behavior:', { userId, playlistId, action, metadata })
  }
}
```

```typescript
// src/controllers/playlistController.ts
import { Request, Response } from 'express'
import { PlaylistService } from '@/services/playlistService'

import { AuthenticatedRequest } from '@/types/request'

export class PlaylistController {
  private playlistService: PlaylistService

  constructor() {
    this.playlistService = new PlaylistService()
  }

  /**
   * 创建播放列表
   */
  createPlaylist = async (req: AuthenticatedRequest, res: Response): Promise<void> => {
    try {
      const userId = req.user!.id
      const { error, value } = validateCreatePlaylist(req.body)
      
      if (error) {
        res.status(400).json({
          success: false,
          message: '参数验证失败',
          errors: error.details
        })
        return
      }

      const playlist = await this.playlistService.createPlaylist(userId, value)

      res.status(201).json({
        success: true,
        message: '播放列表创建成功',
        data: playlist
      })
    } catch (error: any) {
      res.status(500).json({
        success: false,
        message: error.message || '创建播放列表失败'
      })
    }

  /**
   * 获取播放列表详情

        limit: parseInt(limit as string),

      })

      res.json({
        success: true,
        data: result
      })
    } catch (error: any) {
```

```typescript
// src/services/favoriteService.ts
import { Favorite, IFavorite, FavoriteFolder, IFavoriteFolder } from '@/models/Favorite'
import { Music } from '@/models/Music'
import { CacheManager } from '@/utils/cacheManager'

export class FavoriteService {
  private cache: CacheManager

  constructor() {
    this.cache = CacheManager.getInstance()
  }

  /**
   * 收藏音乐
   */
  async favoriteMusic(
    userId: string, 
    musicId: string, 
    options: {
      folderId?: string
      note?: string
      tags?: string[]
    } = {}
  ): Promise<IFavorite> {
    // 检查是否已收藏
    const existing = await Favorite.findOne({ user: userId, music: musicId })
    if (existing) {
      throw new Error('音乐已在收藏中')
    }

    const favorite = new Favorite({
      user: userId,
      music: musicId,
      folder: options.folderId,
      note: options.note,
      tags: options.tags || []
    })

    await favorite.save()
    await favorite.populate('music', 'title artist coverUrl duration')

    // 更新收藏夹计数
    if (options.folderId) {
      await this.incrementFolderCount(options.folderId)
    }

    // 清除用户收藏缓存
    await this.cache.deleteByTag(`user:${userId}:favorites`)

    return favorite
  }

  /**
   * 取消收藏
   */
  async unfavoriteMusic(userId: string, musicId: string): Promise<void> {
    const favorite = await Favorite.findOneAndDelete({ user: userId, music: musicId })
    
    if (!favorite) {
      throw new Error('音乐未在收藏中')
    }

    // 更新收藏夹计数
    if (favorite.folder) {
      await this.decrementFolderCount(favorite.folder.toString())
    }

    // 清除用户收藏缓存
    await this.cache.deleteByTag(`user:${userId}:favorites`)
  }

  /**
   * 检查是否已收藏
   */
  async isFavorited(userId: string, musicId: string): Promise<boolean> {
    const cacheKey = `favorite:${userId}:${musicId}`
    
    let isFavorited = await this.cache.get<boolean>(cacheKey)
    if (isFavorited !== null) {
      return isFavorited
    }

    const favorite = await Favorite.findOne({ user: userId, music: musicId })
    isFavorited = !!favorite

    // 缓存10分钟
    await this.cache.set(cacheKey, isFavorited, { ttl: 600 })

    return isFavorited
  }

  /**
   * 获取用户收藏列表
   */
  async getUserFavorites(
    userId: string, 
    options: {
      page?: number
      limit?: number
      folderId?: string
      tags?: string[]
      sortBy?: 'createdAt' | 'title' | 'artist'
      sortOrder?: 'asc' | 'desc'
    } = {}
  ): Promise<{ favorites: IFavorite[], total: number }> {
    const { 
      page = 1, 
      limit = 20, 
      folderId, 
      tags, 
      sortBy = 'createdAt', 
      sortOrder = 'desc' 
    } = options

    const cacheKey = `user:${userId}:favorites:${page}:${limit}:${folderId || 'all'}:${tags?.join(',') || ''}:${sortBy}:${sortOrder}`
    
    const cached = await this.cache.get(cacheKey)
    if (cached) return cached

    const query: any = { user: userId }
    
    if (folderId) {
      query.folder = folderId
    }
    
    if (tags && tags.length > 0) {
      query.tags = { $in: tags }
    }

    const sort: any = {}
    sort[sortBy] = sortOrder === 'asc' ? 1 : -1

    const favorites = await Favorite.find(query)
      .populate('music', 'title artist coverUrl duration genre')
      .populate('folder', 'name')
      .sort(sort)
      .skip((page - 1) * limit)
      .limit(limit)

    const total = await Favorite.countDocuments(query)

    const result = { favorites, total }
    
    // 缓存5分钟
    await this.cache.set(cacheKey, result, { 
      ttl: 300, 
      tags: [`user:${userId}:favorites`] 
    })

    return result
  }

  /**
   * 创建收藏夹
   */
  async createFolder(
    userId: string, 
    folderData: {
      name: string
      description?: string
      cover?: string
      isPublic?: boolean
    }
  ): Promise<IFavoriteFolder> {
    const folder = new FavoriteFolder({
      ...folderData,
      user: userId
    })

    await folder.save()

    // 清除用户收藏夹缓存
    await this.cache.deleteByTag(`user:${userId}:folders`)

    return folder
  }

  /**
   * 获取用户收藏夹列表
   */
  async getUserFolders(userId: string): Promise<IFavoriteFolder[]> {
    const cacheKey = `user:${userId}:folders`
    
    let folders = await this.cache.get<IFavoriteFolder[]>(cacheKey)
    
    if (!folders) {
      folders = await FavoriteFolder.find({ user: userId })
        .sort({ isDefault: -1, createdAt: -1 })

      // 缓存15分钟
      await this.cache.set(cacheKey, folders, { 
        ttl: 900, 
        tags: [`user:${userId}:folders`] 
      })
    }

    return folders
  }

  /**
   * 更新收藏夹
   */
  async updateFolder(
    folderId: string, 
    userId: string, 
    updateData: Partial<IFavoriteFolder>
  ): Promise<IFavoriteFolder> {
    const folder = await FavoriteFolder.findOne({ _id: folderId, user: userId })
    if (!folder) {
      throw new Error('收藏夹不存在')
    }

    Object.assign(folder, updateData)
    await folder.save()

    // 清除缓存
    await this.cache.deleteByTag(`user:${userId}:folders`)

    return folder
  }

  /**
   * 删除收藏夹
   */
  async deleteFolder(folderId: string, userId: string): Promise<void> {
    const folder = await FavoriteFolder.findOne({ _id: folderId, user: userId })
    if (!folder) {
      throw new Error('收藏夹不存在')
    }

    if (folder.isDefault) {
      throw new Error('默认收藏夹不能删除')
    }

    // 将该收藏夹中的收藏移到默认收藏夹
    const defaultFolder = await FavoriteFolder.findOne({ 
      user: userId, 
      isDefault: true 
    })

    if (defaultFolder) {
      await Favorite.updateMany(
        { user: userId, folder: folderId },
        { folder: defaultFolder._id }
      )
    } else {
      await Favorite.deleteMany({ user: userId, folder: folderId })
    }

    await FavoriteFolder.findByIdAndDelete(folderId)

    // 清除缓存
    await this.cache.deleteByTag(`user:${userId}:folders`)
    await this.cache.deleteByTag(`user:${userId}:favorites`)
  }

  /**
   * 移动收藏到其他收藏夹
   */
  async moveFavorite(
    userId: string, 
    musicId: string, 
    targetFolderId: string
  ): Promise<void> {
    const favorite = await Favorite.findOne({ user: userId, music: musicId })
    if (!favorite) {
      throw new Error('收藏不存在')
    }

    const oldFolderId = favorite.folder?.toString()
    
    // 更新收藏夹
    favorite.folder = targetFolderId
    await favorite.save()

    // 更新计数
    if (oldFolderId) {
      await this.decrementFolderCount(oldFolderId)
    }
    await this.incrementFolderCount(targetFolderId)

    // 清除缓存
    await this.cache.deleteByTag(`user:${userId}:favorites`)
  }

  /**
   * 更新收藏信息
   */
  async updateFavorite(
    userId: string, 
    musicId: string, 
    updateData: {
      note?: string
      tags?: string[]
      folderId?: string
    }
  ): Promise<IFavorite> {
    const favorite = await Favorite.findOne({ user: userId, music: musicId })
    if (!favorite) {
      throw new Error('收藏不存在')
    }

    const oldFolderId = favorite.folder?.toString()

    Object.assign(favorite, updateData)
    await favorite.save()
    await favorite.populate('music', 'title artist coverUrl duration')

    // 更新收藏夹计数
    if (oldFolderId && oldFolderId !== updateData.folderId) {
      await this.decrementFolderCount(oldFolderId)
    }
    if (updateData.folderId && updateData.folderId !== oldFolderId) {
      await this.incrementFolderCount(updateData.folderId)
    }

    // 清除缓存
    await this.cache.deleteByTag(`user:${userId}:favorites`)

    return favorite
  }

  /**
   * 获取收藏统计
   */
  async getFavoriteStats(userId: string): Promise<{
    totalFavorites: number
    totalFolders: number
    recentFavorites: IFavorite[]
    favoriteGenres: any[]
  }> {
    const cacheKey = `user:${userId}:favorite-stats`
    
    let stats = await this.cache.get(cacheKey)
    
    if (!stats) {
      const totalFavorites = await Favorite.countDocuments({ user: userId })
      const totalFolders = await FavoriteFolder.countDocuments({ user: userId })
      
      const recentFavorites = await Favorite.find({ user: userId })
        .populate('music', 'title artist coverUrl genre')
        .sort({ createdAt: -1 })
        .limit(5)

      // 统计收藏的音乐类型
      const genreStats = await Favorite.aggregate([
        { $match: { user: new mongoose.Types.ObjectId(userId) } },
        { $lookup: { from: 'musics', localField: 'music', foreignField: '_id', as: 'music' } },
        { $unwind: '$music' },
        { $group: { _id: '$music.genre', count: { $sum: 1 } } },
        { $sort: { count: -1 } },
        { $limit: 10 }
      ])

      stats = {
        totalFavorites,
        totalFolders,
        recentFavorites,
        favoriteGenres: genreStats
      }

      // 缓存10分钟
      await this.cache.set(cacheKey, stats, { ttl: 600 })
    }

    return stats
  }

  /**
   * 增加收藏夹计数
   */
  private async incrementFolderCount(folderId: string): Promise<void> {
    await FavoriteFolder.findByIdAndUpdate(folderId, {
      $inc: { favoritesCount: 1 }
    })
  }

  /**
   * 减少收藏夹计数
   */
  private async decrementFolderCount(folderId: string): Promise<void> {
    await FavoriteFolder.findByIdAndUpdate(folderId, {
      $inc: { favoritesCount: -1 }
    })
  }
}
```

```typescript
  async createComment(
    userId: string, 
    commentData: {
      content: string
      targetType: string
      targetId: string
      parentId?: string
    }
  ): Promise<IComment> {
    const comment = new Comment({
      ...commentData,
      author: userId
    })

    await comment.save()
    await comment.populate([
      { path: 'author', select: 'username avatar level' },
      { path: 'parentId', populate: { path: 'author', select: 'username' } }
    ])

    // 如果是回复，添加到父评论的回复列表
    if (commentData.parentId) {
      await Comment.findByIdAndUpdate(
        commentData.parentId,
        { $push: { replies: comment._id } }
      )
    }

    // 清除相关缓存
    await this.clearCommentCache(commentData.targetType, commentData.targetId)

    // 发送通知
    await this.sendCommentNotification(comment)

    return comment
  }

  /**
   * 获取评论列表
   */
  async getComments(
    targetType: string,
    targetId: string,
    options: {
      page?: number
      limit?: number
      sortBy?: 'createdAt' | 'likes' | 'replies'
      sortOrder?: 'asc' | 'desc'
      parentId?: string
    } = {}
  ): Promise<{ comments: IComment[], total: number }> {
    const { 
      page = 1, 
      limit = 20, 
      sortBy = 'createdAt', 
      sortOrder = 'desc',
      parentId
    } = options

    const cacheKey = `comments:${targetType}:${targetId}:${page}:${limit}:${sortBy}:${sortOrder}:${parentId || 'root'}`
    
    const cached = await this.cache.get(cacheKey)
    if (cached) return cached

    const query: any = {
      targetType,
      targetId,
      isDeleted: false,
      parentId: parentId || { $exists: false }
    }

    const sort: any = {}
    if (sortBy === 'likes') {
      sort.likes = sortOrder === 'asc' ? 1 : -1
      sort.createdAt = -1 // 二级排序
    } else {
      sort[sortBy] = sortOrder === 'asc' ? 1 : -1
    }

    // 置顶评论优先
    sort.isPinned = -1

    const comments = await Comment.find(query)
      .populate('author', 'username avatar level badge')
      .populate({
        path: 'replies',
        populate: {
          path: 'author',
          select: 'username avatar level'
        },
        options: { sort: { createdAt: 1 } }
      })
      .sort(sort)
      .skip((page - 1) * limit)
      .limit(limit)

    const total = await Comment.countDocuments(query)

    const result = { comments, total }
    
    // 缓存2分钟
    await this.cache.set(cacheKey, result, { ttl: 120 })

    return result
  }

    comment.updatedAt = new Date()
    await comment.save()

    await comment.populate('author', 'username avatar level')

    return comment
  }

  async unlikeComment(commentId: string, userId: string): Promise<void> {
```

```typescript
      }

      // 根据时间范围设置缓存时间
      const ttl = this.getCacheTTL(timeRange)
      await this.cache.set(cacheKey, analytics, { ttl })
    }

    return analytics
  }

  /**
   * 获取用户分析数据
   */
  private async getUserAnalytics(dateRange: { start: Date, end: Date }): Promise<UserAnalytics> {
    const totalUsers = await User.countDocuments()
    
    const activeUsers = await User.countDocuments({
      lastActiveAt: { $gte: dateRange.start }
    })

    const newUsers = await User.countDocuments({
      createdAt: { $gte: dateRange.start, $lte: dateRange.end }
    })

    const previousNewUsers = await User.countDocuments({
      createdAt: { 
        $gte: new Date(dateRange.start.getTime() - (dateRange.end.getTime() - dateRange.start.getTime())),
        $lt: dateRange.start
      }
    })

    const userGrowthRate = previousNewUsers > 0 
      ? ((newUsers - previousNewUsers) / previousNewUsers) * 100 
      : 0

    // 平均会话时长（这里需要从会话日志中获取）
    const avgSessionDuration = await this.getAvgSessionDuration(dateRange)

    // 热门国家
    const topCountries = await this.getTopCountries(dateRange)

    // 用户留存率
    const userRetention = await this.getUserRetention(dateRange)

    return {
      totalUsers,
      activeUsers,
      newUsers,
      userGrowthRate,
      avgSessionDuration,
      topCountries,
      userRetention
    }
  }

   * 获取音乐分析数据
   */
  private async getMusicAnalytics(dateRange: { start: Date, end: Date }): Promise<MusicAnalytics> {

    const totalPlays = await PlayRecord.countDocuments({
      timestamp: { $gte: dateRange.start, $lte: dateRange.end }

          _id: null,
          avgDuration: { $avg: '$duration' }

    // 趋势音乐
    const trendingMusic = await this.getTrendingMusic(dateRange)

      trendingMusic,

  private getDateRange(
    timeRange: string,

    endDate?: Date

    const end = endDate || new Date()
    let start: Date

    switch (timeRange) {
      case 'day':
```

```typescript
  private setupMiddleware(): void {
    // 身份验证中间件
    this.io.use(async (socket: AuthenticatedSocket, next) => {
      try {
        const token = socket.handshake.auth.token
        if (!token) {
          return next(new Error('Authentication token required'))
        }

        const decoded = jwt.verify(token, process.env.JWT_SECRET!) as any
        const user = await User.findById(decoded.userId)

        socket.userId = user._id.toString()
        socket.user = user
        
        next()
      } catch (error) {

      }
    })

    })
```

```typescript
    const notifications = recipients.map(recipientId => ({
      ...notificationData,
      recipient: recipientId
    }))

    await Notification.insertMany(notifications)

    // 批量发送
    for (const recipientId of recipients) {
      const recipient = await User.findById(recipientId)
      if (recipient) {
        const channels = await this.getUserNotificationChannels(recipientId, notificationData.type)
        
        for (const channel of channels) {

          await this.sendNotificationChannel(notification, recipient, channel)
        }
      }
    }

      type?: string
      isRead?: boolean
      sortBy?: 'createdAt' | 'priority'
      sortOrder?: 'asc' | 'desc'
    } = {}
  ): Promise<{ notifications: any[], total: number, unreadCount: number }> {
    const { 
      page = 1, 
      limit = 20, 
      type, 
      isRead, 
      sortBy = 'createdAt', 
      sortOrder = 'desc' 
    } = options

    const cacheKey = `notifications:${userId}:${page}:${limit}:${type || 'all'}:${isRead !== undefined ? isRead : 'all'}:${sortBy}:${sortOrder}`
    
    const cached = await this.cache.get(cacheKey)
    if (cached) return cached

    const query: any = { recipient: userId }
    
    if (type) {
      query.type = type
    }
    
    if (isRead !== undefined) {
      query.isRead = isRead
    }

    const sort: any = {}
    sort[sortBy] = sortOrder === 'asc' ? 1 : -1
    sort.priority = -1 // 优先级排序

    const notifications = await Notification.find(query)
      .sort(sort)
      .skip((page - 1) * limit)
      .limit(limit)

    const unreadCount = await Notification.countDocuments({ 
      recipient: userId, 
      isRead: false 
    })

    await User.findByIdAndUpdate(

      { 
        $set: { 
          'notificationSettings': settings,
          updatedAt: new Date()
        } 
      }
    )

    // 清除相关缓存
    await this.cache.delete(`notification-settings:${userId}`)
  }

  /**
   * 获取未读通知数量
   */
  async getUnreadCount(userId: string, type?: string): Promise<number> {
    const cacheKey = `unread-count:${userId}:${type || 'all'}`
    
    let count = await this.cache.get<number>(cacheKey)
    
    if (count === null) {
      const query: any = { recipient: userId, isRead: false }
      if (type) {
        query.type = type
      }

      count = await Notification.countDocuments(query)
      
      // 缓存30秒
      await this.cache.set(cacheKey, count, { ttl: 30 })
    }

    return count
  }

  // 私有方法
  private async sendNotificationChannel(
    notification: any, 
    recipient: any, 
    channel: string
  ): Promise<void> {
    switch (channel) {
      case 'web':
        await this.sendWebNotification(notification, recipient)
        break
      case 'email':
        await this.sendEmailNotification(notification, recipient)
        break
      case 'push':
        await this.sendPushNotification(notification, recipient)
        break
    }
  }

  private async sendWebNotification(notification: any, recipient: any): Promise<void> {
    // 通过WebSocket发送实时通知
    await this.wsService.sendNotificationToUser(recipient._id.toString(), {
      id: notification._id,
      type: notification.type,
      title: notification.title,
      content: notification.content,
      relatedId: notification.relatedId,
      relatedType: notification.relatedType,
      metadata: notification.metadata,
      priority: notification.priority || 'normal'
    })
  }

  private async sendEmailNotification(notification: any, recipient: any): Promise<void> {
    if (!recipient.email) return

    try {
      await this.emailService.sendEmail({
        to: recipient.email,
        subject: notification.title,
        template: notification.type,
        data: {
          username: recipient.username,
          content: notification.content,
          metadata: notification.metadata
        }
      })
    } catch (error) {
      console.error('Send email notification failed:', error)
    }
  }

  private async sendPushNotification(notification: any, recipient: any): Promise<void> {
    // 实现推送通知（需要集成推送服务）
    console.log('Send push notification:', notification, recipient)
  }

  private async getUserNotificationChannels(
    userId: string, 
    notificationType: string
  ): Promise<string[]> {
    const settings = await this.getNotificationSettings(userId)
    const typeSettings = settings[notificationType] || {}
    
    // 默认渠道
    const defaultChannels = ['web']
    
    // 根据用户设置和通知类型确定渠道
    const channels = typeSettings.channels || defaultChannels
    
    // 检查用户是否启用了相应渠道
    return channels.filter(channel => {
      switch (channel) {
        case 'email':
          return settings.enableEmail !== false && typeSettings.enableEmail !== false
        case 'push':
          return settings.enablePush !== false && typeSettings.enablePush !== false
        case 'web':
        default:
          return true // Web通知默认启用
      }
    })
```

```typescript
import express from 'express'

    this.initializeRoutes()
    this.initializeErrorHandling()

      next()
    })
  }

        success: false,
        message: 'API endpoint not found'
      })

    this.app.use(errorHandler)
```
