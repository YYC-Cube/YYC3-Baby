### 2. 威尔逊区间算法实现 (backend/services/wilsonScore.js)
```javascript
/**
 * 威尔逊区间得分计算服务
 * 基于Wilson Score Interval算法，解决小样本排名不公平问题
 */
class WilsonScoreService {
    /**
     * 计算威尔逊区间下限
     * @param {number} positive - 正面评价数
     * @param {number} total - 总评价数
     * @param {number} confidence - 置信水平 (默认0.95)
     * @returns {number} 威尔逊得分
     */
    static calculateWilsonScore(positive, total, confidence = 0.95) {
        if (total === 0) return 0;
        
        // 计算好评率
        const p = positive / total;
        
        // 获取z值 (95%置信水平对应1.96)
        const z = this.getZValue(confidence);
        
        // 威尔逊区间公式
        const denominator = 1 + (z * z) / total;
        const numerator = p + (z * z) / (2 * total);
        const adjustment = z * Math.sqrt(
            (p * (1 - p)) / total + (z * z) / (4 * total * total)
        );
        
        const lowerBound = (numerator - adjustment) / denominator;
        
        // 确保结果在0-1之间
        return Math.max(0, Math.min(1, lowerBound));
    }

    /**
     * 获取标准正态分布的分位数
     * @param {number} confidence - 置信水平
     * @returns {number} z值
     */
    static getZValue(confidence) {
        const zValues = {
            0.90: 1.645,
            0.95: 1.960,
            0.99: 2.576
        };
        return zValues[confidence] || 1.960;
    }

    /**
     * 计算作品综合得分（结合威尔逊得分和星力助推）
     * @param {Object} contentData - 作品数据
     * @param {string} rankingType - 排行榜类型
     * @returns {number} 综合得分
     */
    static calculateContentScore(contentData, rankingType) {
        const {
            positive_votes = 0,
            total_votes = 0,
            play_count = 0,
            share_count = 0,
            star_power_boost = 0
        } = contentData;

        // 计算威尔逊基础得分
        const wilsonScore = this.calculateWilsonScore(positive_votes, total_votes);
        
        // 根据排行榜类型使用不同的权重
        const weights = this.getRankingWeights(rankingType);
        
        // 计算综合得分
        let finalScore = 0;
        
        // 威尔逊得分权重
        finalScore += wilsonScore * weights.wilson;
        
        // 播放量权重（归一化处理）
        const normalizedPlays = this.normalizeValue(play_count, 10000); // 假设最大1万播放
        finalScore += normalizedPlays * weights.plays;
        
        // 分享量权重
        const normalizedShares = this.normalizeValue(share_count, 100); // 假设最大100分享
        finalScore += normalizedShares * weights.shares;
        
        // 星力助推权重
        const normalizedBoost = this.normalizeValue(star_power_boost, 5000); // 假设最大5000星力助推
        finalScore += normalizedBoost * weights.boost;
        
        return finalScore;
    }

    /**
     * 获取不同排行榜的权重配置
     */
    static getRankingWeights(rankingType) {
        const weightConfigs = {
            'daily': { wilson: 0.3, plays: 0.3, shares: 0.1, boost: 0.3 },     // 日榜：星力助推权重高
            'weekly': { wilson: 0.4, plays: 0.4, shares: 0.2, boost: 0.0 },    // 周榜：更注重作品质量
            'monthly': { wilson: 0.5, plays: 0.3, shares: 0.2, boost: 0.0 },   // 月榜：最注重用户评价
            'potential': { wilson: 0.6, plays: 0.2, shares: 0.2, boost: 0.0 }  // 潜力榜：主要看威尔逊得分
        };
        
        return weightConfigs[rankingType] || weightConfigs.daily;
    }

    /**
     * 数值归一化 (0-1)
     */
    static normalizeValue(value, maxValue) {
        return Math.min(value / maxValue, 1);
    }

    /**
     * 批量计算作品排名
     */
    static async calculateRankings(contents, rankingType) {
        return contents.map(content => {
            const score = this.calculateContentScore(content, rankingType);
            return {
                ...content,
                wilson_score: this.calculateWilsonScore(content.positive_votes, content.total_votes),
                final_score: score
            };
        }).sort((a, b) => b.final_score - a.final_score);
    }
}

module.exports = WilsonScoreService;
```

### 3. 星力核心服务 (backend/services/starPowerService.js)
```javascript
const db = require('../config/database');
const Redis = require('ioredis');
const redis = new Redis(process.env.REDIS_URL);

class StarPowerService {
    /**
     * 获取用户星力余额
     */
    async getUserStarPower(userId) {
        const cacheKey = `star_power:${userId}`;
        
        // 尝试从缓存获取
        const cached = await redis.get(cacheKey);
        if (cached) {
            return JSON.parse(cached);
        }
        
        // 从数据库查询
        const [rows] = await db.execute(
            'SELECT * FROM user_star_power WHERE user_id = ?',
            [userId]
        );
        
        let userStarPower = rows[0];
        
        // 如果用户不存在，初始化
        if (!userStarPower) {
            userStarPower = await this.initializeUserStarPower(userId);
        }
        
        // 写入缓存，过期时间5分钟
        await redis.setex(cacheKey, 300, JSON.stringify(userStarPower));
        
        return userStarPower;
    }

    /**
     * 初始化用户星力账户
     */
    async initializeUserStarPower(userId) {
        const [result] = await db.execute(
            `INSERT INTO user_star_power 
             (user_id, total_star_power, available_star_power, frozen_star_power, vip_level, vip_exp) 
             VALUES (?, 100, 100, 0, 1, 0)`,
            [userId]
        );
        
        return {
            id: result.insertId,
            user_id: userId,
            total_star_power: 100,
            available_star_power: 100,
            frozen_star_power: 0,
            vip_level: 1,
            vip_exp: 0
        };
    }

    /**
     * 增加星力
     */
    async earnStarPower(userId, amount, ruleType, description = '', relatedId = null) {
        // 检查获取限制
        const canEarn = await this.checkEarnLimit(userId, ruleType, amount);
        if (!canEarn) {
            throw new Error(`已达到${ruleType}的获取上限`);
        }

        const connection = await db.getConnection();
        
        try {
            await connection.beginTransaction();

            // 更新用户星力余额
            const [updateResult] = await connection.execute(
                `UPDATE user_star_power 
                 SET total_star_power = total_star_power + ?, 
                     available_star_power = available_star_power + ?,
                     updated_at = CURRENT_TIMESTAMP
                 WHERE user_id = ?`,
                [amount, amount, userId]
            );

            if (updateResult.affectedRows === 0) {
                throw new Error('用户星力账户不存在');
            }

            // 记录交易
            const [userData] = await connection.execute(
                'SELECT available_star_power FROM user_star_power WHERE user_id = ?',
                [userId]
            );

            await connection.execute(
                `INSERT INTO star_power_transactions 
                 (user_id, transaction_type, amount, balance_after, description, related_type, related_id) 
                 VALUES (?, 'earn', ?, ?, ?, ?, ?)`,
                [
                    userId, 
                    amount, 
                    userData[0].available_star_power,
                    description,
                    ruleType,
                    relatedId
                ]
            );

            await connection.commit();

            // 清除缓存
            await redis.del(`star_power:${userId}`);

            return {
                success: true,
                newBalance: userData[0].available_star_power,
                earnedAmount: amount
            };

        } catch (error) {
            await connection.rollback();
            throw error;
        } finally {
            connection.release();
        }
    }

    /**
     * 消费星力
     */
    async spendStarPower(userId, amount, spendType, description = '', relatedId = null) {
        const connection = await db.getConnection();
        
        try {
            await connection.beginTransaction();

            // 检查余额
            const [userData] = await connection.execute(
                'SELECT available_star_power FROM user_star_power WHERE user_id = ? FOR UPDATE',
                [userId]
            );

            if (!userData[0] || userData[0].available_star_power < amount) {
                throw new Error('星力余额不足');
            }

            // 扣除星力
            const [updateResult] = await connection.execute(
                `UPDATE user_star_power 
                 SET available_star_power = available_star_power - ?,
                     updated_at = CURRENT_TIMESTAMP
                 WHERE user_id = ?`,
                [amount, userId]
            );

            // 记录交易
            const newBalance = userData[0].available_star_power - amount;
            await connection.execute(
                `INSERT INTO star_power_transactions 
                 (user_id, transaction_type, amount, balance_after, description, related_type, related_id) 
                 VALUES (?, 'spend', ?, ?, ?, ?, ?)`,
                [
                    userId, 
                    amount, 
                    newBalance,
                    description,
                    spendType,
                    relatedId
                ]
            );

            await connection.commit();

            // 清除缓存
            await redis.del(`star_power:${userId}`);

            return {
                success: true,
                newBalance: newBalance,
                spentAmount: amount
            };

        } catch (error) {
            await connection.rollback();
            throw error;
        } finally {
            connection.release();
        }
    }

    /**
     * 检查星力获取限制
     */
    async checkEarnLimit(userId, ruleType, amount) {
        const today = new Date().toISOString().split('T')[0];
        
        // 获取规则配置
        const [rules] = await db.execute(
            'SELECT * FROM star_power_rules WHERE rule_type = ? AND is_active = TRUE',
            [ruleType]
        );
        
        if (rules.length === 0) return true;
        
        const rule = rules[0];
        
        // 检查日限制
        if (rule.daily_limit) {
            const [todayEarnings] = await db.execute(
                `SELECT SUM(amount) as total 
                 FROM star_power_transactions 
                 WHERE user_id = ? AND related_type = ? AND DATE(created_at) = ? 
                 AND transaction_type = 'earn'`,
                [userId, ruleType, today]
            );
            
            const todayTotal = todayEarnings[0].total || 0;
            if (todayTotal + amount > rule.daily_limit) {
                return false;
            }
        }
        
        // 检查总限制
        if (rule.total_limit) {
            const [totalEarnings] = await db.execute(
                `SELECT SUM(amount) as total 
                 FROM star_power_transactions 
                 WHERE user_id = ? AND related_type = ? AND transaction_type = 'earn'`,
                [userId, ruleType]
            );
            
            const total = totalEarnings[0].total || 0;
            if (total + amount > rule.total_limit) {
                return false;
            }
        }
        
        return true;
    }

    /**
     * 每日签到
     */
    async dailyCheckin(userId) {
        const today = new Date().toISOString().split('T')[0];
        
        // 检查今天是否已经签到
        const [todayCheckins] = await db.execute(
            `SELECT id FROM star_power_transactions 
             WHERE user_id = ? AND related_type = 'daily_checkin' 
             AND DATE(created_at) = ?`,
            [userId, today]
        );
        
        if (todayCheckins.length > 0) {
            throw new Error('今天已经签到过了');
        }
        
        // 计算连续签到天数
        const consecutiveDays = await this.getConsecutiveCheckinDays(userId);
        
        // 基础奖励 + 连续签到额外奖励
        let baseAmount = 10;
        let extraAmount = 0;
        
        if (consecutiveDays >= 7) {
            extraAmount = 50; // 连续7天额外奖励
        }
        
        const totalAmount = baseAmount + extraAmount;
        const description = `每日签到${extraAmount > 0 ? `，连续${consecutiveDays + 1}天奖励` : ''}`;
        
        return await this.earnStarPower(
            userId, 
            totalAmount, 
            'daily_checkin', 
            description
        );
    }

    /**
     * 获取连续签到天数
     */
    async getConsecutiveCheckinDays(userId) {
        const [checkins] = await db.execute(
            `SELECT DATE(created_at) as checkin_date 
             FROM star_power_transactions 
             WHERE user_id = ? AND related_type = 'daily_checkin' 
             ORDER BY created_at DESC 
             LIMIT 7`,
            [userId]
        );
        
        let consecutive = 0;
        let currentDate = new Date();
        
        for (const checkin of checkins) {
            const checkinDate = new Date(checkin.checkin_date);
            const diffTime = currentDate - checkinDate;
            const diffDays = Math.floor(diffTime / (1000 * 60 * 60 * 24));
            
            if (diffDays === consecutive) {
                consecutive++;
            } else {
                break;
            }
            
            currentDate = checkinDate;
        }
        
        return consecutive;
    }

    /**
     * 邀请好友奖励
     */
    async inviteFriendReward(inviterId, inviteeId, inviteeCompletedCreation = false) {
        // 被邀请人注册奖励
        const registerReward = await this.earnStarPower(
            inviterId,
            50,
            'invite_register',
            '邀请好友注册奖励',
            inviteeId
        );
        
        let creationReward = null;
        
        // 被邀请人完成首次创作奖励
        if (inviteeCompletedCreation) {
            creationReward = await this.earnStarPower(
                inviterId,
                100,
                'invite_creation',
                '好友完成首次创作奖励',
                inviteeId
            );
        }
        
        return {
            registerReward,
            creationReward
        };
    }

    /**
     * 作品互动奖励
     */
    async contentInteractionReward(userId, contentType, interactionType, contentId) {
        const rewardRules = {
            'play': { base: 0.01, description: '播放奖励' }, // 每100播放+1星力
            'like': { base: 2, description: '点赞奖励' },
            'collect': { base: 5, description: '收藏奖励' },
            'share': { base: 3, description: '分享奖励' }
        };
        
        const rule = rewardRules[interactionType];
        if (!rule) {
            throw new Error('不支持的互动类型');
        }
        
        // 检查日上限
        const today = new Date().toISOString().split('T')[0];
        const [todayEarnings] = await db.execute(
            `SELECT SUM(amount) as total 
             FROM star_power_transactions 
             WHERE user_id = ? AND related_type = 'content_interaction' 
             AND DATE(created_at) = ?`,
            [userId, today]
        );
        
        const todayTotal = todayEarnings[0].total || 0;
        if (todayTotal + rule.base > 500) { // 日上限500
            throw new Error('今日互动奖励已达上限');
        }
        
        return await this.earnStarPower(
            userId,
            rule.base,
            'content_interaction',
            rule.description,
            contentId
        );
    }
}

module.exports = new StarPowerService();
```

```javascript
import { ref, computed } from 'vue'
import { useUserStore } from '../stores/user'

// 星力系统组合式函数
export function useStarPower() {
  const userStore = useUserStore()
  
  // 响应式数据
  const userStarPower = ref({
    available_star_power: 0,
    total_star_power: 0,
    frozen_star_power: 0,
    vip_level: 1,
    vip_exp: 0
  })
  
  const recentTransactions = ref([])
  const dailyTasks = ref([])
  const loading = ref(false)

  // 计算属性
  const canCheckIn = computed(() => {
    const todayCheckin = recentTransactions.value.find(
      t => t.related_type === 'daily_checkin' && 
      new Date(t.created_at).toDateString() === new Date().toDateString()
    )
    return !todayCheckin
  })

  // 加载星力数据
  const loadStarPowerData = async () => {
    loading.value = true
    try {
      const [starPowerRes, transactionsRes, tasksRes] = await Promise.all([
        fetch('/api/star-power/user').then(r => r.json()),
        fetch('/api/star-power/transactions/recent').then(r => r.json()),
        fetch('/api/star-power/tasks/daily').then(r => r.json())
      ])
      
      userStarPower.value = starPowerRes.data
      recentTransactions.value = transactionsRes.data
      dailyTasks.value = tasksRes.data
    } catch (error) {
      console.error('加载星力数据失败:', error)
    } finally {
      loading.value = false
    }
  }

  // 每日签到
  const checkInDaily = async () => {
    try {
      const response = await fetch('/api/star-power/checkin', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Authorization': `Bearer ${userStore.token}`
        }
      })
      
      const result = await response.json()
      
      if (result.success) {
        // 更新本地数据
        await loadStarPowerData()
        return result
      } else {
        throw new Error(result.message)
      }
    } catch (error) {
      console.error('签到失败:', error)
      throw error
    }
  }

  // 完成任务
  const completeDailyTask = async (taskId) => {
    try {
      const response = await fetch('/api/star-power/complete-task', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Authorization': `Bearer ${userStore.token}`
        },
        body: JSON.stringify({ taskId })
      })
      
      const result = await response.json()
      
      if (result.success) {
        await loadStarPowerData()
        return result
      } else {
        throw new Error(result.message)
      }
    } catch (error) {
      console.error('完成任务失败:', error)
      throw error
    }
  }

  // 消费星力
  const spendStarPower = async (amount, spendType, description = '', relatedId = null) => {
    try {
      const response = await fetch('/api/star-power/spend', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Authorization': `Bearer ${userStore.token}`
        },
        body: JSON.stringify({
          amount,
          spendType,
          description,
          relatedId
        })
      })
      
      const result = await response.json()
      
      if (result.success) {
        // 更新本地余额
        userStarPower.value.available_star_power = result.data.newBalance
        return result
      } else {
        throw new Error(result.message)
      }
    } catch (error) {
      console.error('消费星力失败:', error)
      throw error
    }
  }

  // 充值星力
  const rechargeStarPower = async (packageId, paymentMethod) => {
    try {
      const response = await fetch('/api/star-power/recharge', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Authorization': `Bearer ${userStore.token}`
        },
        body: JSON.stringify({
          packageId,
          paymentMethod
        })
      })
      
      const result = await response.json()
      
      if (result.success) {
        await loadStarPowerData()
        return result
      } else {
        throw new Error(result.message)
      }
    } catch (error) {
      console.error('充值失败:', error)
      throw error
    }
  }

  return {
    // 数据
    userStarPower,
    recentTransactions,
    dailyTasks,
    loading,
    
    // 计算属性
    canCheckIn,
    
    // 方法
    loadStarPowerData,
    checkInDaily,
    completeDailyTask,
    spendStarPower,
    rechargeStarPower
  }
}
```

```javascript
                    [userId, task.rule_type, today]
                );
                
                const completed = completions[0].count > 0;
                
                return {
                    id: task.rule_type,
                    name: task.rule_name,
                    reward: task.base_amount,
                    completed: completed,
                    description: this.getTaskDescription(task.rule_type)
```

## 📊 排行榜服务 (backend/services/rankingService.js)
```javascript
const WilsonScoreService = require('./wilsonScore');
const db = require('../config/database');
const Redis = require('ioredis');
const redis = new Redis(process.env.REDIS_URL);

class RankingService {
    /**
     * 获取排行榜
     */
    async getRankings(rankingType, page = 1, limit = 50) {
        const cacheKey = `rankings:${rankingType}:${page}:${limit}`;
        
        // 尝试从缓存获取
        const cached = await redis.get(cacheKey);
        if (cached) {
            return JSON.parse(cached);
        }
        
        const offset = (page - 1) * limit;
        
        // 获取排行榜数据
        const [rankings] = await db.execute(
            `SELECT rd.*, 
                    c.title as content_title,
                    c.cover_url as content_cover,
                    u.username as creator_name,
                    u.avatar as creator_avatar
             FROM ranking_data rd
             LEFT JOIN contents c ON rd.content_id = c.id
             LEFT JOIN users u ON c.creator_id = u.id
             WHERE rd.ranking_type = ? AND rd.ranking_date = ?
             ORDER BY rd.final_score DESC
             LIMIT ? OFFSET ?`,
            [rankingType, this.getRankingDate(rankingType), limit, offset]
        );
        
        // 处理排名序号
        const rankedList = rankings.map((item, index) => ({
            ...item,
            rank: offset + index + 1
        }));
        
        // 缓存5分钟
        await redis.setex(cacheKey, 300, JSON.stringify(rankedList));
        
        return rankedList;
    }

    /**
     * 更新排行榜
     */
    async updateRanking(rankingType) {
        const rankingDate = this.getRankingDate(rankingType);
        
        // 获取需要排名的作品数据
        const [contents] = await db.execute(
            `SELECT c.id as content_id,
                    c.content_type,
                    COALESCE(SUM(CASE WHEN i.interaction_type = 'like' THEN 1 ELSE 0 END), 0) as positive_votes,
                    COALESCE(COUNT(i.id), 0) as total_votes,
                    COALESCE(c.play_count, 0) as play_count,
                    COALESCE(c.share_count, 0) as share_count,
                    COALESCE(rd.star_power_boost, 0) as star_power_boost
             FROM contents c
             LEFT JOIN interactions i ON c.id = i.content_id
             LEFT JOIN ranking_data rd ON c.id = rd.content_id AND rd.ranking_type = ? AND rd.ranking_date = ?
             WHERE c.status = 'published'
               AND c.created_at >= ?
             GROUP BY c.id`,
            [rankingType, rankingDate, this.getRankingStartDate(rankingType)]
        );
        
        // 计算得分并排序
        const rankedContents = await WilsonScoreService.calculateRankings(contents, rankingType);
        
        // 批量更新数据库
        const connection = await db.getConnection();
        
        try {
            await connection.beginTransaction();
            
            // 清空旧数据
            await connection.execute(
                'DELETE FROM ranking_data WHERE ranking_type = ? AND ranking_date = ?',
                [rankingType, rankingDate]
            );
            
            // 插入新数据
            for (const content of rankedContents) {
                await connection.execute(
                    `INSERT INTO ranking_data 
                     (content_id, content_type, positive_votes, total_votes, play_count, 
                      share_count, star_power_boost, wilson_score, final_score, ranking_type, ranking_date) 
                     VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)`,
                    [
                        content.content_id,
                        content.content_type,
                        content.positive_votes,
                        content.total_votes,
                        content.play_count,
                        content.share_count,
                        content.star_power_boost,
                        content.wilson_score,
                        content.final_score,
                        rankingType,
                        rankingDate
                    ]
                );
            }
            
            await connection.commit();
            
            // 清除相关缓存
            const pattern = `rankings:${rankingType}:*`;
            const keys = await redis.keys(pattern);
            if (keys.length > 0) {
                await redis.del(...keys);
            }
            
            console.log(`排行榜 ${rankingType} 更新完成，共 ${rankedContents.length} 个作品`);
            
        } catch (error) {
            await connection.rollback();
            throw error;
        } finally {
            connection.release();
        }
    }

    /**
     * 获取作品排名
     */
    async getContentRanking(contentId, rankingType) {
        const [ranking] = await db.execute(
            `SELECT rd.*, 
                    (SELECT COUNT(*) + 1 
                     FROM ranking_data 
                     WHERE ranking_type = ? AND ranking_date = ? AND final_score > rd.final_score) as rank
             FROM ranking_data rd
             WHERE rd.content_id = ? AND rd.ranking_type = ? AND rd.ranking_date = ?`,
            [rankingType, this.getRankingDate(rankingType), contentId, rankingType, this.getRankingDate(rankingType)]
        );
        
        return ranking[0] || null;
    }

    /**
     * 获取排行榜日期
     */
    getRankingDate(rankingType) {
        const now = new Date();
        
        switch (rankingType) {
            case 'daily':
                return now.toISOString().split('T')[0];
            case 'weekly':
                // 返回本周一的日期
                const monday = new Date(now);
                monday.setDate(now.getDate() - now.getDay() + 1);
                return monday.toISOString().split('T')[0];
            case 'monthly':
                // 返回本月第一天的日期
                return `${now.getFullYear()}-${(now.getMonth() + 1).toString().padStart(2, '0')}-01`;
            default:
                return now.toISOString().split('T')[0];
        }
    }

    /**
     * 获取排行榜开始日期
     */
    getRankingStartDate(rankingType) {
        const now = new Date();
        
        switch (rankingType) {
            case 'daily':
                return new Date(now.getTime() - 24 * 60 * 60 * 1000);
            case 'weekly':
                return new Date(now.getTime() - 7 * 24 * 60 * 60 * 1000);
            case 'monthly':
                return new Date(now.getFullYear(), now.getMonth() - 1, now.getDate());
            default:
                return new Date(now.getTime() - 24 * 60 * 60 * 1000);
        }
    }

    /**
     * 获取排行榜元数据
     */
    async getRankingMetadata(rankingType) {
        const rankingDate = this.getRankingDate(rankingType);
        
        const [metadata] = await db.execute(
            `SELECT COUNT(*) as total_items,
                    MAX(final_score) as top_score,
                    AVG(final_score) as avg_score
             FROM ranking_data 
             WHERE ranking_type = ? AND ranking_date = ?`,
            [rankingType, rankingDate]
        );
        
        return metadata[0] || { total_items: 0, top_score: 0, avg_score: 0 };
    }

    /**
     * 定时更新所有排行榜
     */
    async updateAllRankings() {
        try {
            console.log('开始更新所有排行榜...');
            
            await this.updateRanking('daily');
            await this.updateRanking('weekly');
            await this.updateRanking('monthly');
            await this.updateRanking('potential');
            
            console.log('所有排行榜更新完成');
        } catch (error) {
            console.error('更新排行榜失败:', error);
        }
    }
}

module.exports = new RankingService();
```

```javascript
class MHeartService {
    /**
     * 计算M❤️值奖励
     */
    async calculateMHeartReward(userId, actionType, metadata = {}) {
        const rewardRules = {
            'track_play': {
                base: 1,
                maxDaily: 50,
                conditions: {
                    minDuration: 30, // 至少播放30秒
                    uniqueTrack: true
                }
            },
            'track_like': {
                base: 5,
                maxDaily: 100
            },
            'daily_checkin': {
                base: 10,
                bonus: {
                    consecutive: [20, 30, 50, 80, 100] // 连续签到奖励
                }
            },
            'ai_creation': {
                base: 20,
                qualityMultiplier: true
            },
            'content_share': {
                base: 8,
                viralBonus: true
            },
            'complete_profile': {
                base: 50,
                oneTime: true
            }
        };

        const rule = rewardRules[actionType];
        if (!rule) return 0;

        // 检查每日上限
        if (rule.maxDaily) {
            const todayEarned = await this.getTodayEarned(userId, actionType);
            if (todayEarned >= rule.maxDaily) return 0;
        }

        let reward = rule.base;

        // 特殊条件处理
        switch (actionType) {
            case 'daily_checkin':
                reward += await this.getConsecutiveBonus(userId, rule.bonus.consecutive);
                break;
            case 'ai_creation':
                if (rule.qualityMultiplier && metadata.qualityScore) {
                    reward *= metadata.qualityScore;
                }
                break;
            case 'track_play':
                if (metadata.duration < rule.conditions.minDuration) {
                    reward = 0;
                }
                break;
        }

        return Math.floor(reward);
    }

    /**
     * 更新用户M❤️值
     */
    async updateMHeartValue(userId, amount, actionType, metadata = {}) {
        const connection = await db.getConnection();
        
        try {
            await connection.beginTransaction();

            // 获取当前值
            const [user] = await connection.execute(
                'SELECT m_heart_value, m_heart_level FROM users WHERE id = ? FOR UPDATE',
                [userId]
            );

            if (!user[0]) throw new Error('用户不存在');

            const currentValue = user[0].m_heart_value;
            const currentLevel = user[0].m_heart_level;
            const newValue = currentValue + amount;

            // 更新用户数据
            const newLevel = Math.floor(newValue / 100) + 1;
            const levelChanged = newLevel > currentLevel;

            await connection.execute(
                `UPDATE users 
                 SET m_heart_value = ?, m_heart_level = ?
                 WHERE id = ?`,
                [newValue, newLevel, userId]
            );

            // 记录交易
            await connection.execute(
                `INSERT INTO star_transactions 
                 (user_id, transaction_type, amount, balance_after, related_type, metadata) 
                 VALUES (?, 'm_heart_earn', ?, ?, ?, ?)`,
                [
                    userId,
                    amount,
                    newValue,
                    actionType,
                    JSON.stringify(metadata)
                ]
            );

            // 处理升级奖励
            if (levelChanged) {
                await this.handleLevelUp(userId, newLevel, connection);
            }

            await connection.commit();

            return {
                newValue,
                newLevel,
                levelChanged,
                rewardAmount: amount
            };

        } catch (error) {
            await connection.rollback();
            throw error;
        } finally {
            connection.release();
        }
    }

    /**
     * 处理等级提升
     */
    async handleLevelUp(userId, newLevel, connection) {
        const levelRewards = {
            2: { star_power: 50, features: ['advanced_visualizer'] },
            3: { star_power: 100, features: ['ai_lyrics_generator'] },
            5: { star_power: 200, features: ['custom_themes'] },
            10: { star_power: 500, features: ['exclusive_content', 'priority_support'] }
        };

        const reward = levelRewards[newLevel];
        if (!reward) return;

        // 发放星力奖励
        if (reward.star_power) {
            await connection.execute(
                'UPDATE users SET star_power = star_power + ? WHERE id = ?',
                [reward.star_power, userId]
            );

            await connection.execute(
                `INSERT INTO star_transactions 
                 (user_id, transaction_type, amount, related_type, description) 
                 VALUES (?, 'star_power_earn', ?, 'level_up', ?)`,
                [userId, reward.star_power, `等级 ${newLevel} 奖励`]
            );
        }

        // 解锁新功能
        if (reward.features) {
            await connection.execute(
                'UPDATE users SET unlocked_features = JSON_ARRAY_APPEND(unlocked_features, "$", ?) WHERE id = ?',
                [reward.features, userId]
            );
        }

        // 记录成就
        await connection.execute(
            `INSERT INTO user_achievements (user_id, achievement_type, achievement_data) 
             VALUES (?, 'level_up', ?)`,
            [userId, JSON.stringify({ level: newLevel, rewards: reward })]
        );
    }
}
```

```javascript
    buildLyricsPrompt(theme, emotions, keywords, style, length, language) {
        const emotionText = emotions.length > 0 ? `情感基调：${emotions.join('、')}` : '';
        const keywordText = keywords.length > 0 ? `关键词：${keywords.join('、')}` : '';
        
        return `
请创作一首${style}风格的歌词，要求如下：

主题：${theme}

语言：${language}
长度：约${length}行

要求：
1. 富有诗意和画面感
2. 情感表达真挚自然
3. 结构清晰（建议：主歌-副歌-主歌-副歌-桥段-副歌）
4. 适当使用修辞手法
5. 朗朗上口，适合演唱

请直接输出歌词内容，不需要额外的说明文字。
        `.trim();

    /**
     * 解析歌词结构
     */
    parseLyricsStructure(lyricsText) {
        const lines = lyricsText.split('\n').filter(line => line.trim());
        const sections = [];
        let currentSection = { type: 'verse', lines: [] };
        
        lines.forEach(line => {
            const trimmedLine = line.trim();
            
            // 检测段落标记
            if (trimmedLine.includes('副歌') || trimmedLine.includes('Chorus')) {
                if (currentSection.lines.length > 0) {
                    sections.push(currentSection);
                }
                currentSection = { type: 'chorus', lines: [trimmedLine] };
            } else if (trimmedLine.includes('主歌') || trimmedLine.includes('Verse')) {
                if (currentSection.lines.length > 0) {
                    sections.push(currentSection);
                }
                currentSection = { type: 'verse', lines: [trimmedLine] };
            } else if (trimmedLine.includes('桥段') || trimmedLine.includes('Bridge')) {
                if (currentSection.lines.length > 0) {
                    sections.push(currentSection);
                }
                currentSection = { type: 'bridge', lines: [trimmedLine] };
            } else {
                currentSection.lines.push(trimmedLine);
            }
        });
        
        if (currentSection.lines.length > 0) {
            sections.push(currentSection);
        }
        
        return {
            rawText: lyricsText,
            sections: sections,
            totalLines: lines.length
        };
    }

        // 计算创新度

        return {
            emotions: emotionResult.emotions,
            dominantEmotion: emotionResult.dominantEmotion,
            valence: emotionResult.valence, // 情感价
            arousal: emotionResult.arousal, // 唤醒度

            innovationScore: innovationScore,

            style = 'pop',
            emotion = 'happy',

        } = params;
```

## 📊 排行榜服务 (backend/services/rankingService.js)
```javascript
const WilsonScoreService = require('./wilsonScore');
const db = require('../config/database');
const Redis = require('ioredis');
const redis = new Redis(process.env.REDIS_URL);

class RankingService {
    /**
     * 获取排行榜
     */
    async getRankings(rankingType, page = 1, limit = 50) {
        const cacheKey = `rankings:${rankingType}:${page}:${limit}`;
        
        // 尝试从缓存获取
        const cached = await redis.get(cacheKey);
        if (cached) {
            return JSON.parse(cached);
        }
        
        const offset = (page - 1) * limit;
        
        // 获取排行榜数据
        const [rankings] = await db.execute(
            `SELECT rd.*, 
                    c.title as content_title,
                    c.cover_url as content_cover,
                    u.username as creator_name,
                    u.avatar as creator_avatar
             FROM ranking_data rd
             LEFT JOIN contents c ON rd.content_id = c.id
             LEFT JOIN users u ON c.creator_id = u.id
             WHERE rd.ranking_type = ? AND rd.ranking_date = ?
             ORDER BY rd.final_score DESC
             LIMIT ? OFFSET ?`,
            [rankingType, this.getRankingDate(rankingType), limit, offset]
        );
        
        // 处理排名序号
        const rankedList = rankings.map((item, index) => ({
            ...item,
            rank: offset + index + 1
        }));
        
        // 缓存5分钟
        await redis.setex(cacheKey, 300, JSON.stringify(rankedList));
        
        return rankedList;
    }

    /**
     * 更新排行榜
     */
    async updateRanking(rankingType) {
        const rankingDate = this.getRankingDate(rankingType);
        
        // 获取需要排名的作品数据
        const [contents] = await db.execute(
            `SELECT c.id as content_id,
                    c.content_type,
                    COALESCE(SUM(CASE WHEN i.interaction_type = 'like' THEN 1 ELSE 0 END), 0) as positive_votes,
                    COALESCE(COUNT(i.id), 0) as total_votes,
                    COALESCE(c.play_count, 0) as play_count,
                    COALESCE(c.share_count, 0) as share_count,
                    COALESCE(rd.star_power_boost, 0) as star_power_boost
             FROM contents c
             LEFT JOIN interactions i ON c.id = i.content_id
             LEFT JOIN ranking_data rd ON c.id = rd.content_id AND rd.ranking_type = ? AND rd.ranking_date = ?
             WHERE c.status = 'published'
               AND c.created_at >= ?
             GROUP BY c.id`,
            [rankingType, rankingDate, this.getRankingStartDate(rankingType)]
        );
        
        // 计算得分并排序
        const rankedContents = await WilsonScoreService.calculateRankings(contents, rankingType);
        
        // 批量更新数据库
        const connection = await db.getConnection();
        
        try {
            await connection.beginTransaction();
            
            // 清空旧数据
            await connection.execute(
                'DELETE FROM ranking_data WHERE ranking_type = ? AND ranking_date = ?',
                [rankingType, rankingDate]
            );
            
            // 插入新数据
            for (const content of rankedContents) {
                await connection.execute(
                    `INSERT INTO ranking_data 
                     (content_id, content_type, positive_votes, total_votes, play_count, 
                      share_count, star_power_boost, wilson_score, final_score, ranking_type, ranking_date) 
                     VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)`,
                    [
                        content.content_id,
                        content.content_type,
                        content.positive_votes,
                        content.total_votes,
                        content.play_count,
                        content.share_count,
                        content.star_power_boost,
                        content.wilson_score,
                        content.final_score,
                        rankingType,
                        rankingDate
                    ]
                );
            }
            
            await connection.commit();
            
            // 清除相关缓存
            const pattern = `rankings:${rankingType}:*`;
            const keys = await redis.keys(pattern);
            if (keys.length > 0) {
                await redis.del(...keys);
            }
            
            console.log(`排行榜 ${rankingType} 更新完成，共 ${rankedContents.length} 个作品`);
            
        } catch (error) {
            await connection.rollback();
            throw error;
        } finally {
            connection.release();
        }
    }

    /**
     * 获取作品排名
     */
    async getContentRanking(contentId, rankingType) {
        const [ranking] = await db.execute(
            `SELECT rd.*, 
                    (SELECT COUNT(*) + 1 
                     FROM ranking_data 
                     WHERE ranking_type = ? AND ranking_date = ? AND final_score > rd.final_score) as rank
             FROM ranking_data rd
             WHERE rd.content_id = ? AND rd.ranking_type = ? AND rd.ranking_date = ?`,
            [rankingType, this.getRankingDate(rankingType), contentId, rankingType, this.getRankingDate(rankingType)]
        );
        
        return ranking[0] || null;
    }

    /**
     * 获取排行榜日期
     */
    getRankingDate(rankingType) {
        const now = new Date();
        
        switch (rankingType) {
            case 'daily':
                return now.toISOString().split('T')[0];
            case 'weekly':
                // 返回本周一的日期
                const monday = new Date(now);
                monday.setDate(now.getDate() - now.getDay() + 1);
                return monday.toISOString().split('T')[0];
            case 'monthly':
                // 返回本月第一天的日期
                return `${now.getFullYear()}-${(now.getMonth() + 1).toString().padStart(2, '0')}-01`;
            default:
                return now.toISOString().split('T')[0];
        }
    }

    /**
     * 获取排行榜开始日期
     */
    getRankingStartDate(rankingType) {
        const now = new Date();
        
        switch (rankingType) {
            case 'daily':
                return new Date(now.getTime() - 24 * 60 * 60 * 1000);
            case 'weekly':
                return new Date(now.getTime() - 7 * 24 * 60 * 60 * 1000);
            case 'monthly':
                return new Date(now.getFullYear(), now.getMonth() - 1, now.getDate());
            default:
                return new Date(now.getTime() - 24 * 60 * 60 * 1000);
        }
    }

    /**
     * 获取排行榜元数据
     */
    async getRankingMetadata(rankingType) {
        const rankingDate = this.getRankingDate(rankingType);
        
        const [metadata] = await db.execute(
            `SELECT COUNT(*) as total_items,
                    MAX(final_score) as top_score,
                    AVG(final_score) as avg_score
             FROM ranking_data 
             WHERE ranking_type = ? AND ranking_date = ?`,
            [rankingType, rankingDate]
        );
        
        return metadata[0] || { total_items: 0, top_score: 0, avg_score: 0 };
    }

    /**
     * 定时更新所有排行榜
     */
    async updateAllRankings() {
        try {
            console.log('开始更新所有排行榜...');
            
            await this.updateRanking('daily');
            await this.updateRanking('weekly');
            await this.updateRanking('monthly');
            await this.updateRanking('potential');
            
            console.log('所有排行榜更新完成');
        } catch (error) {
            console.error('更新排行榜失败:', error);
        }
    }
}

module.exports = new RankingService();
```
